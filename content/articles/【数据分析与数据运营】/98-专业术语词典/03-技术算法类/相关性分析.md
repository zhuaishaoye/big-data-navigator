# 📊 相关性分析 - 探索变量关系的统计方法

> [!info] 💡 **术语速览**
> **相关性分析** = 研究两个或多个变量之间关联强度和方向的统计方法，用于探索变量关系和特征选择。

## 💡 为什么相关性分析是数据科学的核心武器？

相关性分析是**数据挖掘和预测建模的基础工具**，几乎所有高级分析都建立在变量关系理解之上。

**核心价值**：
- 🔍 **变量筛选**：从海量特征中找出真正有意义的变量
- 🎯 **假设验证**：验证业务假设和理论模型的正确性
- ⚡ **快速洞察**：快速识别影响业务结果的关键因素
- 🛠️ **模型优化**：为机器学习模型提供特征工程指导

无论是分析用户活跃度与购买行为的关系，还是探索广告投入与转化效果的关联，相关性分析都是数据分析师不可缺少的分析技能。
## 💡 核心概念解释

> [!abstract] 🎯 什么是相关性分析？
> 相关性分析是通过计算相关系数来衡量两个或多个变量之间关联强弱和方向的统计方法。

### 相关系数类型

| 相关系数 | 适用数据类型 | 取值范围 | 应用场景 |
|---|---|---|---|
| Pearson相关系数 | 连续数值型 | [-1, 1] | 线性关系分析 |
| Spearman相关系数 | 有序分类或可排序数值型 | [-1, 1] | 单调关系分析 |
| Kendall相关系数 | 有序分类或可排序数值型 | [-1, 1] | 秩一致性分析 |

### 相关强度判断标准

> [!note] 📏 相关强度参考标准
> - **|r| > 0.8**：强相关
> - **0.5 < |r| ≤ 0.8**：中等相关
> - **0.3 < |r| ≤ 0.5**：弱相关
> - **|r| ≤ 0.3**：几乎无相关

### 基础相关性计算

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from scipy import stats

# 创建示例数据
np.random.seed(42)
n = 1000

# 模拟电商用户数据
data = {
    'user_age': np.random.normal(30, 8, n),
    'session_duration': np.random.normal(15, 5, n),
    'page_views': np.random.poisson(8, n),
    'purchase_amount': np.random.gamma(2, 50, n),
    'app_rating': np.random.choice([1,2,3,4,5], n, p=[0.05,0.1,0.2,0.4,0.25])
}

# 添加一些相关性
data['purchase_amount'] = data['purchase_amount'] + data['session_duration'] * 2 + np.random.normal(0, 20, n)
data['page_views'] = data['page_views'] + data['session_duration'] * 0.3 + np.random.normal(0, 2, n)

df = pd.DataFrame(data)

print("📊 数据基本信息")
print(df.describe())
```

```python
# 计算Pearson相关系数
def calculate_pearson_correlation(df):
    """计算Pearson相关系数矩阵"""
    correlation_matrix = df.corr(method='pearson')
    return correlation_matrix

# 执行分析
corr_matrix = calculate_pearson_correlation(df)
print("\n📈 Pearson相关系数矩阵：")
print(corr_matrix.round(3))

# 可视化相关性热力图
plt.figure(figsize=(10, 8))
sns.heatmap(corr_matrix, 
            annot=True,

            square=True,
            fmt='.3f')
plt.title('变量间Pearson相关性热力图', fontsize=16)
```

### 相关性显著性检验

```python
def correlation_significance_test(x, y, alpha=0.05):
    """相关性显著性检验"""
 # 计算Pearson相关系数和p值
    r, p_value = stats.pearsonr(x, y)
    
 # 判断显著性
    is_significant = p_value < alpha
    
 # 相关强度判断
    if abs(r) > 0.8:
        strength = "强相关"
    elif abs(r) > 0.5:
        strength = "中等相关"
    elif abs(r) > 0.3:
        strength = "弱相关"
    else:
        strength = "几乎无相关"
    
    result = {
        'correlation': r,
        'p_value': p_value,
        'is_significant': is_significant,
        'strength': strength,
        'sample_size': len(x)
    }
    
    return result

# 分析具体变量对
print("\n🔍 相关性显著性检验结果：")
variables_pairs = [
    ('session_duration', 'purchase_amount'),
    ('session_duration', 'page_views'),
    ('user_age', 'purchase_amount'),
    ('page_views', 'purchase_amount')
]

for var1, var2 in variables_pairs:
    result = correlation_significance_test(df[var1], df[var2])
    print(f"\n{var1} vs {var2}:")
    print(f"  相关系数: {result['correlation']:.3f}")
    print(f"  P值: {result['p_value']:.6f}")
    print(f"  是否显著: {'是' if result['is_significant'] else '否'}")
    print(f"  相关强度: {result['strength']}")
```

## 🚀 实战案例：用户行为相关性分析

### 案例1：电商用户付费行为分析

```python
def ecommerce_correlation_analysis():
    """电商用户付费行为相关性分析"""
    
 # 生成更真实的电商数据
    np.random.seed(123)
    n = 2000
    
 # 基础用户特征
    user_data = {
        'user_id': range(1, n+1),
        'age': np.random.normal(32, 10, n),
        'registration_days': np.random.exponential(365, n),
        'app_version': np.random.choice(['v1.0', 'v1.1', 'v1.2'], n),
        'device_type': np.random.choice(['iOS', 'Android'], n)
    }
    
 # 行为数据（添加相关性）
    base_activity = np.random.gamma(2, 3, n)
    user_data['daily_sessions'] = base_activity + np.random.normal(0, 1, n)
    user_data['avg_session_duration'] = base_activity * 2 + np.random.normal(10, 3, n)
    user_data['page_views_per_session'] = base_activity * 1.5 + np.random.normal(5, 2, n)
    
 # 商业数据（强相关性）
    engagement_score = (user_data['daily_sessions'] * 0.3 + 
                       user_data['avg_session_duration'] * 0.1 + 
                       user_data['page_views_per_session'] * 0.2)
    
    user_data['monthly_purchase_frequency'] = np.maximum(0, engagement_score + np.random.normal(0, 1, n))
    user_data['average_order_value'] = np.maximum(0, engagement_score * 15 + np.random.normal(50, 20, n))
    user_data['total_spend'] = user_data['monthly_purchase_frequency'] * user_data['average_order_value']
    
    df_ecommerce = pd.DataFrame(user_data)
    
 # 相关性分析
    numeric_cols = ['age', 'registration_days', 'daily_sessions', 
                   'avg_session_duration', 'page_views_per_session',
                   'monthly_purchase_frequency', 'average_order_value', 'total_spend']
    
    corr_matrix = df_ecommerce[numeric_cols].corr()
    
 # 可视化
    plt.figure(figsize=(12, 10))
    mask = np.triu(np.ones_like(corr_matrix, dtype=bool))
    sns.heatmap(corr_matrix, 
                mask=mask,
                annot=True, 
                cmap='RdYlBu_r', 
                center=0,
                square=True,
                fmt='.2f',
                cbar_kws={"shrink": .8})
    plt.title('电商用户行为与付费关联性分析', fontsize=16)
    plt.tight_layout()
    plt.show()
    
 # 关键发现
    print("🔍 关键相关性发现：")
    target_var = 'total_spend'
    correlations = corr_matrix[target_var].sort_values(ascending=False)
    
    for var, corr in correlations.items():
        if var != target_var and abs(corr) > 0.3:
            print(f"  {var}: {corr:.3f} ({get_correlation_strength(corr)})")
    
    return df_ecommerce, corr_matrix

def get_correlation_strength(r):
    """判断相关强度"""
    if abs(r) > 0.8:
        return "强相关"
    elif abs(r) > 0.5:
        return "中等相关"
    elif abs(r) > 0.3:
        return "弱相关"
    else:
        return "几乎无相关"

# 执行分析
df_ec, corr_ec = ecommerce_correlation_analysis()
```

### 案例2：广告投放效果相关性

```python
def advertising_correlation_analysis():
    """广告投放效果相关性分析"""
    
 # 模拟广告数据
    np.random.seed(456)
    days = 90
    
    ad_data = {
        'date': pd.date_range('2024-01-01', periods=days),
        'ad_spend': np.random.gamma(3, 1000, days),
        'impressions': np.random.poisson(50000, days),
        'clicks': np.random.poisson(2000, days),
        'conversions': np.random.poisson(100, days)
    }
    
 # 添加相关性
    ad_data['clicks'] = ad_data['impressions'] * 0.04 + np.random.normal(0, 200, days)
    ad_data['conversions'] = ad_data['clicks'] * 0.05 + np.random.normal(0, 10, days)
    ad_data['revenue'] = ad_data['conversions'] * 50 + np.random.normal(0, 500, days)
    
 # 计算衍生指标
    ad_data['ctr'] = ad_data['clicks'] / ad_data['impressions']
    ad_data['cvr'] = ad_data['conversions'] / ad_data['clicks']
    ad_data['cpc'] = ad_data['ad_spend'] / ad_data['clicks']
    ad_data['roas'] = ad_data['revenue'] / ad_data['ad_spend']
    
    df_ad = pd.DataFrame(ad_data)
    
 # 相关性分析
    analysis_cols = ['ad_spend', 'impressions', 'clicks', 'conversions', 
                    'revenue', 'ctr', 'cvr', 'cpc', 'roas']
    
    corr_matrix = df_ad[analysis_cols].corr()
    
 # 重点关注ROI相关性
    roi_correlations = corr_matrix['roas'].sort_values(ascending=False)
    
    print("\n💰 ROAS（广告投资回报率）相关性分析：")
    for metric, corr in roi_correlations.items():
        if metric != 'roas':
            strength = get_correlation_strength(corr)
            print(f"  {metric}: {corr:.3f} ({strength})")
    
 # 绘制关键指标散点图
    fig, axes = plt.subplots(2, 2, figsize=(15, 12))
    
 # CVR vs ROAS
    axes[0,0].scatter(df_ad['cvr'], df_ad['roas'], alpha=0.6)
    axes[0,0].set_xlabel('转化率 (CVR)')
    axes[0,0].set_ylabel('广告投资回报率 (ROAS)')
    axes[0,0].set_title(f'CVR vs ROAS (r={corr_matrix.loc["cvr", "roas"]:.3f})')
    
 # CPC vs ROAS
    axes[0,1].scatter(df_ad['cpc'], df_ad['roas'], alpha=0.6, color='orange')
    axes[0,1].set_xlabel('每次点击成本 (CPC)')
    axes[0,1].set_ylabel('广告投资回报率 (ROAS)')
    axes[0,1].set_title(f'CPC vs ROAS (r={corr_matrix.loc["cpc", "roas"]:.3f})')
    
 # CTR vs ROAS
    axes[1,0].scatter(df_ad['ctr'], df_ad['roas'], alpha=0.6, color='green')
    axes[1,0].set_xlabel('点击率 (CTR)')
    axes[1,0].set_ylabel('广告投资回报率 (ROAS)')
    axes[1,0].set_title(f'CTR vs ROAS (r={corr_matrix.loc["ctr", "roas"]:.3f})')
    
 # Ad Spend vs Revenue
    axes[1,1].scatter(df_ad['ad_spend'], df_ad['revenue'], alpha=0.6, color='red')
    axes[1,1].set_xlabel('广告支出')
    axes[1,1].set_ylabel('收入')
    axes[1,1].set_title(f'广告支出 vs 收入 (r={corr_matrix.loc["ad_spend", "revenue"]:.3f})')
    
    plt.tight_layout()
    plt.show()
    
    return df_ad, corr_matrix

# 执行广告分析
```

## ⚠️ 注意事项与最佳实践

### 常见误区

> [!warning] 🚨 相关性 ≠ 因果性
> - 相关性只表示变量间的关联强度
> - 不能说明因果关系方向
> - 可能存在第三变量影响

### 分析要点

1. **数据质量检查**
   - 检查异常值和缺失值
   - 确保数据分布合理
   - 注意非线性关系

2. **选择合适的相关系数**
   - 数值变量的线性关联：Pearson
   - 有序分类或数值变量的单调关联：Spearman
   - 有序分类或数值变量的秩一致性：Kendall

3. **解释相关性结果**
   - 考虑业务背景
   - 验证统计显著性
   - 探索潜在机制

### 实际应用建议

```python
def correlation_analysis_best_practices(df, target_var):
    """相关性分析最佳实践"""
    
 # 1. 数据预处理
    print("📋 数据质量检查：")
    print(f"缺失值: {df.isnull().sum().sum()}")
    print(f"数据类型: {df.dtypes.value_counts()}")
    
 # 2. 异常值检测
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    Q1 = df[numeric_cols].quantile(0.25)
    Q3 = df[numeric_cols].quantile(0.75)
    IQR = Q3 - Q1
    outliers = ((df[numeric_cols] < (Q1 - 1.5 * IQR)) | 
                (df[numeric_cols] > (Q3 + 1.5 * IQR))).sum()
    print(f"\n📊 异常值统计:\n{outliers[outliers > 0]}")
    
 # 3. 相关性分析
    correlations = df[numeric_cols].corr()[target_var].sort_values(ascending=False)
    
    print(f"\n🎯 与{target_var}的相关性排序：")
    for var, corr in correlations.items():
        if var != target_var and abs(corr) > 0.1:
            significance = correlation_significance_test(df[var], df[target_var])
            print(f"  {var}: {corr:.3f} (p={significance['p_value']:.4f})")
    
 # 4. 可视化建议
    strong_correlations = correlations[abs(correlations) > 0.5].drop(target_var)
    if len(strong_correlations) > 0:
        print(f"\n💡 建议深入分析的强相关变量:")
        for var in strong_correlations.index:
            print(f"  - {var}")
    
    return correlations
```
