class PreAttentiveVisualizer:

class GestaltVisualPrinciples:
    """格式塔视觉原理的可视化实现"""
    
    @staticmethod
    def proximity_grouping(ax, data_groups):
        """邻近性原理：通过空间距离创建视觉分组"""
        x_offset = 0
        group_spacing = 2  # 组间距离
        bar_spacing = 0.8  # 组内距离
        
        for group_name, group_data in data_groups.items():
            positions = np.arange(len(group_data)) * bar_spacing + x_offset
            ax.bar(positions, group_data, label=group_name, width=0.6)
            x_offset += len(group_data) * bar_spacing + group_spacing
    
    @staticmethod
    def similarity_encoding(ax, data, categories):
        """相似性原理：通过一致的视觉属性表示相同类别"""
        color_map = {
            'category_a': '#1f77b4',
            'category_b': '#ff7f0e', 
            'category_c': '#2ca02c'
        }
        
        for i, (value, category) in enumerate(zip(data, categories)):
            ax.bar(i, value, color=color_map[category], 
                  alpha=0.8, edgecolor='white', linewidth=1)

class VisualEncodingEfficiency:

    def __init__(self):

            2: 'position_nonaligned_scale', # 位置（非对齐刻度）

            5: 'area',                      # 面积

### 3. 数据诚实性与统计完整性

**统计可视化的伦理框架**

> [!warning] ⚠️ 数据可视化伦理准则
> 可视化设计必须遵循统计伦理，避免数据误导和认知偏差

```python
class DataIntegrityValidator:
    """数据完整性与可视化伦理验证器"""
    
    def __init__(self):
        self.integrity_checks = {
            'scale_manipulation': self._check_scale_integrity,
            'aspect_ratio': self._check_aspect_ratio,
            'missing_data': self._check_missing_data_handling,
            'statistical_context': self._check_statistical_context,
            'uncertainty_representation': self._check_uncertainty
        }
    
    def _check_scale_integrity(self, fig, ax):
        """检查坐标轴比例完整性"""
        y_min, y_max = ax.get_ylim()
        data_min, data_max = ax.get_children()[0].get_data()[1].min(), ax.get_children()[0].get_data()[1].max()
        
        # 检查零基线原则
        if data_min >= 0 and y_min > 0:
            return {
                'status': 'warning',
                'message': 'Y轴未从零开始，可能夸大数据差异',
                'recommendation': '除非有特殊说明，建议从零开始'
            }
        
        # 检查截断比例
        if (data_max - data_min) / (y_max - y_min) < 0.3:
            return {
                'status': 'error',
                'message': '数据范围占显示范围比例过小，存在误导风险'
            }
        
        return {'status': 'ok'}
    
    def _check_uncertainty_representation(self, data, visualization_type):
        """检查不确定性表示"""
        if 'std' in data.columns or 'error' in data.columns:
            if visualization_type not in ['errorbar', 'confidence_interval']:
                return {
                    'status': 'warning',
                    'message': '数据包含不确定性信息但未在可视化中体现',
                    'recommendation': '添加误差棒或置信区间'
                }
        return {'status': 'ok'}
    
    def create_honest_visualization(self, data, show_uncertainty=True):
        """创建诚实的数据可视化"""
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15, 6))
        
        # 误导性可视化示例（仅用于对比）
        ax1.bar(data.index, data['value'])
        ax1.set_ylim(data['value'].min() * 0.95, data['value'].max() * 1.05)
        ax1.set_title('误导性：截断Y轴夸大差异', color='red')
        
        # 诚实的可视化
        bars = ax2.bar(data.index, data['value'])
        ax2.set_ylim(0, data['value'].max() * 1.1)
        
        # 添加不确定性表示
        if show_uncertainty and 'std' in data.columns:
            ax2.errorbar(data.index, data['value'], 
                        yerr=data['std'], fmt='none', 
                        color='black', capsize=3)
        
        # 添加统计信息
        mean_val = data['value'].mean()
        ax2.axhline(y=mean_val, color='red', linestyle='--', 
                   alpha=0.7, label=f'平均值: {mean_val:.2f}')
        
        ax2.set_title('诚实性：完整比例 + 不确定性', color='green')
        ax2.legend()
        
        return fig
```

class DataQualityAnnotation:
    """数据质量标注与元数据管理"""
    
    def __init__(self):
        self.quality_metrics = {
            'completeness': 0.0,      # 数据完整性
            'accuracy': 0.0,          # 数据准确性  
            'consistency': 0.0,       # 数据一致性
            'timeliness': 0.0,        # 数据时效性
            'validity': 0.0           # 数据有效性
        }
    
    def calculate_quality_score(self, data):
        """计算数据质量综合得分"""
        # 完整性：非空值比例
        completeness = data.notna().sum().sum() / (len(data) * len(data.columns))
        
        # 一致性：数据类型一致性检查
        consistency = self._check_data_consistency(data)
        
        # 有效性：业务规则验证
        validity = self._check_business_rules(data)
        
        return {
            'completeness': completeness,
            'consistency': consistency,
            'validity': validity,
            'overall_score': (completeness + consistency + validity) / 3
        }
    
    def add_quality_annotation(self, ax, quality_score):
        """在可视化中添加数据质量标注"""
        quality_text = f"数据质量评分: {quality_score['overall_score']:.2f}"
        
        # 根据质量分数选择颜色
        if quality_score['overall_score'] > 0.9:
            color = 'green'
        elif quality_score['overall_score'] > 0.7:
            color = 'orange'  
        else:
            color = 'red'
        
        ax.text(0.02, 0.98, quality_text, transform=ax.transAxes,
               verticalalignment='top', bbox=dict(boxstyle='round', 
               facecolor=color, alpha=0.3))

    def create_color_palette(self, n_colors, palette_type='qualitative'):
        """生成企业品牌色彩调色板"""
        if palette_type == 'qualitative':
            # 定性数据调色板
            base_colors = [self.brand_colors['primary'], self.brand_colors['secondary']]
            if n_colors <= 2:
                return base_colors[:n_colors]
            else:
                # 生成更多颜色变体
                import matplotlib.colors as mcolors
                colors = []
                for i in range(n_colors):
                    hue_shift = i * (360 / n_colors)
                    colors.append(self._shift_hue(self.brand_colors['primary'], hue_shift))

            return self._generate_sequential_palette(self.brand_colors['primary'], n_colors)
        
        elif palette_type == 'diverging':

            return self._generate_diverging_palette(n_colors)
    
    def _generate_sequential_palette(self, base_color, n_colors):
        """生成顺序调色板"""
        import matplotlib.colors as mcolors
        
        # 从浅到深的渐变
        colors = []
        for i in range(n_colors):
            alpha = 0.3 + (0.7 * i / (n_colors - 1))
            colors.append(mcolors.to_rgba(base_color, alpha))
        return colors

## ❌ 常见错误与陷阱

### 1. 图表类型选择错误

**错误示例：**
- 用饼图显示超过7个类别
- 需要突出连续时间趋势时，仅用条形图可能不利于观察趋势；比较离散时点的数值时可使用条形图
- 用线图显示分类数据

**正确选择：**
```python
# 数据特征 -> 推荐图表类型
分类数据 -> 条形图、饼图（≤7类）
时间序列 -> 线图、面积图
相关性 -> 散点图、热力图
分布 -> 直方图、箱线图
对比 -> 条形图、雷达图
```

## 🚫 避免的可视化陷阱

### 1. 数据误导陷阱

**截断坐标轴：**
```python
# 以条形长度比较非负数值时，截断零基线可能夸大差异
plt.ylim(80, 100)  # 不适合直接用于此类条形长度比较

# 数据在0到100范围内的非负条形图示例
plt.ylim(0, 100)   # 保留零基线；负值、超范围值及误差棒需另行确定轴范围

# 折线图等趋势图可依据分析任务选择非零轴范围，并明确标示
```

**使用3D效果：**
```python
# 避免使用3D饼图
# 3D效果会扭曲角度和面积比例
# 推荐使用2D饼图或环形图
```

### 2. 视觉噪音陷阱

**过度装饰：**
- 不必要的阴影效果
- 花哨的边框和背景
- 过多的网格线
- 装饰性图标

**信息过载：**
- 同时显示过多数据系列
- 标签过于密集
- 颜色过多且无规律

### 3. 认知负荷陷阱

**复杂图表：**
- 同时使用多种图表类型
- 嵌套过多的子图
- 交互功能过于复杂

**不直观的编码：**
- 使用不常见的视觉编码
- 颜色含义不明确
- 符号含义模糊
