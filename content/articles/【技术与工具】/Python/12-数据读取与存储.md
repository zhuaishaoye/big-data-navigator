```python
print("=== 高性能CSV文件处理系统 ===")

    def demonstrate_streaming(self, file_path: str, chunk_size: int = 10000):
        """演示流式处理大文件"""
        print(f"\n=== 流式处理演示 (chunk_size: {chunk_size:,}) ===")
        
        total_rows = 0
        chunk_count = 0
        aggregated_stats = {}
        
        try:
            start_time = time.time()
            
            # 使用生成器进行流式处理
            for chunk in pd.read_csv(file_path, chunksize=chunk_size):
                chunk_count += 1
                total_rows += len(chunk)
                
                # 对每个chunk进行统计
                chunk_stats = {
                    'mean_value': chunk['value'].mean() if 'value' in chunk else 0,
                    'category_counts': chunk['category'].value_counts().to_dict() if 'category' in chunk else {},
                    'missing_count': chunk.isnull().sum().sum()
                }
                
                # 累积统计
                if not aggregated_stats:
                    aggregated_stats = chunk_stats
                else:
                    # 合并统计信息
                    aggregated_stats['mean_value'] = (
                        (aggregated_stats['mean_value'] * (chunk_count - 1) + chunk_stats['mean_value']) / chunk_count
                    )
                    
                    for cat, count in chunk_stats['category_counts'].items():
                        aggregated_stats['category_counts'][cat] = (
                            aggregated_stats['category_counts'].get(cat, 0) + count
                        )
                
                if chunk_count % 10 == 0:
                    print(f"已处理 {chunk_count} 个chunk, {total_rows:,} 行数据")
            
            processing_time = time.time() - start_time
            
            print(f"\n流式处理完成:")
            print(f"- 总chunk数: {chunk_count}")
            print(f"- 总行数: {total_rows:,}")
            print(f"- 处理时间: {processing_time:.3f}s")
            print(f"- 处理速度: {total_rows/processing_time:,.0f} 行/秒")
            print(f"- 平均值: {aggregated_stats['mean_value']:.2f}")
            print(f"- 类别分布: {aggregated_stats['category_counts']}")
            
        except Exception as e:
            print(f"流式处理错误: {e}")
```

### 结构化文件处理：Excel与复杂格式

```python
from openpyxl import Workbook, load_workbook
from openpyxl.styles import Font, PatternFill, Alignment
from openpyxl.utils.dataframe import dataframe_to_rows
from openpyxl.chart import LineChart, Reference
import xlsxwriter
from typing import List, Dict

print("=== 高级Excel文件处理系统 ===")

class ExcelProcessor:
    """Excel文件高级处理器
    
    功能特性:
    - 多引擎支持 (openpyxl, xlsxwriter, xlrd)
    - 样式和格式化控制
    - 图表和可视化集成
    - 大文件优化处理
    - 数据验证和完整性检查
    """
    
    def __init__(self):
        self.supported_engines = ['openpyxl', 'xlsxwriter']
        self.max_excel_rows = 1048576  # Excel行限制
        self.max_excel_cols = 16384    # Excel列限制
    
    def create_advanced_excel(self, data_dict: Dict[str, pd.DataFrame], filename: str):
        """创建高级Excel文件，包含样式和图表"""
        print(f"\n创建高级Excel文件: {filename}")
        
        with pd.ExcelWriter(filename, engine='xlsxwriter') as writer:
            workbook = writer.book
            
            # 定义样式
            header_format = workbook.add_format({
                'bold': True,
                'text_wrap': True,
                'valign': 'top',
                'fg_color': '#4472C4',
                'font_color': 'white',
                'border': 1
            })
            
            number_format = workbook.add_format({'num_format': '#,##0.00'})
            percent_format = workbook.add_format({'num_format': '0.00%'})
            date_format = workbook.add_format({'num_format': 'yyyy-mm-dd'})
            
            for sheet_name, df in data_dict.items():
                # 检查数据大小限制
                if len(df) > self.max_excel_rows - 1:
                    print(f"警告: {sheet_name} 数据行数 ({len(df)}) 超过扣除表头后的Excel限制，将截断")
                    df = df.head(self.max_excel_rows - 1)
                
                # 写入数据
                df.to_excel(writer, sheet_name=sheet_name, index=False, header=False, startrow=1)
                worksheet = writer.sheets[sheet_name]
                
                # 应用表头样式
                for col_num, value in enumerate(df.columns.values):
                    worksheet.write(0, col_num, value, header_format)
                
                # 自动调整列宽
                for idx, col in enumerate(df.columns):
                    max_length = max(
                        df[col].astype(str).str.len().max(),
                        len(str(col))
                    )
                    worksheet.set_column(idx, idx, min(max_length + 2, 50))
                
                # 应用数字格式
                for idx, col in enumerate(df.columns):
                    if df[col].dtype in ['float64', 'int64']:
                        if col.lower().find('percent') != -1 or col.lower().find('rate') != -1:
                            worksheet.set_column(idx, idx, None, percent_format)
                        else:
                            worksheet.set_column(idx, idx, None, number_format)
                    elif df[col].dtype == 'datetime64[ns]':
                        worksheet.set_column(idx, idx, None, date_format)
                
                # 添加数据筛选
                worksheet.autofilter(0, 0, len(df), len(df.columns) - 1)
                
                # 为数值数据创建图表
                numeric_cols = df.select_dtypes(include=[np.number]).columns
                if len(numeric_cols) > 0 and len(df) > 1:
                    chart = workbook.add_chart({'type': 'line'})
                    
                    for col_idx, col_name in enumerate(numeric_cols[:3]):  # 最多3个系列
                        chart.add_series({
                            'name': col_name,
                            'categories': [sheet_name, 1, 0, len(df), 0],
                            'values': [sheet_name, 1, df.columns.get_loc(col_name), 
                                     len(df), df.columns.get_loc(col_name)],
                        })
                    
                    chart.set_title({'name': f'{sheet_name} 趋势图'})
                    chart.set_x_axis({'name': '数据点'})
                    chart.set_y_axis({'name': '数值'})
                    
                    # 插入图表
                    worksheet.insert_chart('H2', chart)
        
        print(f"✅ 高级Excel文件创建完成: {filename}")
        return filename
    
    def read_excel_optimized(self, filename: str, **kwargs) -> Dict[str, pd.DataFrame]:
        """优化的Excel读取方法"""
        print(f"\n优化读取Excel文件: {filename}")
        
        start_time = time.time()
        
        # 首先获取工作表信息
        excel_file = pd.ExcelFile(filename)
        sheet_names = excel_file.sheet_names
        
        print(f"发现 {len(sheet_names)} 个工作表: {sheet_names}")
        
        dataframes = {}
        
        for sheet_name in sheet_names:
            try:
                # 使用优化参数读取
                df = pd.read_excel(
                    filename,
                    sheet_name=sheet_name,
                    engine='openpyxl',  # 推荐引擎
                    **kwargs
                )
                
                dataframes[sheet_name] = df
                print(f"- {sheet_name}: {df.shape[0]} 行 × {df.shape[1]} 列")
                
            except Exception as e:
                print(f"❌ 读取工作表 {sheet_name} 失败: {e}")
        
        read_time = time.time() - start_time
        print(f"总读取时间: {read_time:.3f}s")
        
        return dataframes
    
    def process_large_excel(self, filename: str, max_rows: int = 100000):
        """处理大型Excel文件的策略"""
        print(f"\n处理大型Excel文件: {filename}")
        
        try:
            # 分块读取策略
            chunk_size = 10000
            all_chunks = []
            
            # 使用openpyxl直接操作
            wb = load_workbook(filename, read_only=True)
            
            for sheet_name in wb.sheetnames:
                ws = wb[sheet_name]
                print(f"处理工作表: {sheet_name}")
                
                # 获取数据范围
                max_row = ws.max_row
                max_col = ws.max_column
                
                print(f"- 数据范围: {max_row} 行 × {max_col} 列")
                
                if max_row > max_rows:
                    print(f"- 文件较大，使用分块处理策略")
                    
                    # 分块处理
                    for start_row in range(2, max_row + 1, chunk_size):  # 跳过标题行
                        end_row = min(start_row + chunk_size - 1, max_row)
                        
                        # 读取数据块
                        chunk_data = []
                        for row in ws.iter_rows(min_row=start_row, max_row=end_row, 
                                              values_only=True):
                            chunk_data.append(row)
                        
                        if chunk_data:
                            # 获取列名
                            headers = [cell.value for cell in ws[1]]
                            chunk_df = pd.DataFrame(chunk_data, columns=headers)
                            
                            # 进行必要的处理
                            processed_chunk = self._process_chunk(chunk_df)
                            all_chunks.append(processed_chunk)
                        
                        print(f"  已处理行 {start_row}-{end_row}")
                
                else:
                    # 直接读取小文件
                    df = pd.read_excel(filename, sheet_name=sheet_name)
                    all_chunks.append(self._process_chunk(df))
            
            wb.close()
            
            # 合并所有块
            if all_chunks:
                final_df = pd.concat(all_chunks, ignore_index=True)
                print(f"\n合并完成: {len(final_df)} 行数据")
                return final_df
            
        except Exception as e:
            print(f"❌ 处理大文件失败: {e}")
            return None
    
    def _process_chunk(self, chunk: pd.DataFrame) -> pd.DataFrame:
        """处理数据块的通用方法"""
        # 基本数据清理
        chunk = chunk.dropna(how='all')  # 删除全空行
        
        # 数据类型优化
        for col in chunk.select_dtypes(include=['object']):
            if chunk[col].nunique() / len(chunk) < 0.5:  # 如果唯一值比例小于50%
                chunk[col] = chunk[col].astype('category')
        
        return chunk
```

```python
class DatabaseManager:

        self._setup_engine()
    
    def _setup_engine(self):
        """配置数据库引擎"""
        print(f"\n初始化数据库连接: {self.db_url}")
        
        # SQLite连接配置
        if 'sqlite' in self.db_url:
            self.engine = create_engine(
                self.db_url,
                poolclass=StaticPool,
                pool_pre_ping=True,
                connect_args={
                    'check_same_thread': False,
                    'timeout': 30,
                    'isolation_level': None  # 启用autocommit模式
                },
                echo=False  # 设置为True可看到SQL语句
            )
        else:
            # 其他数据库连接配置
            self.engine = create_engine(
                self.db_url,
                pool_size=10,
                max_overflow=20,
                pool_pre_ping=True,
                pool_recycle=3600
            )
        
        print(f"✅ 数据库引擎配置完成")
    
    @contextmanager
    def get_connection(self):
        """获取数据库连接的上下文管理器"""
        conn = self.engine.connect()
        try:
            yield conn
        finally:
            conn.close()
    
    @contextmanager
    def transaction(self):
        """事务上下文管理器"""
        with self.get_connection() as conn:
            trans = conn.begin()
            try:
                yield conn
                trans.commit()
            except Exception:
                trans.rollback()
                raise
```

```python
print("=== 并发文件访问控制系统 ===")

class ConcurrentFileManager:
    """并发文件访问管理器
    
    功能特性:
    - 文件锁定机制
    - 多进程文件访问示例，当前锁文件生命周期存在竞争风险
    - 原地覆盖写入，不保证原子替换
    - 尚未实现锁超时及死锁检测
    - 共享读锁与独占写锁的使用示例，写操作不应使用共享锁
    """
    
    def __init__(self, base_dir: str = None):
        self.base_dir = Path(base_dir) if base_dir else Path(tempfile.mkdtemp())
        self.lock_timeout = 30  # 锁超时时间（秒）
        self._ensure_directory()
    
    def _ensure_directory(self):
        """确保目录存在"""
        self.base_dir.mkdir(parents=True, exist_ok=True)
        print(f"工作目录: {self.base_dir}")
    
    def write_with_lock(self, filename: str, data: str, exclusive: bool = True) -> bool:
        """使用文件锁的写操作"""
        file_path = self.base_dir / filename
        lock_path = self.base_dir / f"{filename}.lock"
        
        try:
            with open(lock_path, 'w') as lock_file:
                # 获取独占锁
                if exclusive:
                    portalocker.lock(lock_file, portalocker.LOCK_EX)
                else:
                    portalocker.lock(lock_file, portalocker.LOCK_SH)
                
                print(f"[{threading.current_thread().name}] 获取锁: {filename}")
                
                # 原地覆盖写入；这不是临时文件加原子替换
                with open(file_path, 'w') as f:
                    f.write(data)
                    f.flush()  # 将Python缓冲区的数据提交给底层文件对象；后续调用fsync请求同步
                    os.fsync(f.fileno())
                
                # 模拟一些处理时间
                time.sleep(0.1)
                
                print(f"[{threading.current_thread().name}] 写入完成: {filename}")
                return True
                
        except Exception as e:
            print(f"❌ 写入失败 {filename}: {e}")
            return False
        finally:
            # 锁会在with语句结束时自动释放
            if lock_path.exists():
                lock_path.unlink()
    
    def read_with_lock(self, filename: str) -> Optional[str]:
        """使用文件锁的读操作"""
        file_path = self.base_dir / filename
        lock_path = self.base_dir / f"{filename}.lock"
        
        if not file_path.exists():
            return None
        
        try:
            with open(lock_path, 'w') as lock_file:
                # 获取共享锁
                portalocker.lock(lock_file, portalocker.LOCK_SH)
                
                print(f"[{threading.current_thread().name}] 获取读锁: {filename}")
                
                with open(file_path, 'r') as f:
                    content = f.read()
                
                print(f"[{threading.current_thread().name}] 读取完成: {filename}")
                return content
                
        except Exception as e:
            print(f"❌ 读取失败 {filename}: {e}")
            return None
        finally:
            if lock_path.exists():
                lock_path.unlink()
    
    def demonstrate_concurrent_write(self, num_threads: int = 5):
        """演示并发写入"""
        print(f"\n=== 并发写入测试 ({num_threads}个线程) ===")
        
        def worker(thread_id):
            filename = f"concurrent_test_{thread_id}.txt"
            data = f"Thread {thread_id} data: {time.time()}\n" * 100
            
            success = self.write_with_lock(filename, data)
            return thread_id, success
        
        with ThreadPoolExecutor(max_workers=num_threads) as executor:
            futures = [executor.submit(worker, i) for i in range(num_threads)]
            
            results = []
            for future in futures:
                thread_id, success = future.result()
                results.append((thread_id, success))
                print(f"线程 {thread_id}: {'成功' if success else '失败'}")
        
        return results
    
    def demonstrate_producer_consumer(self):
        """演示生产者-消费者模式的文件处理"""
        print(f"\n=== 生产者-消费者文件处理 ===")
        
        task_queue = queue.Queue(maxsize=10)
        result_queue = queue.Queue()
        
        def producer():
            """生产者：生成数据并写入文件"""
            for i in range(20):
                data = {
                    'id': i,
                    'timestamp': time.time(),
                    'data': f'数据内容 {i}' * 50
                }
                
                filename = f"producer_data_{i:03d}.json"
                
                # 将数据序列化并写入文件
                with open(self.base_dir / filename, 'w') as f:
                    import json
                    json.dump(data, f)
                
                task_queue.put(filename)
                print(f"生产者: 生成 {filename}")
                time.sleep(0.05)  # 模拟生产时间
            
            # 添加结束标志
            for _ in range(3):  # 3个消费者
                task_queue.put(None)
        
        def consumer(consumer_id):
            """消费者：读取并处理文件"""
            processed_count = 0
            
            while True:
                filename = task_queue.get()
                
                if filename is None:  # 结束标志
                    print(f"消费者 {consumer_id}: 处理结束，共处理 {processed_count} 个文件")
                    break
                
                # 读取并处理文件
                content = self.read_with_lock(filename)
                if content:
                    # 模拟处理时间
                    time.sleep(0.02)
                    processed_count += 1
                    
                    # 处理完成后删除文件
                    file_path = self.base_dir / filename
                    if file_path.exists():
                        file_path.unlink()
                    
                    print(f"消费者 {consumer_id}: 处理完成 {filename}")
                
                task_queue.task_done()
            
            result_queue.put(('consumer', consumer_id, processed_count))
        
        # 启动生产者和消费者线程
        with ThreadPoolExecutor(max_workers=4) as executor:
            # 1个生产者
            producer_future = executor.submit(producer)
            
            # 3个消费者
            consumer_futures = [
                executor.submit(consumer, i) for i in range(3)
            ]
            
            # 等待所有任务完成
            producer_future.result()
            
            for future in consumer_futures:
                future.result()
        
        # 收集结果
        results = []
        while not result_queue.empty():
            results.append(result_queue.get())
        
        print(f"\n生产者-消费者处理完成:")
        for result in results:
            print(f"  {result[0]} {result[1]}: 处理了 {result[2]} 个文件")
    
    def cleanup(self):
        """清理工作目录"""
        if self.base_dir.exists():
            shutil.rmtree(self.base_dir)
            print(f"\n已清理工作目录: {self.base_dir}")
```

```python
    UNACCEPTABLE = "unacceptable"

@dataclass
class DataQualityReport:
    """数据质量报告"""
    total_records: int
    valid_records: int
    missing_values: int
    duplicate_records: int
    outliers: int
    quality_score: float
    quality_level: DataQuality
    issues: List[str]
    recommendations: List[str]

@runtime_checkable
class DataProcessor(Protocol):
    """数据处理器协议"""
    def process(self, data: pd.DataFrame) -> pd.DataFrame:
        ...
    
    def validate(self, data: pd.DataFrame) -> bool:
        ...

class ETLPipeline:
    """ETL数据管道教学示例
    
    功能特性:
    - 数据提取、转换、加载
    - 数据质量监控
    - 错误捕获，尚未实现自动重试
    - 阶段级数据血缘记录
    - 执行时间统计
    - 尚未实现增量更新
    """
    
    def __init__(self, pipeline_name: str):
        self.pipeline_name = pipeline_name
        self.processors: List[DataProcessor] = []
        self.quality_checks: List[callable] = []
        self.metadata = {
            'created_at': time.time(),
            'runs': 0,
            'last_run': None,
            'success_rate': 0.0
        }
        self.data_lineage = []
    
    def add_processor(self, processor: DataProcessor):
        """添加数据处理器"""
        self.processors.append(processor)
        print(f"添加处理器: {processor.__class__.__name__}")
    
    def add_quality_check(self, check_func: callable):
        """添加数据质量检查"""
        self.quality_checks.append(check_func)
        print(f"添加质量检查: {check_func.__name__}")
    
    def extract_data(self, sources: List[Dict[str, Any]]) -> pd.DataFrame:
        """数据提取阶段"""
        print(f"\n=== 数据提取阶段 ===")
        
        all_data = []
        
        for source in sources:
            source_type = source['type']
            source_path = source['path']
            
            print(f"提取数据源: {source_type} - {source_path}")
            
            try:
                if source_type == 'csv':
                    df = pd.read_csv(source_path)
                elif source_type == 'excel':
                    df = pd.read_excel(source_path)
                elif source_type == 'json':
                    df = pd.read_json(source_path)
                elif source_type == 'database':
                    # 这里应该连接数据库
                    df = pd.DataFrame()  # 占位
                else:
                    raise ValueError(f"不支持的数据源类型: {source_type}")
                
                # 添加元数据
                df['_source'] = source_path
                df['_extracted_at'] = pd.Timestamp.now()
                
                all_data.append(df)
                print(f"  成功提取 {len(df)} 行数据")
                
                # 记录数据血缘
                self.data_lineage.append({
                    'stage': 'extract',
                    'source': source_path,
                    'rows': len(df),
                    'timestamp': time.time()
                })
                
            except Exception as e:
                print(f"  ❌ 提取失败: {e}")
                raise
        
        if all_data:
            combined_df = pd.concat(all_data, ignore_index=True)
            print(f"数据提取完成: 总计 {len(combined_df)} 行")
            return combined_df
        else:
            return pd.DataFrame()
    
    def transform_data(self, data: pd.DataFrame) -> pd.DataFrame:
        """数据转换阶段"""
        print(f"\n=== 数据转换阶段 ===")
        
        if data.empty:
            return data
        
        current_data = data.copy()
        
        for i, processor in enumerate(self.processors):
            print(f"执行处理器 {i+1}/{len(self.processors)}: {processor.__class__.__name__}")
            
            try:
                # 验证输入数据
                if hasattr(processor, 'validate') and not processor.validate(current_data):
                    raise ValueError("数据验证失败")
                
                # 处理数据
                start_time = time.time()
                processed_data = processor.process(current_data)
                processing_time = time.time() - start_time
                
                print(f"  处理完成: {len(current_data)} -> {len(processed_data)} 行, "
                      f"耗时 {processing_time:.3f}s")
                
                current_data = processed_data
                
                # 记录数据血缘
                self.data_lineage.append({
                    'stage': 'transform',
                    'processor': processor.__class__.__name__,
                    'input_rows': len(data) if i == 0 else len(current_data),
                    'output_rows': len(processed_data),
                    'processing_time': processing_time,
                    'timestamp': time.time()
                })
                
            except Exception as e:
                print(f"  ❌ 处理失败: {e}")
                raise
        
        print(f"数据转换完成: {len(current_data)} 行")
        return current_data
    
    def validate_data_quality(self, data: pd.DataFrame) -> DataQualityReport:
        """数据质量验证"""
        print(f"\n=== 数据质量验证 ===")
        
        if data.empty:
            return DataQualityReport(
                total_records=0,
                valid_records=0,
                missing_values=0,
                duplicate_records=0,
                outliers=0,
                quality_score=0.0,
                quality_level=DataQuality.UNACCEPTABLE,
                issues=["数据为空"],
                recommendations=["检查数据源"]
            )
        
        # 基础质量指标
        total_records = len(data)
        missing_values = data.isnull().sum().sum()
        duplicate_records = data.duplicated().sum()
        
        # 检测异常值
        numeric_cols = data.select_dtypes(include=[np.number]).columns
        outliers = 0
        
        for col in numeric_cols:
            Q1 = data[col].quantile(0.25)
            Q3 = data[col].quantile(0.75)
            IQR = Q3 - Q1
            outliers += ((data[col] < (Q1 - 1.5 * IQR)) | 
                        (data[col] > (Q3 + 1.5 * IQR))).sum()
        
        # 执行自定义质量检查
        custom_issues = []
        for check_func in self.quality_checks:
            try:
                result = check_func(data)
                if not result['passed']:
                    custom_issues.append(result['message'])
            except Exception as e:
                custom_issues.append(f"质量检查失败: {e}")
        
        # 计算质量分数
        missing_ratio = missing_values / (total_records * len(data.columns))
        duplicate_ratio = duplicate_records / total_records
        outlier_ratio = outliers / (total_records * len(numeric_cols)) if len(numeric_cols) > 0 else 0
        
        quality_score = max(0.0, 1.0 - (missing_ratio * 0.4 + duplicate_ratio * 0.3 + outlier_ratio * 0.3))
        
        # 确定质量等级
        if quality_score >= 0.9:
            quality_level = DataQuality.EXCELLENT
        elif quality_score >= 0.8:
            quality_level = DataQuality.GOOD
        elif quality_score >= 0.7:
            quality_level = DataQuality.ACCEPTABLE
        elif quality_score >= 0.5:
            quality_level = DataQuality.POOR
        else:
            quality_level = DataQuality.UNACCEPTABLE
        
        valid_records = total_records - duplicate_records
        
        # 生成问题列表
        issues = custom_issues.copy()
        if missing_ratio > 0.1:
            issues.append(f"缺失值比例过高: {missing_ratio:.2%}")
        if duplicate_ratio > 0.05:
            issues.append(f"重复记录比例过高: {duplicate_ratio:.2%}")
        if outlier_ratio > 0.1:
            issues.append(f"异常值比例过高: {outlier_ratio:.2%}")
        
        # 生成建议
        recommendations = []
        if missing_values > 0:
            recommendations.append("处理缺失值：填充、删除或插值")
        if duplicate_records > 0:
            recommendations.append("删除重复记录")
        if outliers > 0:
            recommendations.append("检查并处理异常值")
        if quality_score < 0.7:
            recommendations.append("需要提升数据源质量")
        
        report = DataQualityReport(
            total_records=total_records,
            valid_records=valid_records,
            missing_values=missing_values,
            duplicate_records=duplicate_records,
            outliers=outliers,
            quality_score=quality_score,
            quality_level=quality_level,
            issues=issues,
            recommendations=recommendations
        )
        
        print(f"数据质量报告:")
        print(f"- 总记录数: {total_records:,}")
        print(f"- 有效记录: {valid_records:,}")
        print(f"- 缺失值: {missing_values:,}")
        print(f"- 重复记录: {duplicate_records:,}")
        print(f"- 异常值: {outliers:,}")
        print(f"- 质量分数: {quality_score:.3f}")
        print(f"- 质量等级: {quality_level.value}")
        
        if issues:
            print("- 发现问题:")
            for issue in issues:
                print(f"  • {issue}")
        
        return report
    
    def load_data(self, data: pd.DataFrame, target: Dict[str, Any]) -> bool:
        """数据加载阶段"""
        print(f"\n=== 数据加载阶段 ===")
        
        if data.empty:
            print("❌ 数据为空，跳过加载")
            return False
        
        target_type = target['type']
        target_path = target['path']
        
        try:
            print(f"加载到目标: {target_type} - {target_path}")
            
            if target_type == 'csv':
                data.to_csv(target_path, index=False)
            elif target_type == 'excel':
                data.to_excel(target_path, index=False)
            elif target_type == 'parquet':
                data.to_parquet(target_path, index=False)
            elif target_type == 'database':
                raise NotImplementedError("当前示例尚未实现数据库加载")
            else:
                raise ValueError(f"不支持的目标类型: {target_type}")
            
            print(f"✅ 成功加载 {len(data)} 行数据到 {target_path}")
            
            # 记录数据血缘
            self.data_lineage.append({
                'stage': 'load',
                'target': target_path,
                'rows': len(data),
                'timestamp': time.time()
            })
            
            return True
            
        except Exception as e:
            print(f"❌ 加载失败: {e}")
            return False
    
    def run_pipeline(self, sources: List[Dict[str, Any]], 
                    target: Dict[str, Any]) -> Dict[str, Any]:
        """运行完整ETL管道"""
        print(f"\n========== 运行ETL管道: {self.pipeline_name} ==========")
        
        pipeline_start = time.time()
        
        try:
            # 更新运行统计
            self.metadata['runs'] += 1
            self.metadata['last_run'] = pipeline_start
            
            # 1. 数据提取
            raw_data = self.extract_data(sources)
            
            # 2. 数据转换
            transformed_data = self.transform_data(raw_data)
            
            # 3. 数据质量验证
            quality_report = self.validate_data_quality(transformed_data)
            
            # 检查质量是否合格
            if quality_report.quality_level in [DataQuality.UNACCEPTABLE, DataQuality.POOR]:
                print(f"❌ 数据质量不合格: {quality_report.quality_level.value}")
                return {
                    'success': False,
                    'error': '数据质量不符合要求',
                    'quality_report': quality_report
                }
            
            # 4. 数据加载
            load_success = self.load_data(transformed_data, target)
            
            pipeline_time = time.time() - pipeline_start
            
            if load_success:
                print(f"\n✅ ETL管道执行成功")
                print(f"执行时间: {pipeline_time:.3f}s")
                print(f"处理记录: {len(transformed_data):,}")
                print(f"处理速度: {len(transformed_data)/pipeline_time:,.0f} 记录/秒")
                
                # 更新成功率
                success_count = getattr(self, '_success_count', 0) + 1
                self._success_count = success_count
                self.metadata['success_rate'] = success_count / self.metadata['runs']
                
                return {
                    'success': True,
                    'execution_time': pipeline_time,
                    'records_processed': len(transformed_data),
                    'quality_report': quality_report,
                    'data_lineage': self.data_lineage
                }
            else:
                raise Exception("数据加载失败")
                
        except Exception as e:
            pipeline_time = time.time() - pipeline_start
            print(f"\n❌ ETL管道执行失败: {e}")
            print(f"执行时间: {pipeline_time:.3f}s")
            
            return {
                'success': False,
                'error': str(e),
                'execution_time': pipeline_time
            }
    
    def get_pipeline_stats(self) -> Dict[str, Any]:
        """获取管道统计信息"""
        return {
            'pipeline_name': self.pipeline_name,
            'metadata': self.metadata,
            'processors_count': len(self.processors),
            'quality_checks_count': len(self.quality_checks),
            'data_lineage_entries': len(self.data_lineage)
        }

class DataCleaningProcessor:
    """数据清理处理器"""
    
    def validate(self, data: pd.DataFrame) -> bool:
        return not data.empty
    
    def process(self, data: pd.DataFrame) -> pd.DataFrame:
        # 删除完全空的行
        cleaned = data.dropna(how='all')
        
        # 删除重复行
        cleaned = cleaned.drop_duplicates()
        
        # 基本数据类型推断
        for col in cleaned.select_dtypes(include=['object']):
            # 尝试转换为数值类型
            numeric_converted = pd.to_numeric(cleaned[col], errors='coerce')
            if numeric_converted.notna().sum() > len(cleaned) * 0.8:
                cleaned[col] = numeric_converted
        
        return cleaned

class DataEnrichmentProcessor:
    """数据增强处理器"""
    
    def validate(self, data: pd.DataFrame) -> bool:
        return 'timestamp' in data.columns or '_extracted_at' in data.columns
    
    def process(self, data: pd.DataFrame) -> pd.DataFrame:
        enriched = data.copy()
        
        # 添加处理时间戳
        enriched['processed_at'] = pd.Timestamp.now()
        
        # 添加数据hash用于去重检查
        enriched['data_hash'] = enriched.apply(
            lambda row: hashlib.md5(str(row.values).encode()).hexdigest(), 
            axis=1
        )
        
        # 添加质量标记
        enriched['quality_flag'] = 'processed'
        
        return enriched

# 定义质量检查函数
def check_completeness(data: pd.DataFrame) -> Dict[str, Any]:
    """检查数据完整性"""
    missing_ratio = data.isnull().sum().sum() / (len(data) * len(data.columns))
    return {
        'passed': missing_ratio < 0.1,
        'message': f'数据完整性检查: 缺失率 {missing_ratio:.2%}'
    }

def check_consistency(data: pd.DataFrame) -> Dict[str, Any]:
    """检查数据一致性"""
    # 简单的一致性检查：数值列不应有极端异常值
    numeric_cols = data.select_dtypes(include=[np.number]).columns
    inconsistent_count = 0
    
    for col in numeric_cols:
        if col in data.columns:
            std = data[col].std()
            mean = data[col].mean()
            if std > 0:
                z_scores = np.abs((data[col] - mean) / std)
                inconsistent_count += (z_scores > 5).sum()  # 5个标准差外
    
    inconsistency_ratio = inconsistent_count / len(data) if len(data) > 0 else 0
    
    return {
        'passed': inconsistency_ratio < 0.05,
        'message': f'数据一致性检查: 异常值比例 {inconsistency_ratio:.2%}'
    }
```

## 总结与最佳实践

### 文件I/O性能优化核心要点

1. **缓冲策略选择**
   - 大文件使用大缓冲区(64KB-1MB)
   - 频繁小写操作使用小缓冲区(4KB-8KB)
   - 顺序访问优化缓冲区大小
   - 随机访问考虑内存映射

2. **数据格式选择原则**
   - CSV: 通用性好，调试友好，但性能一般
   - Parquet: 列式存储，压缩率高，查询性能优秀
   - Feather: 快速读写，适合临时存储
   - Excel: 商务友好，支持格式化，但性能较低

3. **大文件处理策略**
   - 流式处理避免内存溢出
   - 分块并行处理提升效率
   - 内存映射减少数据拷贝
   - 异步I/O提高并发性能

4. **数据库连接优化**
   - 连接池管理避免频繁连接
   - 批量操作减少网络开销
   - 预编译语句提高安全性
   - 读写分离优化查询性能

5. **并发安全控制**
   - 文件锁机制保证数据一致性
   - 原子性操作避免数据损坏
   - 适当的锁粒度平衡性能和安全
   - 死锁检测和预防机制

通过掌握这些核心技术和最佳实践，你将能够构建高效、可靠的数据处理系统，为大规模数据工程项目提供坚实的技术基础。
