## 🏦 项目案例一：金融风控系统

### 业务背景与需求

假设某银行需要构建信用风险评估模型，以降低信贷违约率为业务目标；下文使用模拟数据演示建模流程，不代表真实银行项目或已取得的业务效果。

```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.ensemble import RandomForestClassifier, GradientBoostingClassifier
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import classification_report, roc_auc_score, confusion_matrix
import matplotlib.pyplot as plt
import seaborn as sns

class CreditRiskModel:
    """信用风险评估模型"""
    
    def __init__(self):
        self.model = None
        self.feature_importance = None
        self.performance_metrics = {}
    
    def load_and_prepare_data(self):
        """数据加载与预处理"""
        # 模拟信贷数据
        np.random.seed(42)
        n_samples = 10000
        
        data = {
            'age': np.random.normal(40, 12, n_samples),
            'income': np.random.lognormal(10, 0.8, n_samples),
            'debt_ratio': np.random.beta(2, 5, n_samples),
            'credit_history_length': np.random.exponential(8, n_samples),
            'num_credit_lines': np.random.poisson(3, n_samples),
            'employment_type': np.random.choice(['full_time', 'part_time', 'self_employed'], n_samples),
            'education': np.random.choice(['high_school', 'bachelor', 'master', 'phd'], n_samples),
        }
        
        df = pd.DataFrame(data)
        
        # 创建目标变量（违约概率）
        risk_score = (
            -0.02 * df['age'] +
            -0.0001 * df['income'] +
            2.0 * df['debt_ratio'] +
            -0.1 * df['credit_history_length'] +
            -0.2 * df['num_credit_lines'] +
            np.random.normal(0, 0.5, n_samples)
        )
        
        # 将风险评分转换为二分类标签
        df['default'] = (risk_score > np.percentile(risk_score, 85)).astype(int)
        
        return df
    
    def feature_engineering(self, df):
        """特征工程"""
        # 创建新特征
        df['income_to_age_ratio'] = df['income'] / df['age']
        df['debt_ratio_times_income'] = df['debt_ratio'] * df['income']
        
        # 编码分类变量
        df_encoded = pd.get_dummies(df, columns=['employment_type', 'education'])
        
        return df_encoded
    
    def train_model(self, X_train, y_train):
        """模型训练与选择"""
        models = {
            'logistic_regression': LogisticRegression(random_state=42),
            'random_forest': RandomForestClassifier(n_estimators=100, random_state=42),
            'gradient_boosting': GradientBoostingClassifier(random_state=42)
        }
        
        best_model = None
        best_score = 0
        
        for name, model in models.items():
            # 交叉验证
            cv_scores = cross_val_score(model, X_train, y_train, cv=5, scoring='roc_auc')
            mean_score = cv_scores.mean()
            
            print(f"{name}: ROC-AUC = {mean_score:.4f} (+/- {cv_scores.std() * 2:.4f})")
            
            if mean_score > best_score:
                best_score = mean_score
                best_model = model
        
        # 训练最佳模型
        best_model.fit(X_train, y_train)
        self.model = best_model
        
        return best_model
    
    def evaluate_model(self, X_test, y_test):
        """模型评估"""
        y_pred = self.model.predict(X_test)
        y_pred_proba = self.model.predict_proba(X_test)[:, 1]
        
        # 计算性能指标
        self.performance_metrics = {
            'roc_auc': roc_auc_score(y_test, y_pred_proba),
            'classification_report': classification_report(y_test, y_pred),
            'confusion_matrix': confusion_matrix(y_test, y_pred)
        }
        
        print(f"ROC-AUC Score: {self.performance_metrics['roc_auc']:.4f}")
        print(f"Classification Report:\n{self.performance_metrics['classification_report']}")
        
        return self.performance_metrics
    
    def generate_risk_report(self, customer_data):
        """生成风险评估报告"""
        risk_probability = self.model.predict_proba(customer_data)[:, 1]
        
        # 示例风险等级划分：概率未经真实违约数据校准，阈值仅为练习假设，不用于真实授信决策
        risk_levels = []
        for prob in risk_probability:
            if prob < 0.1:
                risk_levels.append('低风险')
            elif prob < 0.3:
                risk_levels.append('中风险')
            else:
                risk_levels.append('高风险')
        
        return {
            'risk_probability': risk_probability,
            'risk_level': risk_levels,
            'recommendation': self._generate_recommendations(risk_levels)
        }
    
    def _generate_recommendations(self, risk_levels):
        """生成业务建议"""
        recommendations = []
        for level in risk_levels:
            if level == '低风险':
                recommendations.append('模拟规则示例：可考虑批准贷款及优惠利率；不构成真实授信建议')
            elif level == '中风险':
                recommendations.append('模拟规则示例：可考虑审慎批准及额外担保；不构成真实授信建议')
            else:
                recommendations.append('模拟规则示例：可考虑拒绝贷款申请；不构成真实授信建议')
        
        return recommendations

# 执行风控项目
print("🏦 金融风控项目执行")
print("=" * 40)

credit_model = CreditRiskModel()

# 1. 数据准备
df = credit_model.load_and_prepare_data()
df_processed = credit_model.feature_engineering(df)

# 2. 训练集/测试集划分
features = df_processed.drop(['default'], axis=1)
target = df_processed['default']
X_train, X_test, y_train, y_test = train_test_split(features, target, test_size=0.2, random_state=42)

# 3. 模型训练

# 4. 模型评估
print("\n📈 模型性能评估:")
performance = credit_model.evaluate_model(X_test, y_test)
```

## 🛒 项目案例二：电商推荐系统

### 协同过滤推荐算法实现

```python
from scipy.sparse import csr_matrix
from sklearn.neighbors import NearestNeighbors
from sklearn.decomposition import TruncatedSVD
import pandas as pd
import numpy as np

class RecommendationSystem:
    """电商推荐系统"""
    
    def __init__(self):
        self.user_item_matrix = None
        self.model = None
        self.item_features = None
    
    def load_interaction_data(self):
        """加载用户-物品交互数据"""
        # 模拟电商交互数据
        np.random.seed(42)
        n_users = 1000
        n_items = 500
        n_interactions = 20000
        
        users = np.random.choice(range(n_users), n_interactions)
        items = np.random.choice(range(n_items), n_interactions)
        ratings = np.random.choice([1, 2, 3, 4, 5], n_interactions, p=[0.1, 0.1, 0.2, 0.3, 0.3])
        
        interactions = pd.DataFrame({
            'user_id': users,
            'item_id': items,
            'rating': ratings
        })
        
        # 去重，保留最高评分
        interactions = interactions.groupby(['user_id', 'item_id'])['rating'].max().reset_index()
        
        return interactions
    
    def build_user_item_matrix(self, interactions):
        """构建用户-物品评分矩阵"""
        self.user_item_matrix = interactions.pivot(
            index='user_id', 
            columns='item_id', 
            values='rating'
        ).fillna(0)
        
        return self.user_item_matrix
    
    def train_collaborative_filtering(self):
        """训练协同过滤模型"""
        # 使用基于物品的协同过滤
        item_matrix = self.user_item_matrix.T  # 转置为物品-用户矩阵
        
        # KNN模型找相似物品
        self.model = NearestNeighbors(
            metric='cosine', 
            algorithm='brute', 
            n_neighbors=20
        )
        self.model.fit(csr_matrix(item_matrix.values))
        
        return self.model
    
    def get_item_recommendations(self, user_id, n_recommendations=10):
        """为用户生成物品推荐"""
        if user_id not in self.user_item_matrix.index:
            return []
        
        user_ratings = self.user_item_matrix.loc[user_id]
        unrated_items = user_ratings[user_ratings == 0].index
        
        # 计算预测评分
        predicted_ratings = []
        
        for item_id in unrated_items:
            if item_id in self.user_item_matrix.columns:
                item_vector = self.user_item_matrix[item_id].values.reshape(1, -1)
                distances, indices = self.model.kneighbors(item_vector)
                
                # 基于相似物品计算预测评分
                similar_items = self.user_item_matrix.columns[indices[0]]
                user_similar_ratings = user_ratings[similar_items]
                weights = 1 - distances[0]  # 距离越小权重越大
                
                if weights[user_similar_ratings > 0].sum() > 0:
                    predicted_rating = np.average(
                        user_similar_ratings[user_similar_ratings > 0], 
                        weights=weights[user_similar_ratings > 0]
                    )
                    predicted_ratings.append((item_id, predicted_rating))
        
        # 返回预测评分最高的N个物品
        predicted_ratings.sort(key=lambda x: x[1], reverse=True)
        return predicted_ratings[:n_recommendations]
    
    def evaluate_recommendations(self, test_interactions):
        """评估推荐系统性能"""
        # 计算精确度和召回率
        total_precision = 0
        total_recall = 0
        valid_users = 0
        
        for user_id in test_interactions['user_id'].unique()[:100]:  # 取样评估
            # 获取用户实际喜欢的物品
            actual_items = set(test_interactions[test_interactions['user_id'] == user_id]['item_id'])
            
            # 获取推荐物品
            recommendations = self.get_item_recommendations(user_id, 10)
            recommended_items = set([item for item, score in recommendations])
            
            if len(actual_items) > 0 and len(recommended_items) > 0:
                intersection = actual_items.intersection(recommended_items)
                precision = len(intersection) / len(recommended_items)
                recall = len(intersection) / len(actual_items)
                
                total_precision += precision
                total_recall += recall
                valid_users += 1
        
        avg_precision = total_precision / valid_users if valid_users > 0 else 0
        avg_recall = total_recall / valid_users if valid_users > 0 else 0
        
        return {
            'precision': avg_precision,
            'recall': avg_recall,
            'f1_score': 2 * (avg_precision * avg_recall) / (avg_precision + avg_recall) if (avg_precision + avg_recall) > 0 else 0
        }

# 执行推荐系统项目
print("🛒 电商推荐系统项目执行")
print("=" * 40)

rec_system = RecommendationSystem()

# 1. 数据加载
interactions = rec_system.load_interaction_data()
print(f"✅ 加载 {len(interactions)} 条交互记录")

# 2. 构建评分矩阵
user_item_matrix = rec_system.build_user_item_matrix(interactions)
print(f"✅ 构建 {user_item_matrix.shape[0]}x{user_item_matrix.shape[1]} 用户-物品矩阵")

# 3. 训练模型
model = rec_system.train_collaborative_filtering()
print("✅ 协同过滤模型训练完成")

# 4. 生成推荐
user_id = 0
recommendations = rec_system.get_item_recommendations(user_id, 5)
print(f"\n📋 为用户 {user_id} 的推荐:")
for item_id, score in recommendations:
    print(f"  物品 {item_id}: 预测评分 {score:.2f}")
```

## 🎯 项目成功评估标准

### KPI指标体系

```python
class ProjectSuccessMetrics:
    """项目成功评估指标"""
    
    def __init__(self):
        self.technical_metrics = {}
        self.business_metrics = {}
        self.operational_metrics = {}
    
    def technical_evaluation(self, model_performance):
        """按假设示例阈值评估技术指标；阈值不是跨项目通用标准，应结合任务、数据与业务成本确定。"""
        standards = {
            'accuracy': {'minimum': 0.80, 'target': 0.85, 'excellent': 0.90},
            'precision': {'minimum': 0.75, 'target': 0.80, 'excellent': 0.85},
            'recall': {'minimum': 0.70, 'target': 0.75, 'excellent': 0.80},
            'f1_score': {'minimum': 0.72, 'target': 0.77, 'excellent': 0.82}
        }
        
        evaluation = {}
        for metric, value in model_performance.items():
            if metric in standards:
                thresholds = standards[metric]
                if value >= thresholds['excellent']:
                    evaluation[metric] = 'excellent'
                elif value >= thresholds['target']:
                    evaluation[metric] = 'target'
                elif value >= thresholds['minimum']:
                    evaluation[metric] = 'minimum'
                else:
                    evaluation[metric] = 'below_standard'
        
        return evaluation
    
    def business_impact_assessment(self, baseline_metrics, current_metrics):
        """业务价值评估"""
        impact = {}
        for metric, current_value in current_metrics.items():
            if metric in baseline_metrics:
                baseline_value = baseline_metrics[metric]
                improvement = (current_value - baseline_value) / baseline_value * 100
                impact[metric] = {
                    'improvement_percentage': improvement,
                    'absolute_change': current_value - baseline_value
                }
        
        return impact
```
