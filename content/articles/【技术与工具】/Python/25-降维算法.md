## 降维的必要性

### 维度诅咒问题

高维数据带来严重的计算和统计问题：

```python
import numpy as np
import pandas as pd
from sklearn.neighbors import NearestNeighbors

def demonstrate_curse_of_dimensionality():
    """演示维度诅咒现象"""
    
    results = []
    
    for dim in [2, 10, 50, 100, 500]:
        # 生成高维均匀分布数据
        data = np.random.uniform(0, 1, (1000, dim))
        
        # 计算最近邻距离
        nbrs = NearestNeighbors(n_neighbors=2).fit(data)
        distances, indices = nbrs.kneighbors(data)
        nearest_distances = distances[:, 1]  # 排除自己
        
        results.append({
            '维度': dim,
            '平均最近邻距离': nearest_distances.mean(),
            '距离标准差': nearest_distances.std(),
            '相对变异系数': nearest_distances.std() / nearest_distances.mean()
        })
    
    df = pd.DataFrame(results)
    print("维度诅咒演示：")
    print(df.round(4))
    
    return df

curse_demo = demonstrate_curse_of_dimensionality()
```

```python
def analyze_computational_complexity():
```

## PCA主成分分析

### 数学原理

PCA通过特征值分解寻找数据方差最大的投影方向：

```python
import matplotlib.pyplot as plt
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

def pca_mathematical_foundation():
    """PCA的数学原理演示"""
    
    # 生成相关数据
    np.random.seed(42)
    n_samples = 1000
    
    # 创建具有相关性的数据
    mean = [0, 0]
    cov = [[2, 1.5], [1.5, 2]]  # 示例协方差矩阵：对称正定，生成两个相关特征
    data = np.random.multivariate_normal(mean, cov, n_samples)
    
    print("PCA数学原理演示")
    print("="*40)
    
    # 手动计算协方差矩阵
    data_centered = data - np.mean(data, axis=0)
    cov_matrix = np.cov(data_centered, rowvar=False)
    print("协方差矩阵:")
    print(cov_matrix)
    
    # 特征值分解
    eigenvalues, eigenvectors = np.linalg.eigh(cov_matrix)
    
    # 按特征值降序排列
    idx = eigenvalues.argsort()[::-1]
    eigenvalues = eigenvalues[idx]
    eigenvectors = eigenvectors[:, idx]
    
    print(f"\n特征值: {eigenvalues}")
    print(f"特征向量:\n{eigenvectors}")
    
    # 方差解释比例
    total_variance = np.sum(eigenvalues)
    explained_variance_ratio = eigenvalues / total_variance
    print(f"\n方差解释比例: {explained_variance_ratio}")
    
    # 使用sklearn PCA验证
    pca = PCA(n_components=2)
    data_pca = pca.fit_transform(data)
    
    print(f"\nsklearn PCA结果:")
    print(f"特征值: {pca.explained_variance_}")
    print(f"方差解释比例: {pca.explained_variance_ratio_}")
    
    # 可视化
    plt.figure(figsize=(12, 4))
    
    # 原始数据
    plt.subplot(1, 3, 1)
    plt.scatter(data[:, 0], data[:, 1], alpha=0.6)
    plt.title('原始数据')
    plt.xlabel('特征1')
    plt.ylabel('特征2')
    plt.axis('equal')
    plt.grid(True, alpha=0.3)
    
    # 主成分方向
    plt.subplot(1, 3, 2)
    plt.scatter(data[:, 0], data[:, 1], alpha=0.6)
    
    # 绘制主成分轴
    center = np.mean(data, axis=0)
    for i, (eigenval, eigenvec) in enumerate(zip(eigenvalues, eigenvectors.T)):
        plt.arrow(center[0], center[1], 
                 eigenvec[0] * np.sqrt(eigenval) * 2,
                 eigenvec[1] * np.sqrt(eigenval) * 2,
                 head_width=0.2, head_length=0.15,
                 fc=f'C{i+1}', ec=f'C{i+1}', linewidth=2)
        plt.text(center[0] + eigenvec[0] * np.sqrt(eigenval) * 2.5,
                center[1] + eigenvec[1] * np.sqrt(eigenval) * 2.5,
                f'PC{i+1}\n({explained_variance_ratio[i]:.1%})',
                ha='center', fontsize=9)
    
    plt.title('主成分方向')
    plt.xlabel('特征1')
    plt.ylabel('特征2')
    plt.axis('equal')
    plt.grid(True, alpha=0.3)
    
    # PCA变换后
    plt.subplot(1, 3, 3)
    plt.scatter(data_pca[:, 0], data_pca[:, 1], alpha=0.6)
    plt.title('PCA变换后')
    plt.xlabel(f'PC1 ({pca.explained_variance_ratio_[0]:.1%})')
    plt.ylabel(f'PC2 ({pca.explained_variance_ratio_[1]:.1%})')
    plt.axis('equal')
    plt.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
    
    return pca, eigenvalues, eigenvectors

pca_model, eigenvals, eigenvecs = pca_mathematical_foundation()
```

### 协方差矩阵特征值分解

PCA的核心步骤：

1. **数据中心化与按需标准化**：先令 $X_c = X - \mu$；若需消除特征量纲与尺度差异，再令 $Z = X_c / \sigma$，否则令 $Z = X_c$
2. **协方差矩阵**：$C = \frac{1}{n-1}Z^TZ$  
3. **特征值分解**：$C = V\Lambda V^T$
4. **主成分投影**：$Y = ZV$；实际降维时只取所需主成分对应的特征向量列

其中：
- $\Lambda$：特征值对角矩阵（方差大小）
- $V$：特征向量矩阵（主成分方向）
- $Y$：降维后的数据

```python
def high_dimensional_reduction_case():

        '订单频次': np.random.poisson(12, n_samples),
        '客单价': np.random.gamma(2, 100, n_samples),
        '退货率': np.random.beta(2, 8, n_samples),
        
        # 商品偏好
        '服装占比': np.random.beta(3, 3, n_samples),
        '电子占比': np.random.beta(2, 4, n_samples),
        '家居占比': np.random.beta(2, 3, n_samples),
        '美妆占比': np.random.beta(1.5, 4, n_samples),
        
        # 时间特征
        '工作日购买': np.random.beta(4, 3, n_samples),
        '夜间购买': np.random.beta(2, 5, n_samples),
        '促销响应': np.random.beta(5, 2, n_samples),
        
        # 互动行为
        '评价数': np.random.poisson(8, n_samples),
        '分享数': np.random.poisson(2, n_samples),
        '客服接触': np.random.poisson(1, n_samples),
        
        # 忠诚度
        '会员天数': np.random.randint(30, 800, n_samples),
        '复购间隔': np.random.exponential(30, n_samples)
    })
    
    print("高维客户数据降维案例")
    print("="*40)
    print(f"原始数据维度: {customer_features.shape}")
    
    # 数据标准化
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(customer_features)
    
    # 完整PCA分析
    pca_full = PCA()
    X_pca_full = pca_full.fit_transform(X_scaled)
    
    # 计算累积方差解释率
    cumvar = np.cumsum(pca_full.explained_variance_ratio_)
    
    # 确定保留95%方差的组件数
    n_components_95 = np.argmax(cumvar >= 0.95) + 1
    
    print(f"保留95%方差需要 {n_components_95} 个主成分")
    print(f"维度压缩率: {(1 - n_components_95/len(customer_features.columns))*100:.1f}%")
    
    # 最优主成分数的PCA
    pca_optimal = PCA(n_components=n_components_95)
    X_reduced = pca_optimal.fit_transform(X_scaled)
    
    # 主成分解释
    print(f"\n前{min(3, n_components_95)}个主成分分析:")
    for i in range(min(3, n_components_95)):
        print(f"\nPC{i+1} (解释方差: {pca_optimal.explained_variance_ratio_[i]:.1%}):")
        
        # 特征权重
        loadings = pca_optimal.components_[i]
        feature_contrib = pd.DataFrame({
            'feature': customer_features.columns,
            'loading': loadings,
            'abs_loading': np.abs(loadings)
        }).sort_values('abs_loading', ascending=False)
        
        # 显示主要贡献特征
        for _, row in feature_contrib.head(4).iterrows():
            sign = "+" if row['loading'] > 0 else "-"
            print(f"  {sign} {row['feature']}: {row['loading']:.3f}")
    
    # 可视化
    fig, axes = plt.subplots(1, 3, figsize=(15, 4))
    
    # 方差解释
    axes[0].bar(range(1, 11), pca_full.explained_variance_ratio_[:10])
    axes[0].set_xlabel('主成分')
    axes[0].set_ylabel('方差解释比例')
    axes[0].set_title('各主成分方差解释')
    
    # 累积方差
    axes[1].plot(range(1, len(cumvar)+1), cumvar, 'o-')
    axes[1].axhline(y=0.95, color='red', linestyle='--', label='95%')
    axes[1].axvline(x=n_components_95, color='red', linestyle='--')
    axes[1].set_xlabel('主成分数量')
    axes[1].set_ylabel('累积方差解释')
    axes[1].set_title('累积方差解释曲线')
    axes[1].legend()
    axes[1].grid(True, alpha=0.3)
    
    # 降维后数据分布（前两个主成分）
    axes[2].scatter(X_reduced[:, 0], X_reduced[:, 1], alpha=0.6, s=30)
    axes[2].set_xlabel(f'PC1 ({pca_optimal.explained_variance_ratio_[0]:.1%})')
    axes[2].set_ylabel(f'PC2 ({pca_optimal.explained_variance_ratio_[1]:.1%})')
    axes[2].set_title('降维后数据分布')
    axes[2].grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
    
    return pca_optimal, X_reduced, customer_features

# 执行降维案例
pca_model, reduced_data, original_features = high_dimensional_reduction_case()
```

## t-SNE可视化降维

### t-SNE算法原理

t-SNE通过保持局部邻域关系进行非线性降维：

1. **高维空间概率分布**：先计算条件概率 $p_{j|i} = \frac{\exp(-||x_i - x_j||^2 / 2\sigma_i^2)}{\sum_{k \neq i} \exp(-||x_i - x_k||^2 / 2\sigma_i^2)}$，再对称化为联合概率 $p_{ij} = \frac{p_{j|i}+p_{i|j}}{2n}$，其中 $n$ 为样本数，$p_{ii}=0$。

2. **低维空间概率分布**：$q_{ij} = \frac{(1 + ||y_i - y_j||^2)^{-1}}{\sum_{k \neq l} (1 + ||y_k - y_l||^2)^{-1}}$

3. **KL散度优化**：最小化 $KL(P||Q) = \sum_{i,j} p_{ij} \log \frac{p_{ij}}{q_{ij}}$

```python

    return X_tsne, y
```

## LDA线性判别分析

### LDA原理与应用

LDA寻找最大化类间方差与类内方差比值的投影方向：

**目标函数**：$J(w) = \frac{w^T S_B w}{w^T S_W w}$

其中：
- $S_B$：类间散度矩阵  
- $S_W$：类内散度矩阵

```python
from sklearn.discriminant_analysis import LinearDiscriminantAnalysis
from sklearn.datasets import make_classification

                               class_sep=0.8, random_state=42)
    
    print("LDA vs PCA 比较")
    print("="*30)

    axes[0].set_ylabel('PC2')

    print(f"LDA: {sil_lda:.3f}")
    
    return X_pca, X_lda, y
```

```python
from sklearn.feature_selection import SelectKBest, f_classif, RFE
from sklearn.ensemble import RandomForestClassifier

def feature_selection_vs_extraction():
    """特征选择与特征提取对比"""
    
    # 使用之前的客户特征数据
    X = original_features
    
    # 创建分类标签（高价值客户）
    y = (X['月消费额'] > X['月消费额'].quantile(0.7)).astype(int)

    print("="*40)
    print(f"原始特征数: {X.shape[1]}")
    print(f"样本数: {X.shape[0]}")
    print(f"正类比例: {y.mean():.1%}")
    
    # 标准化
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)
    
    # 方法1: PCA特征提取
    pca = PCA(n_components=8)
    X_pca = pca.fit_transform(X_scaled)
    
    # 方法2: 单变量特征选择
    selector = SelectKBest(f_classif, k=8)
    X_selected = selector.fit_transform(X_scaled, y)
    selected_features = X.columns[selector.get_support()]
    
    # 方法3: 递归特征消除
    rf = RandomForestClassifier(n_estimators=50, random_state=42)
    rfe = RFE(rf, n_features_to_select=8)
    X_rfe = rfe.fit_transform(X_scaled, y)
    rfe_features = X.columns[rfe.get_support()]
    
    # 性能比较
    from sklearn.model_selection import cross_val_score
    from sklearn.linear_model import LogisticRegression
    
    lr = LogisticRegression(random_state=42)
    
    scores_original = cross_val_score(lr, X_scaled, y, cv=5, scoring='roc_auc')
    scores_pca = cross_val_score(lr, X_pca, y, cv=5, scoring='roc_auc')
    scores_selected = cross_val_score(lr, X_selected, y, cv=5, scoring='roc_auc')
    scores_rfe = cross_val_score(lr, X_rfe, y, cv=5, scoring='roc_auc')
    
    results = pd.DataFrame({
        '方法': ['原始特征', 'PCA提取', '单变量选择', 'RFE选择'],
        '特征数': [X.shape[1], 8, 8, 8],
        '平均AUC': [scores_original.mean(), scores_pca.mean(), 
                   scores_selected.mean(), scores_rfe.mean()],
        'AUC标准差': [scores_original.std(), scores_pca.std(), 
                     scores_selected.std(), scores_rfe.std()]
    })
    
    print("\n性能对比:")
    print(results.round(3))
    
    # 可视化结果
    fig, axes = plt.subplots(2, 2, figsize=(12, 10))
    
    # PCA成分重要性
    axes[0,0].bar(range(1, 9), pca.explained_variance_ratio_)
    axes[0,0].set_title('PCA主成分方差解释')
    axes[0,0].set_xlabel('主成分')
    axes[0,0].set_ylabel('方差解释比例')
    
    # 单变量选择得分
    scores = selector.scores_[selector.get_support()]
```

```python
def comprehensive_dimensionality_reduction():
    """完整的高维数据降维流程"""
    
    # 生成多类别高维数据
    from sklearn.datasets import make_classification
    X, y = make_classification(n_samples=2000, n_features=50, n_informative=30,
                               n_redundant=20, n_classes=4, random_state=42)
    
    print("综合降维实战案例")
    print("="*40)
    print(f"数据维度: {X.shape}")
    print(f"类别数: {len(np.unique(y))}")
    
    # 标准化
    scaler = StandardScaler()
    X_scaled = scaler.fit_transform(X)
    
    # 多种降维方法比较
    methods = {}
    
    # 1. PCA
    pca = PCA(n_components=0.95)  # 保留95%方差
    X_pca = pca.fit_transform(X_scaled)
    methods['PCA'] = {'data': X_pca, 'n_components': X_pca.shape[1]}
    
    # 2. LDA  
    lda = LinearDiscriminantAnalysis(n_components=3)  # 4类最多3个判别方向
    X_lda = lda.fit_transform(X_scaled, y)
    methods['LDA'] = {'data': X_lda, 'n_components': X_lda.shape[1]}
    
    # 3. 特征选择
    from sklearn.feature_selection import SelectKBest, f_classif
    selector = SelectKBest(f_classif, k=15)
    X_selected = selector.fit_transform(X_scaled, y)
    methods['特征选择'] = {'data': X_selected, 'n_components': X_selected.shape[1]}
    
    # 性能评估
    from sklearn.ensemble import RandomForestClassifier
    from sklearn.model_selection import cross_val_score
    
    rf = RandomForestClassifier(n_estimators=100, random_state=42)
    
    results = []
    for method_name, method_data in methods.items():
        scores = cross_val_score(rf, method_data['data'], y, cv=5, scoring='accuracy')
        results.append({
            '方法': method_name,
            '维度数': method_data['n_components'],
            '准确率': scores.mean(),
            '标准差': scores.std(),
            '压缩率': 1 - method_data['n_components'] / X.shape[1]
        })
    
    # 原始数据基准
    original_scores = cross_val_score(rf, X_scaled, y, cv=5, scoring='accuracy')
    results.append({
        '方法': '原始数据',
        '维度数': X.shape[1],
        '准确率': original_scores.mean(),
        '标准差': original_scores.std(),
        '压缩率': 0
    })
    
    results_df = pd.DataFrame(results)
    print("\n性能对比结果:")
    print(results_df.round(3))
    
    # 可视化对比
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))
    
    # 维度压缩效果
    axes[0].bar(results_df['方法'], results_df['维度数'])
    axes[0].set_title('维度数对比')
    axes[0].set_ylabel('特征维度')
    axes[0].tick_params(axis='x', rotation=45)
    
    # 分类性能
    axes[1].bar(results_df['方法'], results_df['准确率'], 
                yerr=results_df['标准差'], capsize=5)
    axes[1].set_title('分类准确率')
    axes[1].set_ylabel('准确率')
    axes[1].tick_params(axis='x', rotation=45)
    
    # PCA方差解释
    axes[2].plot(range(1, len(pca.explained_variance_ratio_)+1), 
                np.cumsum(pca.explained_variance_ratio_), 'o-')
    axes[2].axhline(y=0.95, color='red', linestyle='--', label='95%阈值')
    axes[2].set_xlabel('主成分数')
    axes[2].set_ylabel('累积方差解释比例')
    axes[2].set_title('PCA方差解释')
    axes[2].legend()
    axes[2].grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
    
    return results_df, methods
```

## 总结与最佳实践

### 核心技术要点

> [!success] 降维算法核心价值
> **降维是高维数据处理的核心技术**，通过保留关键信息来简化数据结构：
> - **解决维度诅咒**：缓解高维数据带来的计算和统计问题
> - **提升计算效率**：减少存储空间和计算时间复杂度  
> - **增强可解释性**：将复杂高维关系映射到可理解的低维空间
> - **改善模型性能**：去除噪声和冗余特征，提高模型泛化能力

### 算法选择决策

| 场景 | 推荐算法 | 关键考量 |
|------|---------|---------|
| **无监督数据压缩** | PCA | 保留95%方差，线性假设成立 |
| **数据可视化** | t-SNE/UMAP | 结合样本规模、计算资源和邻域保真需求比较t-SNE与UMAP，不以固定样本数作为唯一分界 |
| **监督学习预处理** | LDA | 分类任务，类别数较少时优先 |
| **稀疏高维数据** | TruncatedSVD | 文本数据、推荐系统 |
| **保持特征含义** | 特征选择 | 可解释性要求高的业务场景 |

### 实施最佳实践

1. **数据预处理**
   - PCA前应评估特征量纲和尺度，量纲差异明显时通常需要标准化；LDA并非必须标准化，应结合数据尺度与数值稳定性决定
   - 处理缺失值和异常值
   - 检查数据分布和相关性

2. **维度确定策略**
   ```python
   # PCA维度选择参考
   def optimal_components(pca, threshold=0.95):
       cumvar = np.cumsum(pca.explained_variance_ratio_)
       return np.argmax(cumvar >= threshold) + 1
   ```

3. **性能评估**
   - 信息损失率：重构误差或方差解释比例
   - 下游任务性能：分类准确率、聚类质量
   - 计算效率：训练时间和内存占用

4. **参数调优要点**
   - t-SNE: perplexity=5-50，learning_rate=10-1000
   - PCA: n_components根据方差解释率确定  
   - 特征选择: k值通过交叉验证优化

### 常见陷阱与解决方案

| 问题 | 原因 | 解决方案 |
|------|------|---------|
| **PCA效果差** | 数据非线性相关 | 考虑核PCA或非线性方法 |
| **t-SNE不稳定** | 参数设置不当 | 调整perplexity和learning_rate |
| **特征选择过拟合** | 测试集泄露 | 严格划分训练/验证集 |
| **降维后性能下降** | 丢失关键信息 | 增加保留维度或改用特征选择 |
