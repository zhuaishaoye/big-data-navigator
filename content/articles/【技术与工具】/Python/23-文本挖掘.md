## 文本预处理技术

### 1. 文本清洗与标准化

文本清洗是NLP的基础步骤，包括去除HTML标签、特殊字符、标准化格式等。

```python
import re
import jieba

def text_cleaner(text):
    """文本清洗函数"""
    # 去除HTML标签
    text = re.sub(r'<[^>]+>', '', text)
    # 标准化空白字符
    text = re.sub(r'\s+', ' ', text.strip())
    # 保留中文、英文、数字
    text = re.sub(r'[^一-龥a-zA-Z0-9\s]', '', text)
    # 去除重复字符
    text = re.sub(r'(.)\1{2,}', r'\1\1', text)
    return text.strip()

# 示例
raw_text = "这个产品真的很好用！！！体验<strong>非常</strong>棒@#$"
cleaned = text_cleaner(raw_text)
print(f"清洗前: {raw_text}")
print(f"清洗后: {cleaned}")
# 输出: 清洗后: 这个产品真的很好用体验非常棒
```

### 2. 中文分词技术

**jieba分词原理**：基于前缀词典和动态规划算法，支持精确、全模式、搜索引擎模式。

```python
def chinese_text_processor(text):
    """中文文本预处理管道"""
    # 定义停用词表
    stopwords = {'的', '了', '在', '是', '我', '有', '和', '就', '不', '人'}
    
    # 1. 文本清洗
    text = text_cleaner(text)
    # 2. jieba分词
    words = jieba.lcut(text)
    # 3. 过滤停用词和短词
    filtered_words = [word for word in words 
                     if word not in stopwords and len(word) >= 2]
    return filtered_words

# 分词示例
text = "人工智能技术在自然语言处理领域取得了重大突破"
words = chinese_text_processor(text)
print(f"分词结果: {words}")
# 输出: ['人工智能', '技术', '自然语言', '处理', '领域', '取得', '重大', '突破']
```

### 3. 词干化与词形还原

**技术原理**：
- **词干化(Stemming)**：通过去除前后缀获得词干
- **词形还原(Lemmatization)**：根据词典还原到基本形式

```python
try:
    import nltk
    from nltk.stem import PorterStemmer, WordNetLemmatizer
    
    stemmer = PorterStemmer()
    lemmatizer = WordNetLemmatizer()
    
    words = ['running', 'runs', 'ran', 'easily', 'fairly']
    
    print("词干化结果:")
    for word in words:
        stemmed = stemmer.stem(word)
        lemmatized = lemmatizer.lemmatize(word, pos='v')  # 动词
        print(f"{word} -> {stemmed} (stem) / {lemmatized} (lemma)")
        
except ImportError:
    print("NLTK未安装，跳过英文预处理演示")
```

```python
def bag_of_words_demo():
    """词袋模型演示"""
 # 示例文档
    documents = 
        "机器学习是人工智能的一个分支",
        "深度学习是机器学习的一个子领域",

    print(f"文档-词矩阵形状: {X.shape}")
    
 # 转换为DataFrame便于查看
    df_bow = pd.DataFrame(X.toarray(), columns=feature_names)
    print("\n文档-词矩阵:")
    print(df_bow)
    
    return vectorizer, X, feature_names, df_bow
```

### 2. TF-IDF特征

**TF-IDF向量化：**
```python
def tfidf_demo():
    """TF-IDF演示"""
 # 使用之前的文档
    documents = 
        "机器学习是人工智能的一个分支",
        "深度学习是机器学习的一个子领域",
        "自然语言处理是人工智能的重要应用",
        "计算机视觉和自然语言处理都是AI的应用"
    ]
    
 # 使用TF-IDF向量化
    tfidf_vectorizer = TfidfVectorizer(
        max_features=20,
        token_pattern=r'[一-龥]+',
        use_idf=True,
        smooth_idf=True
    )
    
 # 构建TF-IDF矩阵
    X_tfidf = tfidf_vectorizer.fit_transform(documents)
    
 # 获取特征名称
    feature_names = tfidf_vectorizer.get_feature_names_out()
    
    print("TF-IDF结果:")
    print(f"特征数量: {len(feature_names)}")
    print(f"特征名称: {feature_names}")
    print(f"文档-特征矩阵形状: {X_tfidf.shape}")
    
 # 转换为DataFrame
    df_tfidf = pd.DataFrame(X_tfidf.toarray(), columns=feature_names)
    print("\nTF-IDF矩阵:")
    print(df_tfidf.round(3))
    
    return tfidf_vectorizer, X_tfidf, feature_names, df_tfidf
```

## 🧬 特征工程与向量化

### 1. 常见文本特征表示方法

| 方法         | 原理简介                         | 优点                   | 缺点                   | 适用场景           |
|---|---|---|---|---|
| One-hot      | 每个词独立编码，稀疏向量         | 简单直观               | 维度高，稀疏，无法表达语义 | 小型词表、简单场景   |
| Bag of Words | 统计词频，忽略顺序               | 实现简单，效果稳定      | 丢失语序，稀疏          | 文本分类、聚类      |
| TF-IDF       | 词频-逆文档频率加权              | 兼顾词频与区分度        | 稀疏，无法表达上下文      | 信息检索、文本分类  |
| Word2Vec     | 词嵌入，低维稠密向量             | 能表达词语语义关系      | 需大量语料，忽略上下文    | 语义分析、聚类      |
| BERT         | 预训练深度模型，动态上下文表示    | 语义丰富，效果强大      | 计算资源消耗大            | 高级NLP任务         |

## 🏷️ 文本分类实战

### 1. 任务简介
文本分类是将文本自动分配到一个或多个类别的过程，常见于新闻分类、垃圾邮件识别、情感分类等。

### 2. 完整流程与代码示例

```python
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import classification_report, confusion_matrix

# 示例数据
texts = 
    "人工智能正在改变世界",
    "机器学习和深度学习是AI的核心",
    "垃圾邮件太多了，影响正常工作",
    "深度学习推动图像识别进步",
    "收到很多广告短信，很烦"
]
labels = ["科技", "科技", "垃圾", "科技", "垃圾"]

# 1. 特征提取
vectorizer = TfidfVectorizer()
X = vectorizer.fit_transform(texts)

# 2. 划分训练集和测试集
X_train, X_test, y_train, y_test = train_test_split(X, labels, test_size=0.4, random_state=42)

# 3. 训练模型
clf = MultinomialNB()
clf.fit(X_train, y_train)

# 4. 预测与评估
y_pred = clf.predict(X_test)
print("分类报告：\n", classification_report(y_test, y_pred))
print("混淆矩阵：\n", confusion_matrix(y_test, y_pred))

# 5. 新文本预测
new_texts = ["收到中奖信息，是真的吗？", "深度学习在医疗领域有新突破"]
X_new = vectorizer.transform(new_texts)
preds = clf.predict(X_new)
for text, label in zip(new_texts, preds):
    print(f"文本: {text} => 预测类别: {label}")
```

### 3. 实战建议
- 中文文本分类建议先用TF-IDF+朴素贝叶斯/[逻辑回归]，复杂场景可用BERT等深度模型。
- 特征工程和分词质量对分类效果影响极大。
- 可用GridSearchCV等工具调参提升模型表现。

```python
def topic_modeling_demo():
    """主题建模演示"""
 # 创建示例文档集合
    documents = 
        "人工智能机器学习深度学习神经网络",
        "机器学习算法监督学习无监督学习",
        "深度学习卷积神经网络循环神经网络",
        "自然语言处理文本分析语音识别",
        "计算机视觉图像识别目标检测",
        "数据挖掘聚类分析关联规则",
        "大数据处理分布式计算云计算",
        "数据库SQL查询数据仓库",
        "网络编程Web开发移动应用",
        "操作系统Linux内核进程管理",
        "软件工程设计模式测试方法",
        "算法数据结构排序算法搜索算法",
        "数学统计概率论线性代数",
        "物理力学电磁学量子力学",
        "化学有机化学无机化学分析化学",
        "生物基因DNA蛋白质细胞",
        "医学诊断治疗药物疾病",
        "经济金融投资股票市场",
        "政治国际关系外交政策",
        "历史古代文明现代史"
    ]
    
 # 特征提取
    vectorizer = CountVectorizer(
        max_features=50,
        token_pattern=r'[一-龥]+',
        stop_words=None
    )
    
    X = vectorizer.fit_transform(documents)
    feature_names = vectorizer.get_feature_names_out()
    
 # LDA主题建模
    n_topics = 5
    lda = LatentDirichletAllocation(
        n_components=n_topics,
        random_state=42,
        max_iter=100
    )
    
    lda.fit(X)
    
 # 获取主题-词分布
    topic_word_dist = lda.components_
    
 # 获取文档-主题分布
    doc_topic_dist = lda.transform(X)
    
    print("LDA主题建模结果:")
    print(f"主题数量: {n_topics}")
    print(f"词汇表大小: {len(feature_names)}")
    
 # 显示每个主题的top词
    n_top_words = 5
    for topic_idx, topic in enumerate(topic_word_dist):
        top_words_idx = topic.argsort()[-n_top_words:][::-1]
        top_words = [feature_names[i] for i in top_words_idx]
        print(f"主题 {topic_idx + 1}: {', '.join(top_words)}")
    
    return lda, vectorizer, feature_names, doc_topic_dist
```

## 🚀 实战案例：评论情感分析

### 案例背景

分析电商平台用户评论的情感倾向，为产品改进提供数据支持。

```python
def ecommerce_sentiment_analysis():
    """电商评论情感分析案例"""
 # 创建模拟电商评论数据
    np.random.seed(42)
    n_reviews = 1000
    
 # 生成评论数据
    positive_templates = [
        "产品质量很好，{feature}很满意",
        "物流很快，{feature}不错",
        "客服态度很好，{feature}推荐",
        "价格实惠，{feature}值得购买",
        "包装很好，{feature}超出预期"
    ]
    
    negative_templates = [
        "质量很差，{feature}很失望",
        "物流太慢，{feature}不推荐",
        "客服态度差，{feature}很生气",
        "价格太贵，{feature}不值",
        "包装破损，{feature}很糟糕"
    ]
    
    neutral_templates = [
        "产品一般，{feature}还可以",
        "物流正常，{feature}凑合",
        "客服一般，{feature}还行",
        "价格适中，{feature}一般",
        "包装普通，{feature}正常"
    ]
    
    features = ['外观', '功能', '性能', '服务', '体验', '性价比', '耐用性', '实用性']
    
    reviews = []
    sentiments = []
    
 # 生成评论
    for i in range(n_reviews):
        if i < 400:  # 40% 正面
            template = np.random.choice(positive_templates)
            sentiment = 'positive'
        elif i < 700:  # 30% 负面
            template = np.random.choice(negative_templates)
            sentiment = 'negative'
        else:  # 30% 中性
            template = np.random.choice(neutral_templates)
            sentiment = 'neutral'
        
        feature = np.random.choice(features)
        review = template.format(feature=feature)
        
        reviews.append(review)
        sentiments.append(sentiment)
    
 # 创建数据框
    df = pd.DataFrame({
        'review': reviews,
        'sentiment': sentiments
    })
    
    print("电商评论情感分析数据:")
    print(f"评论数量: {len(df)}")
    print(f"情感分布:\n{df['sentiment'].value_counts()}")
    print(f"情感比例:\n{df['sentiment'].value_counts(normalize=True)}")
    
    return df
```
