> - 🎯 **学习目标**：像教练评估运动员一样系统评估Python技能

import matplotlib.pyplot as plt
import matplotlib.patches as patches
from matplotlib.patches import FancyBboxPatch
import numpy as np

class PythonSkillMap:
    """Python技能地图可视化"""
    
    def __init__(self):
        self.skill_levels = {
            '入门': ['基础语法', '数据结构', '函数编程', '面向对象'],
            '进阶': ['NumPy', 'Pandas', '数据可视化', '文件操作'],
            '应用': ['机器学习', '数据分析', 'Web开发', '自动化'],
            '专业': ['深度学习', '大数据', '模型部署', '项目管理']
        }
        
        self.skill_assessment = {}
    
    def create_skill_map(self):
        """创建技能地图"""
        
        fig, ax = plt.subplots(figsize=(12, 8))
        
 # 颜色配置
        colors = {
            '入门': '#E8F4FD',
            '进阶': '#B8E6B8', 
            '应用': '#FFD93D',
            '专业': '#FF6B6B'
        }
        
        y_positions = {'入门': 0.7, '进阶': 0.5, '应用': 0.3, '专业': 0.1}
        
        for level, skills in self.skill_levels.items():
            y_pos = y_positions[level]
            
 # 绘制级别标题
            level_box = FancyBboxPatch(
                (0.05, y_pos), 0.15, 0.08,
                boxstyle="round,pad=0.01",
                facecolor=colors[level],
                edgecolor='black',
                linewidth=2
            )
            ax.add_patch(level_box)
            ax.text(0.125, y_pos + 0.04, level, ha='center', va='center', 
                   fontsize=12, fontweight='bold')
            
 # 绘制技能点
            for i, skill in enumerate(skills):
                x_pos = 0.25 + i * 0.15
                
 # 技能圆圈
                circle = patches.Circle((x_pos, y_pos + 0.04), 0.03, 
                                      facecolor='white', edgecolor=colors[level], 
                                      linewidth=2)
                ax.add_patch(circle)
                
 # 技能名称
                ax.text(x_pos, y_pos - 0.02, skill, ha='center', va='top', 
                       fontsize=9, rotation=45)
                
 # 连接线
                if i > 0:
                    ax.plot([x_pos - 0.15, x_pos - 0.03], 
                           [y_pos + 0.04, y_pos + 0.04], 
                           color='gray', alpha=0.5, linewidth=1)
        
 # 级别间连接线
        for i, level in enumerate(list(y_positions.keys())[:-1]):
            y1 = y_positions[level]
            y2 = y_positions[list(y_positions.keys())[i + 1]]
            ax.plot([0.125, 0.125], [y1, y2 + 0.08], 
                   color='gray', alpha=0.5, linewidth=2)
        
        ax.set_xlim(0, 1)
        ax.set_ylim(0, 0.85)
        ax.set_title('Python学习技能地图', fontsize=16, fontweight='bold', pad=20)
        ax.axis('off')
        
        plt.tight_layout()
        plt.show()
        
        return fig

def sum_even_numbers(numbers):
    """

    def run_basic_assessment(self):
        """运行基础评估"""
        
        print("📝 基础阶段技能评估")
        print("=" * 20)
        
 # 理论知识测试
        theory_score = self._test_theoretical_knowledge()
        
 # 编程实践测试
        practice_score = self._test_coding_skills()
        
 # 综合评分
        overall_score = (theory_score + practice_score) / 2
        
 # 生成评估报告
        return self._generate_assessment_report(overall_score, theory_score, practice_score)
    
    def _test_theoretical_knowledge(self):
        """理论知识测试"""
        
        print("\n📚 理论知识测试:")
        correct_count = 0
        total_count = len(self.questions['python_basics'])
        
        for i, q in enumerate(self.questions['python_basics'], 1):
            print(f"\n{i}. {q['question']}")
            for option in q['options']:
                print(f"   {option}")
            
 # 模拟答题 (实际应用中可以要求用户输入)
            import random
            simulated_answer = random.choice(['A', 'B', 'C', 'D'])
            
            if simulated_answer == q['answer']:
                correct_count += 1
                print(f"   ✅ 正确! {q['explanation']}")
            else:
                print(f"   ❌ 错误. 正确答案: {q['answer']} - {q['explanation']}")
        
        score = (correct_count / total_count) * 100
        print(f"\n📊 理论知识得分: {score:.1f}% ({correct_count}/{total_count})")
        
        return score
    
    def _test_coding_skills(self):
        """编程技能测试"""
        
        print("\n💻 编程实践测试:")
        
        task = self.questions['coding_practice']
        print(f"任务: {task['task']}")
        print("\n代码模板:")
        print(task['template'])
        
 # 展示解决方案
        print("\n参考解决方案:")
        print(task['solution'])
        
 # 测试解决方案
        print("\n🧪 测试结果:")
        solution_namespace = {}
        exec(task['solution'], solution_namespace)  # 执行文章内的参考解法，仅演示测试流程
        
        passed_tests = 0
        for test_input, expected in task['test_cases']:
            try:
                result = solution_namespace['sum_even_numbers'](test_input)
                if result == expected:
                    print(f"   ✅ 测试通过: {test_input} -> {result}")
                    passed_tests += 1
                else:
                    print(f"   ❌ 测试失败: {test_input} -> {result} (期望: {expected})")
            except Exception as e:
                print(f"   ❌ 运行错误: {e}")
        
        score = (passed_tests / len(task['test_cases'])) * 100
        print(f"\n📊 编程实践得分: {score:.1f}% ({passed_tests}/{len(task['test_cases'])})")
        
        return score
    
    def _generate_assessment_report(self, overall_score, theory_score, practice_score):
        """生成评估报告"""
        
 # 等级评定
        if overall_score >= 90:
            level = "优秀"
            recommendation = "基础扎实，可以进入进阶学习"
        elif overall_score >= 80:
            level = "良好"
            recommendation = "基础较好，建议加强编程实践"
        elif overall_score >= 70:
            level = "中等"
            recommendation = "需要加强基础概念理解和练习"
        elif overall_score >= 60:
            level = "及格"
            recommendation = "建议重新学习基础部分，多做练习"
        else:
            level = "不及格"
            recommendation = "需要系统重新学习Python基础"
        
        report = f"""
        📋 基础阶段评估报告
        -----------------------------------

        📊 综合得分: {overall_score:.1f}分
        📚 理论知识: {theory_score:.1f}分
        💻 编程实践: {practice_score:.1f}分
        🏷️ 技能等级: {level}

        💡 学习建议:
        • {recommendation}
        • 建议每天练习编程1-2小时
        • 重点关注: {'理论知识' if theory_score < practice_score else '编程实践'}

        📖 推荐学习资源:
        • 复习文档: Python基本概念语法
        • 练习项目: Python数据结构
        • 进阶准备: Python函数编程
        """
        
        print(report)
        return {
            'overall_score': overall_score,
            'theory_score': theory_score,
            'practice_score': practice_score,
            'level': level,
            'recommendation': recommendation
        }

class PersonalizedLearningPlanner:
    """个性化学习计划生成器"""
    
    def __init__(self):
        self.learning_modules = {

            },
            '数据可视化': {
                'modules': ['13-Matplotlib基础', '14-Seaborn统计图', '15-Plotly交互图表'],
                'estimated_hours': 18,
                'difficulty': '进阶',
                'prerequisites': ['数据处理']
            },
            '机器学习': {
                'modules': ['18-机器学习基础', '20-回归算法', '21-分类算法'],
                'estimated_hours': 25,
                'difficulty': '中高级',
                'prerequisites': ['数据处理', '数据可视化']
            },
            '深度学习': {
                'modules': ['28-深度学习入门', '29-高级分析技术'],
                'estimated_hours': 30,
                'difficulty': '高级',
                'prerequisites': ['机器学习']
            },
            '工程实践': {
                'modules': ['30-Flask Web开发', '32-模型部署', '33-系统监控'],
                'estimated_hours': 22,
                'difficulty': '中高级',
                'prerequisites': ['机器学习']
            }
        }
        
        self.user_profile = {}
    
    def assess_user_profile(self, goals, available_hours_per_week, current_level, interests):
        """评估用户画像"""
        
        self.user_profile = {
            'goals': goals,  # 'job_ready', 'academic', 'hobby', 'career_change'
            'available_hours': available_hours_per_week,
            'current_level': current_level,  # 'beginner', 'intermediate', 'advanced'
            'interests': interests,  # ['data_analysis', 'machine_learning', 'web_dev', 'automation']
            'learning_style': self._detect_learning_style(available_hours_per_week)
        }
        
        return self.user_profile
    
    def _detect_learning_style(self, hours_per_week):
        """检测学习风格"""
        if hours_per_week >= 20:
            return 'intensive'  # 密集型学习
        elif hours_per_week >= 10:
            return 'regular'    # 常规学习
        else:
            return 'casual'     # 轻松学习
    
    def generate_learning_path(self):
        """生成个性化学习路径"""
        
        if not self.user_profile:
            raise ValueError("请先完成用户画像评估")
        
 # 根据目标和兴趣筛选模块
        recommended_modules = self._select_modules_by_interest()
        
 # 根据当前水平调整起点
        starting_modules = self._adjust_starting_point()
        
 # 生成时间规划
        timeline = self._create_timeline(recommended_modules)
        
 # 生成详细计划
        detailed_plan = self._create_detailed_plan(timeline)
        
        return detailed_plan
    
    def _select_modules_by_interest(self):
        """根据兴趣选择模块"""
        
        interest_mapping = {
            'data_analysis': ['数据处理', '数据可视化'],
            'machine_learning': ['机器学习', '深度学习'],
            'web_dev': ['工程实践'],
            'automation': ['基础巩固', '工程实践']
        }
        
        recommended = set()
        
 # 基础模块总是需要的
        recommended.add('基础巩固')
        
 # 根据兴趣添加模块
        for interest in self.user_profile['interests']:
            if interest in interest_mapping:
                recommended.update(interest_mapping[interest])
        
        return list(recommended)
    
    def _adjust_starting_point(self):
        """根据当前水平调整起点"""
        
        level = self.user_profile['current_level']
        
        if level == 'beginner':
            return ['基础巩固']
        elif level == 'intermediate':
            return ['数据处理']
        else:  # advanced
            return ['机器学习']
    
    def _create_timeline(self, modules):
        """创建时间规划"""
        
        weekly_hours = self.user_profile['available_hours']
        learning_style = self.user_profile['learning_style']
        
 # 调整学习强度
        intensity_multiplier = {
            'intensive': 1.0,
            'regular': 1.2,
            'casual': 1.5
        }
        
        timeline = []
        current_week = 1
        
        for module_name in modules:
            module = self.learning_modules[module_name]
            
 # 计算需要的周数
            adjusted_hours = module['estimated_hours'] * intensity_multiplier[learning_style]
            weeks_needed = max(1, int(np.ceil(adjusted_hours / weekly_hours)))
            
            timeline.append({
                'module': module_name,
                'start_week': current_week,
                'end_week': current_week + weeks_needed - 1,
                'estimated_hours': adjusted_hours,
                'difficulty': module['difficulty']
            })
            
            current_week += weeks_needed
        
        return timeline
    
    def _create_detailed_plan(self, timeline):
        """创建详细学习计划"""
        
        plan = {
            'user_profile': self.user_profile,
            'total_weeks': max(item['end_week'] for item in timeline),
            'timeline': timeline,
            'weekly_schedule': self._create_weekly_schedule(),
            'milestones': self._create_milestones(timeline),
            'resources': self._recommend_resources()
        }
        
        return plan
    
    def _create_weekly_schedule(self):

        hours = self.user_profile['available_hours']
        
        if hours >= 20:  # 全职学习

                '周二': '编程练习 (4小时)',
                '周三': '项目实践 (4小时)',
                '周四': '理论学习 (3小时)',
                '周五': '编程练习 (4小时)',
                '周六': '项目实践 (2小时)',
                '周日': '复习总结 (1小时)'
            }
        elif hours >= 10:  # 兼职学习
            schedule = {
                '周一': '理论学习 (1.5小时)',
                '周三': '编程练习 (2小时)',
                '周五': '项目实践 (1.5小时)',
                '周六': '理论学习 (2小时)',
                '周日': '复习总结 (1小时)'
            }
        else:  # 休闲学习
            schedule = {
                '周三': '理论学习 (1小时)',
                '周六': '编程练习 (1.5小时)',
                '周日': '项目实践 (1小时)'
            }
        
        return schedule
    
    def _create_milestones(self, timeline):
        """创建学习里程碑"""
        
        milestones = []
        
        for item in timeline:
            milestone = {
                'week': item['end_week'],
                'achievement': f"完成{item['module']}模块",
                'skills': f"掌握{item['difficulty']}难度技能",
                'assessment': f"通过阶段性测试"
            }
            milestones.append(milestone)
        
        return milestones
    
    def _recommend_resources(self):
        """推荐学习资源"""
        
        resources = {
            'official_docs': [
                'Python学习路线图',
                '阶段自测与练习'
            ],
            'practice_platforms': [
                'LeetCode - 算法练习',
                'Kaggle - 数据科学竞赛',
                'GitHub - 开源项目学习'
            ],
            'communities': [
                 'Python中文社区',
                'Stack Overflow',
                '掘金技术社区'
            ]
        }
        
        return resources

        """展示学习计划"""
        
        print("🎯 个性化Python学习计划")
        print("=" * 30)
        
        print(f"\n👤 学习者画像:")

            print(f"  第{item['start_week']}-{item['end_week']}周: {item['module']} "

        print(f"\n📚 每周学习安排:")
        for day, activity in plan['weekly_schedule'].items():
