## 📐 回归模型的基本假设

### 线性回归的核心假设

> [!note] 🔍 **回归分析的五大假设**
> 
> 1. **条件均值与线性设定**：模型对参数线性，条件均值由所选自变量及变换项表示，并要求E[误差|自变量]=0；这不等于每个原始自变量与因变量都呈直线关系。
> 2. **误差相关结构**：常规独立样本推断需要相应的独立性条件；时间序列或聚类数据需要适配其误差相关结构，不能仅凭残差图或DW经验区间确认独立性。
> 3. **同方差性**：经典同方差标准误要求误差的条件方差恒定；异方差不必然使OLS系数失去有效性，但通常需要调整标准误或估计方法。
> 4. **正态性**：条件误差正态性主要支持经典有限样本精确推断，并非拟合线性回归或所有大样本推断的统一必要条件；拟合残差只是误差的诊断代理。
> 5. **无完全多重共线性**：设计矩阵需满列秩才能唯一估计系数；高但非完全共线性主要影响估计精度与稳定性，不是禁止拟合的统一条件。

def comprehensive_residual_analysis():
    """
    全面的残差分析
    """
    print("\n=== 全面残差分析 ===")
    
 # 生成示例数据：房价预测
    np.random.seed(123)
    n = 150
    
 # 自变量：面积、楼层、房龄
    area = np.random.normal(100, 30, n)
    floor = np.random.randint(1, 31, n)
    age = np.random.exponential(5, n)
    
 # 因变量：房价（万元），含一些异常值
    price = (2.5 * area + 0.3 * floor - 0.8 * age + 50 + 
             np.random.normal(0, 20, n))
    
 # 添加几个异常值
    outlier_indices = [10, 25, 87, 134]
    for idx in outlier_indices:
        price[idx] += np.random.choice([-80, 80])  # 价格异常
    
 # 创建数据框
    df = pd.DataFrame({
        'area': area,
        'floor': floor, 
        'age': age,
        'price': price
    })
    
    print("房价预测模型数据概况:")
    print(df.describe().round(2))
    
 # 拟合多元线性回归
    X = df[['area', 'floor', 'age']]
    y = df['price']
    
 # 添加常数项
    X_with_const = sm.add_constant(X)
    model = sm.OLS(y, X_with_const).fit()
    
    print(f"\n模型拟合结果:")
    print(model.summary())
    
 # 计算各种残差
    fitted_values = model.fittedvalues
    residuals = model.resid  # 普通残差
    standardized_residuals = model.resid_pearson  # 标准化残差
    studentized_residuals = model.get_influence().resid_studentized_internal  # 学生化残差
    
 # 计算其他诊断统计量
    influence = model.get_influence()
    leverage = influence.hat_matrix_diag  # 杠杆值
    cooks_distance = influence.cooks_distance[0]  # Cook距离
    
    print(f"\n残差统计摘要:")
    print(f"残差均值: {np.mean(residuals):.6f}")
    print(f"残差标准差: {np.std(residuals):.3f}")
    print(f"最大标准化残差: {np.max(np.abs(standardized_residuals)):.3f}")
    print(f"最大杠杆值: {np.max(leverage):.3f}")
    print(f"最大Cook距离: {np.max(cooks_distance):.3f}")
    
 # 识别异常点
    print(f"\n异常点识别:")
    
 # 1. 高残差点（|标准化残差| > 2）
    high_residual_idx = np.where(np.abs(standardized_residuals) > 2)[0]
    print(f"高残差点 (|标准化残差|>2): {len(high_residual_idx)}个")
    if len(high_residual_idx) > 0:
        print(f"   索引: {high_residual_idx}")
    
 # 2. 高杠杆点（杠杆值 > 2(p+1)/n）
    p = X.shape[1]  # 自变量个数
    leverage_threshold = 2 * (p + 1) / n
    high_leverage_idx = np.where(leverage > leverage_threshold)[0]
    print(f"高杠杆点 (杠杆值>{leverage_threshold:.3f}): {len(high_leverage_idx)}个")
    if len(high_leverage_idx) > 0:
        print(f"   索引: {high_leverage_idx}")
    
 # 3. 高影响点（Cook距离 > 4/n）
    cook_threshold = 4 / n
    high_influence_idx = np.where(cooks_distance > cook_threshold)[0]
    print(f"高影响点 (Cook距离>{cook_threshold:.3f}): {len(high_influence_idx)}个")
    if len(high_influence_idx) > 0:
        print(f"   索引: {high_influence_idx}")
    
    return df, model, {
        'fitted': fitted_values,
        'residuals': residuals,
        'std_residuals': standardized_residuals,
        'stud_residuals': studentized_residuals,
        'leverage': leverage,
        'cooks_d': cooks_distance
    }

# 执行残差分析
df_housing, housing_model, residual_data = comprehensive_residual_analysis()

def multicollinearity_diagnosis():
    """
    多重共线性诊断
    """
    print("\n=== 多重共线性诊断 ===")
    
 # 创建包含共线性的数据
    np.random.seed(456)
    n = 200
    
 # 生成高度相关的自变量
    x1 = np.random.normal(0, 1, n)
    x2 = 0.8 * x1 + np.random.normal(0, 0.6, n)  # 与x1高度相关
    x3 = np.random.normal(0, 1, n)
    x4 = 2 * x1 + 3 * x2 + np.random.normal(0, 0.1, n)  # 几乎完全由x1和x2决定
    x5 = np.random.normal(0, 1, n)
    
 # 因变量
    y = 2 + 1.5*x1 + 2*x2 + 0.5*x3 - 1*x4 + 0.8*x5 + np.random.normal(0, 1, n)
    
 # 创建数据框
    df_mc = pd.DataFrame({
        'x1': x1, 'x2': x2, 'x3': x3, 'x4': x4, 'x5': x5, 'y': y
    })
    
    print("变量间相关系数矩阵:")
    correlation_matrix = df_mc.corr()
    print(correlation_matrix.round(3))
    
 # 计算VIF
    X = df_mc[['x1', 'x2', 'x3', 'x4', 'x5']]
    
    print(f"\n方差膨胀因子(VIF):")
    print(f"{'变量':<8} {'VIF':<10} {'诊断结果':<20}")
    print("-" * 40)
    
    vif_data = []
    for i in range(X.shape[1]):
        vif = variance_inflation_factor(X.values, i)
        vif_data.append((X.columns[i], vif))
        
 # VIF诊断标准
        if vif < 5:
            diagnosis = "无共线性问题"
        elif vif < 10:
            diagnosis = "中等共线性"
        else:
            diagnosis = "严重共线性"
            
        print(f"{X.columns[i]:<8} {vif:<10.2f} {diagnosis:<20}")
    
 # 特征值诊断
    print(f"\n相关矩阵特征值诊断:")
    eigenvalues = np.linalg.eigvals(X.corr())
    condition_indices = np.sqrt(eigenvalues.max() / eigenvalues)
    
    print(f"{'特征值':<10} {'条件指数':<10} {'诊断':<15}")
    print("-" * 35)
    
    for i, (eigenval, ci) in enumerate(zip(eigenvalues, condition_indices)):
        if ci < 10:
            diagnosis = "无问题"
        elif ci < 30:
            diagnosis = "中度共线性"
        else:
            diagnosis = "严重共线性"
            
        print(f"{eigenval:<10.4f} {ci:<10.2f} {diagnosis:<15}")
    
 # 展示共线性影响
    print(f"\n共线性对回归结果的影响:")
    
 # 完整模型
    X_full = sm.add_constant(X)
    model_full = sm.OLS(y, X_full).fit()
    
 # 移除高VIF变量后的模型
    X_reduced = X[['x1', 'x3', 'x5']]  # 移除x2和x4
    X_reduced_const = sm.add_constant(X_reduced)
    model_reduced = sm.OLS(y, X_reduced_const).fit()
    
    print(f"\n模型比较:")
    print(f"{'模型':<15} {'R²':<8} {'调整R²':<10} {'AIC':<10}")
    print("-" * 45)
    print(f"{'完整模型':<15} {model_full.rsquared:<8.3f} {model_full.rsquared_adj:<10.3f} {model_full.aic:<10.1f}")
    print(f"{'简化模型':<15} {model_reduced.rsquared:<8.3f} {model_reduced.rsquared_adj:<10.3f} {model_reduced.aic:<10.1f}")
    
 # 可视化相关性矩阵
    plt.figure(figsize=(10, 8))
    mask = np.triu(np.ones_like(correlation_matrix, dtype=bool))
    sns.heatmap(correlation_matrix, mask=mask, annot=True, cmap='coolwarm', 
                center=0, square=True, linewidths=0.5, cbar_kws={"shrink": .8})
    plt.title('变量相关性热图')
    plt.tight_layout()
    plt.show()
    
    return df_mc, vif_data, model_full, model_reduced

# 执行多重共线性诊断

def normality_tests(residuals):
    """
    残差正态性检验
    """

 # 3. Kolmogorov-Smirnov检验
    ks_stat, ks_p = stats.kstest(residuals, 'norm', args=(np.mean(residuals), np.std(residuals)))
    print(f"\nKolmogorov-Smirnov检验:")
    print(f"  统计量: {ks_stat:.4f}")
    print(f"  p值: {ks_p:.6f}")
    print(f"  结论: {'拒绝正态性' if ks_p < 0.05 else '接受正态性'} (α=0.05)")
    
 # 4. Jarque-Bera检验

def heteroscedasticity_tests(model, df):
    """
    异方差检验
    """
    print("\n=== 异方差检验 ===")
    
 # 1. Breusch-Pagan检验
    bp_stat, bp_p, bp_f_stat, bp_f_p = het_breuschpagan(model.resid, model.model.exog)
    print(f"Breusch-Pagan检验:")
    print(f"  LM统计量: {bp_stat:.4f}")
    print(f"  p值: {bp_p:.6f}")
    print(f"  结论: {'存在异方差' if bp_p < 0.05 else '同方差'} (α=0.05)")
    
 # 2. White检验
    white_stat, white_p, white_f_stat, white_f_p = het_white(model.resid, model.model.exog)
    print(f"\nWhite检验:")
    print(f"  LM统计量: {white_stat:.4f}")
    print(f"  p值: {white_p:.6f}")
    print(f"  结论: {'存在异方差' if white_p < 0.05 else '同方差'} (α=0.05)")
    
 # 3. 戈德菲尔德-匡特检验（针对特定变量）
    if 'area' in df.columns:
 # 按面积排序后分组检验
        df_sorted = df.sort_values('area')
        n = len(df_sorted)
        n1 = n // 3
        n2 = n - n1
        
 # 前1/3和后1/3的方差比较
        group1 = df_sorted.iloc[:n1]
        group2 = df_sorted.iloc[-n1:]
        
 # 分别拟合回归
        X1 = sm.add_constant(group1[['area', 'floor', 'age']])
        X2 = sm.add_constant(group2[['area', 'floor', 'age']])
        
        model1 = sm.OLS(group1['price'], X1).fit()
        model2 = sm.OLS(group2['price'], X2).fit()
        
 # 计算F统计量（方差比）
        var1 = model1.ssr / model1.df_resid
        var2 = model2.ssr / model2.df_resid
        f_stat = var2 / var1 if var2 > var1 else var1 / var2
        
 # 计算p值
        df1 = model1.df_resid
        df2 = model2.df_resid
        p_value = 2 * (1 - stats.f.cdf(f_stat, max(df1, df2), min(df1, df2)))
        
        print(f"\nGoldfeld-Quandt检验 (基于面积):")
        print(f"  F统计量: {f_stat:.4f}")
        print(f"  p值: {p_value:.6f}")
        print(f"  结论: {'存在异方差' if p_value < 0.05 else '同方差'} (α=0.05)")
    
 # 异方差性综合评价
    tests_results = [bp_p < 0.05, white_p < 0.05]
    hetero_count = sum(tests_results)
    
    print(f"\n📊 异方差性综合评价:")
    print(f"检测出异方差的测试: {hetero_count}/2")

def outlier_treatment_strategies():
    """
    异常值处理策略演示
    """
    print("\n=== 异常值处理策略 ===")
    
 # 使用之前的住房数据
    df_original = df_housing.copy()

    influence = housing_model.get_influence()
    cooks_d = influence.cooks_distance[0]
    leverage = influence.hat_matrix_diag
    std_residuals = housing_model.resid_pearson
    
 # 定义异常值标准
    cook_threshold = 4 / len(df_original)
    leverage_threshold = 2 * (3 + 1) / len(df_original)  # 3个自变量
    residual_threshold = 2
    
 # 综合异常值识别
    outliers_cook = cooks_d > cook_threshold
    outliers_leverage = leverage > leverage_threshold
    outliers_residual = np.abs(std_residuals) > residual_threshold
    
 # 组合标准：满足任意两个条件
    outliers_combined = (outliers_cook.astype(int) + 
                        outliers_leverage.astype(int) + 
                        outliers_residual.astype(int)) >= 2
    
    outlier_indices = np.where(outliers_combined)[0]
    
    print(f"识别出的异常值数量: {len(outlier_indices)}")
    print(f"异常值索引: {outlier_indices}")
    
 # 策略1: 移除异常值
    df_removed = df_original.drop(df_original.index[outlier_indices])
    X_removed = sm.add_constant(df_removed[['area', 'floor', 'age']])
    model_removed = sm.OLS(df_removed['price'], X_removed).fit()
    
 # 策略2: Winsorizing（缩尾处理）
    from scipy.stats import mstats

    for col in ['area', 'price']:

    model_winsor = sm.OLS(df_winsor['price'], X_winsor).fit()

    from sklearn.linear_model import HuberRegressor

 # 计算阈值
    cook_threshold = 4 / len(df)
    leverage_threshold = 2 * (len(X_cols) + 1) / len(df)
    
 # 统计异常点
    high_cook = np.sum(cooks_d > cook_threshold)
    high_leverage = np.sum(leverage > leverage_threshold)
    high_residual = np.sum(np.abs(std_residuals) > 2)
    
    print(f"高Cook距离点 (>{cook_threshold:.3f}): {high_cook}个")
    print(f"高杠杆点 (>{leverage_threshold:.3f}): {high_leverage}个")
    print(f"高残差点 (|标准化残差|>2): {high_residual}个")
    
    outliers_acceptable = (high_cook + high_leverage + high_residual) < len(df) * 0.05  # 少于5%
    print(f"综合结论: {'✅ 异常点数量可接受' if outliers_acceptable else '⚠️ 异常点较多，需要关注'}")
    results['outliers'] = outliers_acceptable
    
 # 8. 总体评估
    print("\n8️⃣ 总体模型评估")
    print("-" * 30)
    
    passed_assumptions = sum([
        results['linearity'],
        results['independence'], 
        results['homoscedasticity'],
        results['normality'],
        results['multicollinearity'],
        results['outliers']
    ])
    
    total_assumptions = 6
    pass_rate = passed_assumptions / total_assumptions
    
    print(f"通过的假设检验: {passed_assumptions}/{total_assumptions} ({pass_rate*100:.1f}%)")
    
    if pass_rate >= 0.8:
        overall_conclusion = "示例规则下诊断通过率较高；不代表模型可靠，仍需结合业务与独立验证评估"
        color = "green"
    elif pass_rate >= 0.6:
        overall_conclusion = "示例规则下部分诊断触发警示；是否可用需结合用途和具体问题判断"
        color = "yellow"
    else:
        overall_conclusion = "示例规则下较多诊断触发警示；不能仅凭通过率认定必须重新建模"
        color = "red"
    
    print(f"\n🎯 最终结论: {overall_conclusion}")
    
 # 9. 改进建议
    print("\n9️⃣ 改进建议")
    print("-" * 30)
    
    suggestions = []
    
    if not results['linearity']:
        suggestions.append("• 考虑添加多项式项或使用非线性回归")
    
    if not results['independence']:
        suggestions.append("• 检查数据收集过程，考虑时间序列模型")
    
    if not results['homoscedasticity']:
        suggestions.append("• 使用稳健标准误或对因变量进行变换")
    
    if not results['normality']:
        suggestions.append("• 考虑Box-Cox变换或使用非参数方法")
    
    if not results['multicollinearity']:
        suggestions.append("• 移除高度相关的变量或使用正则化方法")
    
    if not results['outliers']:
        suggestions.append("• 仔细检查异常值，考虑稳健回归方法")
    
    if suggestions:

    return results

## ⚠️ 注意事项与最佳实践

> [!warning] 🚨 **回归诊断注意事项**
> 
> **诊断顺序很重要：**
> 1. 先检查线性关系和异常值
> 2. 再检查残差的分布特性
> 3. 最后检查多重共线性
> 
> **常见误区：**
> - 过度依赖单一诊断指标
> - 忽视业务背景和数据收集过程
> - 盲目追求完美的统计假设
> - 不考虑样本量对检验结果的影响
> 
> **最佳实践：**
> - 结合图形诊断和统计检验
> - 优先考虑实际意义而非统计显著性
> - 记录所有诊断步骤和决策过程
> - 对关键模型进行交叉验证
