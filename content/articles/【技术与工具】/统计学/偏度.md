## 🤔 什么是偏度？

> [!note] 📖 **偏度的定义**
> 偏度是衡量数据分布不对称程度的统计量，描述分布相对于对称分布（如正态分布）的偏斜方向和程度。

**数学表达式**：
$$\text{Skewness} = \frac{E[(X-\mu)^3]}{\sigma^3} = \frac{\mu_3}{\sigma^3}$$

其中：
- $\mu_3$ 是三阶中心矩
- $\sigma$ 是标准差
- $E[·]$ 表示期望值

**偏度值的含义**：
- **偏度 = 0**：三阶中心矩为零，但不必然说明分布对称；对称分布在三阶绝对矩有限且标准差非零时偏度为零（如非退化正态分布）
- **偏度 > 0**：右偏（正偏），尾部向右延伸
- **偏度 < 0**：左偏（负偏），尾部向左延伸

**生活例子：**
- **收入分布**：通常右偏，少数高收入者拉长右尾
- **考试成绩**：可能左偏，大多数学生分数较高
- **房价分布**：通常右偏，豪宅拉高平均价格
- **年龄分布**：在某些群体中可能呈现不同偏斜

### 1. 三种偏度类型
```python title="偏度类型可视化"
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
import seaborn as sns

def plot_skewness_types():
    """展示不同类型的偏度"""
    fig, ((ax1, ax2, ax3), (ax4, ax5, ax6)) = plt.subplots(2, 3, figsize=(15, 10))
    
 # 1. 对称分布（偏度≈0）
    np.random.seed(42)
    symmetric_data = np.random.normal(0, 1, 1000)
    skew_sym = stats.skew(symmetric_data)
    
    ax1.hist(symmetric_data, bins=30, density=True, alpha=0.7, color='blue', edgecolor='black')
    ax1.set_title(f'对称分布\n偏度 = {skew_sym:.3f}')
    ax1.set_ylabel('密度')
    ax1.grid(True, alpha=0.3)
    
 # 标注均值、中位数、众数
    mean_sym = np.mean(symmetric_data)
    median_sym = np.median(symmetric_data)
    ax1.axvline(mean_sym, color='red', linestyle='-', linewidth=2, label=f'均值={mean_sym:.2f}')
    ax1.axvline(median_sym, color='green', linestyle='--', linewidth=2, label=f'中位数={median_sym:.2f}')
    ax1.legend()
    
 # 2. 右偏分布（正偏度）
    right_skewed_data = np.random.lognormal(0, 0.8, 1000)
    skew_right = stats.skew(right_skewed_data)
    
    ax2.hist(right_skewed_data, bins=30, density=True, alpha=0.7, color='orange', edgecolor='black')
    ax2.set_title(f'右偏分布（正偏）\n偏度 = {skew_right:.3f}')
    ax2.grid(True, alpha=0.3)
    
    mean_right = np.mean(right_skewed_data)
    median_right = np.median(right_skewed_data)
    ax2.axvline(mean_right, color='red', linestyle='-', linewidth=2, label=f'均值={mean_right:.2f}')
    ax2.axvline(median_right, color='green', linestyle='--', linewidth=2, label=f'中位数={median_right:.2f}')
    ax2.legend()
    
 # 3. 左偏分布（负偏度）
    left_skewed_data = 10 - np.random.lognormal(0, 0.6, 1000)
    skew_left = stats.skew(left_skewed_data)
    
    ax3.hist(left_skewed_data, bins=30, density=True, alpha=0.7, color='green', edgecolor='black')
    ax3.set_title(f'左偏分布（负偏）\n偏度 = {skew_left:.3f}')
    ax3.grid(True, alpha=0.3)
    
    mean_left = np.mean(left_skewed_data)
    median_left = np.median(left_skewed_data)
    ax3.axvline(mean_left, color='red', linestyle='-', linewidth=2, label=f'均值={mean_left:.2f}')
    ax3.axvline(median_left, color='green', linestyle='--', linewidth=2, label=f'中位数={median_left:.2f}')
    ax3.legend()
    
 # 4-6. 箱线图对比
    data_sets = [symmetric_data, right_skewed_data, left_skewed_data]
    labels = ['对称分布', '右偏分布', '左偏分布']
    colors = ['blue', 'orange', 'green']
    
    for i, (data, label, color) in enumerate(zip(data_sets, labels, colors)):
        ax = [ax4, ax5, ax6][i]
        bp = ax.boxplot(data, patch_artist=True, labels=[label])
        bp['boxes'][0].set_facecolor(color)
        bp['boxes'][0].set_alpha(0.7)
        
 # 标注关键统计量
        q1, median, q3 = np.percentile(data, [25, 50, 75])
        mean_val = np.mean(data)
        
        ax.scatter([1], [mean_val], color='red', s=100, marker='D', label='均值')
        ax.set_title(f'{label}\n偏度 = {stats.skew(data):.3f}')
        ax.grid(True, alpha=0.3)
        if i == 0:
            ax.legend()
    
    plt.tight_layout()
    plt.show()
    
 # 数值比较
    print("偏度类型对比:")
    print("=" * 50)
    
    datasets = [
        ("对称分布", symmetric_data),
        ("右偏分布", right_skewed_data), 
        ("左偏分布", left_skewed_data)
    ]
    
    for name, data in datasets:
        mean_val = np.mean(data)
        median_val = np.median(data)
        skew_val = stats.skew(data)
        
        print(f"\n{name}:")
        print(f"  均值: {mean_val:.3f}")
        print(f"  中位数: {median_val:.3f}")
        print(f"  偏度: {skew_val:.3f}")
        print(f"  均值-中位数: {mean_val - median_val:.3f}")
        
        if abs(skew_val) < 0.5:
            skew_desc = "近似对称"
        elif skew_val > 0:
            if skew_val < 1:
                skew_desc = "轻度右偏"
            elif skew_val < 2:
                skew_desc = "中度右偏"
            else:
                skew_desc = "高度右偏"
        else:
            if skew_val > -1:
                skew_desc = "轻度左偏"
            elif skew_val > -2:
                skew_desc = "中度左偏"
            else:
                skew_desc = "高度左偏"
        
        print(f"  偏度程度: {skew_desc}")

plot_skewness_types()
```

### 2. 偏度的计算方法
```python title="偏度的不同计算方法"
def calculate_skewness_methods():
    """演示偏度的不同计算方法"""
    np.random.seed(42)
    
 # 生成右偏数据
    data = np.random.lognormal(1, 0.5, 1000)
    
    print("偏度的计算方法:")
    print("=" * 40)
    
 # 1. 手动计算 - 第三中心矩方法
    n = len(data)
    mean = np.mean(data)
    std = np.std(data, ddof=0)  # 与三阶样本中心矩使用相同的n分母，对应stats.skew默认的未校正矩偏度
    
 # 第三中心矩
    third_moment = np.mean((data - mean)**3)
    skew_manual = third_moment / (std**3)
    
    print(f"1. 手动计算（第三中心矩法）:")
    print(f"   均值: {mean:.4f}")
    print(f"   标准差: {std:.4f}")
    print(f"   第三中心矩: {third_moment:.4f}")
    print(f"   偏度: {skew_manual:.4f}")
    
 # 2. Pearson第一偏度系数
    median = np.median(data)
    from scipy.stats import gaussian_kde
    kde_first = gaussian_kde(data)
    mode_grid = np.linspace(data.min(), data.max(), 1000)
    mode_first = mode_grid[np.argmax(kde_first(mode_grid))]
    pearson_skew1 = (mean - mode_first) / std
    
    print(f"\n2. Pearson第一偏度系数:")
    print(f"   公式: (均值 - 众数) / 标准差；本例众数由核密度峰值近似")
    print(f"   偏度: {pearson_skew1:.4f}")
    
 # 3. Pearson第二偏度系数
 # 使用均值、中位数和标准差，不需要估计众数；下方核密度计算不参与第二系数
    from scipy.stats import gaussian_kde
    kde = gaussian_kde(data)
    x_range = np.linspace(data.min(), data.max(), 1000)
    kde_values = kde(x_range)
    mode_approx = x_range[np.argmax(kde_values)]
    
    pearson_skew2 = 3 * (mean - median) / std
    
    print(f"\n3. Pearson第二偏度系数:")
    print(f"   公式: 3 × (均值 - 中位数) / 标准差")
    print(f"   偏度: {pearson_skew2:.4f}")
    
 # 4. scipy.stats 计算
    scipy_skew = stats.skew(data)
    
    print(f"\n4. SciPy统计函数:")
    print(f"   偏度: {scipy_skew:.4f}")
    
 # 5. 未校正与偏差校正的样本偏度
    unadjusted_skew = stats.skew(data, bias=True)
    adjusted_skew = stats.skew(data, bias=False)
    
    print(f"\n5. 样本偏度口径对比:")
    print(f"   未校正矩偏度: {unadjusted_skew:.4f}")
    print(f"   偏差校正样本偏度: {adjusted_skew:.4f}")
    
 # 6. 偏度的置信区间
 # 使用bootstrap方法
    n_bootstrap = 1000
    bootstrap_skews = []
    
    for _ in range(n_bootstrap):
        bootstrap_sample = np.random.choice(data, size=len(data), replace=True)
        bootstrap_skews.append(stats.skew(bootstrap_sample))
    
    skew_ci_lower = np.percentile(bootstrap_skews, 2.5)
    skew_ci_upper = np.percentile(bootstrap_skews, 97.5)
    
    print(f"\n6. 偏度的95%置信区间 (Bootstrap):")
    print(f"   [{skew_ci_lower:.4f}, {skew_ci_upper:.4f}]")
    
 # 7. 样本量对偏度估计的影响
    print(f"\n7. 样本量对偏度估计的影响:")
    sample_sizes = [50, 100, 500, 1000, 5000]
    true_data = np.random.lognormal(1, 0.5, 10000)  # 真实总体
    true_skew = stats.skew(true_data)
    
    print(f"   真实偏度: {true_skew:.4f}")
    print("   样本量 -> 估计偏度 (标准误差)")
    
    for size in sample_sizes:
        sample_skews = []
        for _ in range(100):  # 100次重复采样
            sample = np.random.choice(true_data, size=size, replace=False)
            sample_skews.append(stats.skew(sample))
        
        mean_skew = np.mean(sample_skews)
        std_skew = np.std(sample_skews, ddof=1)
        
        print(f"   {size:4d} -> {mean_skew:.4f} ({std_skew:.4f})")

calculate_skewness_methods()
```

```python title="偏度分析工具函数"
def skewness_analysis_toolkit():
    """偏度分析的完整工具包"""
    
    def analyze_skewness(data, name="数据"):
        """分析数据的偏度"""
 # 基本统计量
        mean_val = np.mean(data)
        median_val = np.median(data)
        std_val = np.std(data, ddof=1)
        skew_val = stats.skew(data)
        
 # 偏度解释
        if abs(skew_val) < 0.5:
            skew_desc = "近似对称"
            color = 'blue'
        elif skew_val > 0:
            if skew_val < 1:
                skew_desc = "轻度右偏"
                color = 'orange'
            elif skew_val < 2:
                skew_desc = "中度右偏"
                color = 'red'
            else:
                skew_desc = "高度右偏"
                color = 'darkred'
        else:
            if skew_val > -1:
                skew_desc = "轻度左偏"
                color = 'lightgreen'
            elif skew_val > -2:
                skew_desc = "中度左偏"
                color = 'green'
            else:
                skew_desc = "高度左偏"
                color = 'darkgreen'
        
        print(f"{name}偏度分析:")
        print("-" * 30)
        print(f"样本量: {len(data)}")
        print(f"均值: {mean_val:.4f}")
        print(f"中位数: {median_val:.4f}")
        print(f"标准差: {std_val:.4f}")
        print(f"偏度: {skew_val:.4f}")
        print(f"偏度程度: {skew_desc}")
        print(f"均值-中位数: {mean_val - median_val:.4f}")
        
        return {
            'mean': mean_val,
            'median': median_val,
            'std': std_val,
            'skewness': skew_val,
            'description': skew_desc,
            'color': color
        }
    
    def plot_skewness_analysis(data, name="数据", ax=None):
        """绘制偏度分析图"""
        if ax is None:
            fig, ax = plt.subplots(1, 1, figsize=(10, 6))
        
 # 分析偏度
        result = analyze_skewness(data, name)
        
 # 绘制直方图
        ax.hist(data, bins=30, density=True, alpha=0.7, 
               color=result['color'], edgecolor='black')
        
 # 添加统计线
        ax.axvline(result['mean'], color='red', linestyle='-', 
                  linewidth=2, label=f"均值 = {result['mean']:.2f}")
        ax.axvline(result['median'], color='blue', linestyle='--', 
                  linewidth=2, label=f"中位数 = {result['median']:.2f}")
        
 # 尝试拟合正态分布作为对比
        x_range = np.linspace(data.min(), data.max(), 100)
        normal_fit = stats.norm.pdf(x_range, result['mean'], result['std'])
        ax.plot(x_range, normal_fit, 'g--', linewidth=2, alpha=0.7, 
               label="正态分布拟合")
        
        ax.set_title(f'{name} - {result["description"]}\n偏度 = {result["skewness"]:.3f}')
        ax.set_xlabel('值')
        ax.set_ylabel('密度')
        ax.legend()
        ax.grid(True, alpha=0.3)
        
        return result
    
 # 示例应用：分析不同类型的数据
    print("偏度分析工具包演示:")
    print("=" * 50)
    
 # 生成不同偏度的示例数据
    np.random.seed(42)
    
    datasets = {
        "正态分布": np.random.normal(50, 10, 1000),
        "收入分布": np.random.lognormal(3.5, 0.8, 1000),
        "考试成绩": 100 - np.random.exponential(15, 1000),
        "年龄分布": np.random.gamma(2, 15, 1000) + 18
    }
    
 # 分析每个数据集
    fig, axes = plt.subplots(2, 2, figsize=(15, 12))
    axes = axes.flatten()
    
    results = {}
    for i, (name, data) in enumerate(datasets.items()):
 # 演示固定分位数截尾，不等同于异常值识别；后续偏度仅描述截尾样本
        data = data[(data > np.percentile(data, 1)) & (data < np.percentile(data, 99))]
        
        print(f"\n{i+1}. {name}:")
        result = plot_skewness_analysis(data, name, axes[i])
        results[name] = result
    
    plt.tight_layout()
    plt.show()
    
 # 偏度对比总结
    print(f"\n偏度对比总结:")
    print("=" * 50)
    print(f"{'数据类型':<10} {'偏度':<8} {'程度':<10} {'均值-中位数':<12}")
    print("-" * 50)
    
    for name, result in results.items():
        diff = result['mean'] - result['median']
        print(f"{name:<10} {result['skewness']:<8.3f} {result['description']:<10} {diff:<12.3f}")
    
    return analyze_skewness, plot_skewness_analysis
```

### 2. 偏度检验
```python title="偏度的统计检验"
def skewness_tests():
    """偏度的统计检验方法"""
    print("偏度统计检验:")
    print("=" * 40)
    
 # 生成测试数据
    np.random.seed(42)
    
 # 不同偏度的数据
    test_datasets = {
        "正态分布": np.random.normal(0, 1, 500),
        "右偏分布": np.random.exponential(1, 500),
        "左偏分布": -np.random.exponential(1, 500),
        "轻度右偏": np.random.gamma(3, 1, 500)
    }
    
    for name, data in test_datasets.items():
        print(f"\n{name}:")
        print("-" * 20)
        
 # 计算偏度
        skew_val = stats.skew(data)
        n = len(data)
        
        print(f"样本偏度: {skew_val:.4f}")
        
 # 1. D'Agostino偏度检验
 # 使用正态总体参照检验偏度是否偏离零，不是任意分布的通用对称性检验
 # stats.skewtest要求至少8个观测；本例各组有500个观测
        z_skew, p_value_skew = stats.skewtest(data)
        
        print(f"D'Agostino偏度检验（正态参照）:")
        print(f"  检验统计量 Z = {z_skew:.4f}")
        print(f"  p值 = {p_value_skew:.4f}")
        
        if p_value_skew < 0.05:
            print(f"  结论: 在正态参照下，样本偏度显著偏离零 (p < 0.05)；这不是一般对称性检验")
        else:
            print(f"  结论: 在正态参照下，未发现偏度显著偏离零 (p ≥ 0.05)；不能证明分布对称或正态")
        
 # 2. Jarque-Bera检验 (偏度和峰度的联合检验)
        jb_stat, jb_pvalue = stats.jarque_bera(data)
        
        print(f"Jarque-Bera检验:")
        print(f"  JB统计量 = {jb_stat:.4f}")
        print(f"  p值 = {jb_pvalue:.4f}")
        
        if jb_pvalue < 0.05:
            print(f"  结论: 拒绝正态性假设 (p < 0.05)")
        else:
            print(f"  结论: 未拒绝正态性假设 (p ≥ 0.05)，不代表已证明正态性")
        
 # 3. 偏度的置信区间
 # 使用bootstrap方法
        n_bootstrap = 1000
        bootstrap_skews = []
        
        for _ in range(n_bootstrap):
            bootstrap_sample = np.random.choice(data, size=len(data), replace=True)
            bootstrap_skews.append(stats.skew(bootstrap_sample))
        
        ci_lower = np.percentile(bootstrap_skews, 2.5)
        ci_upper = np.percentile(bootstrap_skews, 97.5)
        
        print(f"偏度95%置信区间: [{ci_lower:.4f}, {ci_upper:.4f}]")
        
 # 判断偏度区间是否包含0；包含0不能证明对称性
        if ci_lower <= 0 <= ci_upper:
            print(f"  置信区间包含0，当前区间估计未排除零偏度；不能据此证明分布对称")
        else:
            print(f"  置信区间不包含0，支持非对称性")

skewness_tests()
```

### 1. 金融数据分析
```python title="股票收益率偏度分析"
def financial_skewness_analysis():
    """金融数据的偏度分析"""
    np.random.seed(42)
    
    print("金融数据偏度分析:")
    print("=" * 40)
    
 # 模拟不同类型股票的收益率数据
    trading_days = 252  # 一年的交易日
    
 # 不同风险特征的股票
    stocks = {
        "蓝筹股": np.random.normal(0.0008, 0.02, trading_days),  # 低风险，接近对称
        "成长股": np.random.lognormal(-0.001, 0.03, trading_days) - 1,  # 轻度右偏
        "小盘股": np.concatenate([
            np.random.normal(0.001, 0.025, int(trading_days*0.8)),  # 大部分正常交易
            np.random.normal(-0.05, 0.01, trading_days - int(trading_days*0.8))   # 偶尔大跌
        ]),  # 左偏
        "科技股": np.concatenate([
            np.random.normal(0.002, 0.03, int(trading_days*0.9)),
            np.random.normal(0.08, 0.02, trading_days - int(trading_days*0.9))   # 偶尔暴涨
        ])  # 右偏
    }
    
 # 分析每只股票
    analysis_results = {}
    
    for stock_name, returns in stocks.items():
 # 转换为百分比
        returns_pct = returns * 100
        
 # 基本统计
        mean_return = np.mean(returns_pct)
        std_return = np.std(returns_pct, ddof=1)
        skew_return = stats.skew(returns_pct)
        
 # 风险指标
        var_95 = np.percentile(-returns_pct, 95)  # 损失=-收益率，采用95%损失分位数作为VaR；单位为百分比
        
 # 偏度的风险含义
        if skew_return > 0:
```

```python title="处理偏斜数据的方法"
def handle_skewed_data():
    """处理偏斜数据的各种方法"""
    np.random.seed(42)
    
 # 生成强右偏数据
    skewed_data = np.random.lognormal(2, 1, 1000)
    
    print("处理偏斜数据的方法:")
    print("=" * 40)
    
 # 原始数据分析
    original_skew = stats.skew(skewed_data)
    print(f"原始数据偏度: {original_skew:.4f}")
    
 # 方法1: 对数变换
    log_data = np.log(skewed_data)
    log_skew = stats.skew(log_data)
    
 # 方法2: 平方根变换
    sqrt_data = np.sqrt(skewed_data)
    sqrt_skew = stats.skew(sqrt_data)
    
 # 方法3: Box-Cox变换
    from scipy.stats import boxcox
    boxcox_data, lambda_param = boxcox(skewed_data)
    boxcox_skew = stats.skew(boxcox_data)
    
 # 方法4: Yeo-Johnson变换（可处理负值）
    from scipy.stats import yeojohnson
    yj_data, yj_lambda = yeojohnson(skewed_data)
    yj_skew = stats.skew(yj_data)
    
 # 方法5: 分位数变换
    from sklearn.preprocessing import QuantileTransformer
    qt = QuantileTransformer(output_distribution='normal', random_state=42)
    qt_data = qt.fit_transform(skewed_data.reshape(-1, 1)).flatten()
    qt_skew = stats.skew(qt_data)
    
 # 结果对比
    transformations = {
        "原始数据": (skewed_data, original_skew),
        "对数变换": (log_data, log_skew),
        "平方根变换": (sqrt_data, sqrt_skew),
        "Box-Cox变换": (boxcox_data, boxcox_skew),
        "Yeo-Johnson变换": (yj_data, yj_skew),
        "分位数变换": (qt_data, qt_skew)
    }
    
    print(f"\n变换效果对比:")
    print(f"{'方法':<15} {'偏度':<10} {'改善程度':<10}")
    print("-" * 40)
    
    for method, (data, skew_val) in transformations.items():
        improvement = (abs(original_skew) - abs(skew_val)) / abs(original_skew) * 100
        if method == "原始数据":
            print(f"{method:<15} {skew_val:<10.4f} {'--':<10}")
        else:
            print(f"{method:<15} {skew_val:<10.4f} {improvement:<10.1f}%")
    
 # 可视化对比
    fig, axes = plt.subplots(2, 3, figsize=(18, 12))
    axes = axes.flatten()
    
    for i, (method, (data, skew_val)) in enumerate(transformations.items()):
        axes[i].hist(data, bins=30, density=True, alpha=0.7, 
                    color=plt.cm.tab10(i), edgecolor='black')
        axes[i].set_title(f'{method}\n偏度 = {skew_val:.3f}')
        axes[i].set_ylabel('密度')
        axes[i].grid(True, alpha=0.3)
        
 # 添加正态分布拟合
        mean_val = np.mean(data)
        std_val = np.std(data, ddof=1)
        x_range = np.linspace(data.min(), data.max(), 100)
        normal_fit = stats.norm.pdf(x_range, mean_val, std_val)
        axes[i].plot(x_range, normal_fit, 'r--', alpha=0.7, label='正态拟合')
        axes[i].legend()
    
    plt.tight_layout()
    plt.show()
    
 # 变换选择建议
    print(f"\n变换方法选择建议:")
    print("-" * 20)
    print("1. 对数变换: 适用于右偏数据，数据必须为正")
    print("2. 平方根变换: 适用于轻度右偏，数据必须非负")
    print("3. Box-Cox变换: 自动寻找最优变换，数据必须为正")
    print("4. Yeo-Johnson变换: 类似Box-Cox，但可处理负值")
    print("5. 分位数变换: 强制转换为指定分布，但会改变数据关系")
    
    print(f"\n最佳变换参数:")
    print(f"Box-Cox λ = {lambda_param:.4f}")
    print(f"Yeo-Johnson λ = {yj_lambda:.4f}")
    
    return transformations
```

### 2. 偏度监控系统
```python title="偏度监控预警系统"
def skewness_monitoring_system():
    """构建偏度监控预警系统"""
    
    class SkewnessMonitor:
        def __init__(self, threshold=0.5, window_size=100):
            self.threshold = threshold
            self.window_size = window_size
            self.data_buffer = []
            self.skewness_history = []
            self.alerts = []
        
        def add_data(self, new_data):
            """添加新数据点"""
            if isinstance(new_data, (list, np.ndarray)):
                self.data_buffer.extend(new_data)
            else:
                self.data_buffer.append(new_data)
            
 # 保持缓冲区大小
            if len(self.data_buffer) > self.window_size * 2:
                self.data_buffer = self.data_buffer[-self.window_size:]
            
 # 计算当前偏度
            if len(self.data_buffer) >= self.window_size:
                current_skew = stats.skew(self.data_buffer[-self.window_size:])
                self.skewness_history.append(current_skew)
                
 # 检查是否需要预警
                self._check_alert(current_skew)
                
                return current_skew
            return None
        
        def _check_alert(self, current_skew):
            """检查是否触发预警"""
            if abs(current_skew) > self.threshold:
                alert_type = "右偏预警" if current_skew > 0 else "左偏预警"
                self.alerts.append({
                    'timestamp': len(self.skewness_history) - 1,
                    'type': alert_type,
                    'skewness': current_skew,
                    'severity': self._get_severity(current_skew)
                })
        
        def _get_severity(self, skew_val):
            """获取预警严重程度"""
            abs_skew = abs(skew_val)
            if abs_skew > 2:
                return "严重"
            elif abs_skew > 1:
                return "中等"
            else:
                return "轻微"
        
        def get_status(self):
            """获取当前监控状态"""
            if not self.skewness_history:
                return "无数据"
            
            current_skew = self.skewness_history[-1]
            
            if abs(current_skew) <= self.threshold:
                return "正常"
            elif abs(current_skew) <= 1.0:
                return "注意"
            elif abs(current_skew) <= 2.0:
                return "警告"
            else:
                return "严重"
        
        def plot_monitoring(self):
            """绘制监控图表"""
            if not self.skewness_history:
                print("无数据可绘制")
                return
            
            fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(12, 10))
            
 # 偏度时间序列
            timestamps = range(len(self.skewness_history))
            ax1.plot(timestamps, self.skewness_history, 'b-', linewidth=2, label='偏度')
            ax1.axhline(y=self.threshold, color='orange', linestyle='--', 
                       label=f'预警阈值 ±{self.threshold}')
            ax1.axhline(y=-self.threshold, color='orange', linestyle='--')
            ax1.axhline(y=1, color='red', linestyle=':', alpha=0.7, label='高风险线 ±1')
            ax1.axhline(y=-1, color='red', linestyle=':', alpha=0.7)
            
 # 标记预警点
            for alert in self.alerts:
                color = {'轻微': 'yellow', '中等': 'orange', '严重': 'red'}[alert['severity']]
                ax1.scatter(alert['timestamp'], alert['skewness'], 
                          color=color, s=100, marker='^', 
                          label=f"{alert['type']} ({alert['severity']})")
            
            ax1.set_title('偏度监控时间序列')
            ax1.set_xlabel('时间点')
            ax1.set_ylabel('偏度值')
            ax1.legend()
            ax1.grid(True, alpha=0.3)
            
 # 最近数据分布
            recent_data = self.data_buffer[-self.window_size:]
            ax2.hist(recent_data, bins=30, density=True, alpha=0.7, 
                    color='skyblue', edgecolor='black')
            ax2.set_title(f'最近{len(recent_data)}个数据点的分布')
            ax2.set_xlabel('数值')
            ax2.set_ylabel('密度')
            ax2.grid(True, alpha=0.3)
            
 # 标注当前偏度
            if self.skewness_history:
                current_skew = self.skewness_history[-1]
                ax2.text(0.02, 0.98, f'当前偏度: {current_skew:.3f}', 
                        transform=ax2.transAxes, verticalalignment='top',
                        bbox=dict(boxstyle="round,pad=0.3", facecolor="white", alpha=0.8))
            
            plt.tight_layout()
            plt.show()
        
        def generate_report(self):
            """生成监控报告"""
            print("偏度监控报告")
            print("=" * 30)
            
            if not self.skewness_history:
                print("无监控数据")
                return
            
            current_skew = self.skewness_history[-1]
            status = self.get_status()
            
            print(f"当前状态: {status}")
            print(f"当前偏度: {current_skew:.4f}")
            print(f"监控点数: {len(self.skewness_history)}")
            print(f"预警次数: {len(self.alerts)}")
            
            if self.alerts:
                print(f"\n最近预警:")
                for alert in self.alerts[-3:]:  # 显示最近3次预警
                    print(f"  时间点{alert['timestamp']}: {alert['type']} "
                          f"(偏度={alert['skewness']:.3f}, {alert['severity']})")
    
 # 演示监控系统
    print("偏度监控系统演示:")
    print("=" * 40)
    
 # 创建监控器
    monitor = SkewnessMonitor(threshold=0.5, window_size=50)
    
 # 模拟数据流
    np.random.seed(42)
    
 # 第1阶段：正常数据
    normal_data = np.random.normal(100, 10, 100)
    for data_point in normal_data:
        monitor.add_data(data_point)
    
    print("第1阶段 (正常数据) 完成")
    print(f"状态: {monitor.get_status()}")
    
 # 第2阶段：逐渐变偏
    skewed_data = np.random.lognormal(4.5, 0.3, 100)
    for data_point in skewed_data:
        monitor.add_data(data_point)
    
    print(f"\n第2阶段 (偏斜数据) 完成")
    print(f"状态: {monitor.get_status()}")
    
 # 第3阶段：恢复正常
    recovery_data = np.random.normal(100, 10, 50)
    for data_point in recovery_data:
        monitor.add_data(data_point)
    
    print(f"\n第3阶段 (恢复正常) 完成")
    print(f"状态: {monitor.get_status()}")
    
 # 生成报告和图表
    monitor.generate_report()
    monitor.plot_monitoring()
    
    return monitor

# 使用监控系统
monitor = skewness_monitoring_system()
```
