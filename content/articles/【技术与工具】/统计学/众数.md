## 🤔 什么是众数？

> [!note] 📖 **众数的定义**
> 众数是在一组数据中出现频率最高的数值。它反映了数据中最典型、最常见的取值。

**生活例子：**
- 班级里最多人穿的鞋码就是众数（比如39码）
- 商店里最畅销的商品价位就是众数
- 调查问卷中被选择最多的答案就是众数

## 📊 众数的类型

### 1. 单众数（Unimodal）
只有一个出现频率最高的值

```python title="单众数示例"
import numpy as np
from scipy import stats

data = [1, 2, 2, 2, 3, 4, 5]
mode_result = stats.mode(data)
print(f"众数: {mode_result.mode[0]}, 出现次数: {mode_result.count[0]}")
# 输出: 众数: 2, 出现次数: 3
```

### 2. 双众数（Bimodal）
有两个出现频率相同且最高的值

```python title="双众数示例"
data = [1, 1, 1, 2, 3, 4, 4, 4, 5]
# 1和4都出现3次，都是众数
frequencies = {}
for x in data:
    frequencies[x] = frequencies.get(x, 0) + 1

max_freq = max(frequencies.values())
modes = [k for k, v in frequencies.items() if v == max_freq]
print(f"众数: {modes}, 最高频次: {max_freq}")
# 输出: 众数: [1, 4], 最高频次: 3
```

### 3. 多众数（Multimodal）
有多个出现频率相同且最高的值

### 4. 无众数
所有值的出现频率都相同

```python title="无众数示例"
data = [1, 2, 3, 4, 5]  # 每个值都只出现1次
print("此数据集无众数（或每个值都是众数）")
```

## 💻 Python实现方法

### 方法1：使用scipy.stats
```python title="scipy.stats方法"
from scipy import stats
import numpy as np

# 数值型数据
data = [1, 2, 2, 3, 3, 3, 4, 5]
mode_result = stats.mode(data)
print(f"众数: {mode_result.mode[0]}")
print(f"出现次数: {mode_result.count[0]}")
```

### 方法2：使用statistics模块
```python title="statistics模块方法"
import statistics

data = [1, 2, 2, 3, 3, 3, 4, 5]
try:
    mode_val = statistics.mode(data)
    print(f"众数: {mode_val}")
    # Python 3.8+：并列最高频时返回首次出现的值，不表示众数唯一。
    # 判断唯一性应使用statistics.multimode并检查返回值个数。
except statistics.StatisticsError:
    print("无法计算众数：空数据，或旧版Python遇到无唯一众数的情况")

# 多众数情况
data_multi = [1, 1, 2, 2, 3]
try:
    mode_val = statistics.multimode(data_multi)  # Python 3.8+
    print(f"所有众数: {mode_val}")
except AttributeError:
    print("multimode函数需要Python 3.8+")
```

### 方法3：手动实现
```python title="手动计算众数"
def find_mode(data):
    """计算众数的完整实现"""
    if not data:
        return None, 0
    
 # 计算频次
    frequencies = {}
    for item in data:
        frequencies[item] = frequencies.get(item, 0) + 1
    
 # 找到最高频次
    max_freq = max(frequencies.values())
    
 # 找到所有众数
    modes = [k for k, v in frequencies.items() if v == max_freq]
    
    return modes, max_freq

# 测试不同情况
test_data = [
    [1, 2, 2, 3, 3, 3, 4, 5],           # 单众数
    [1, 1, 2, 2, 3],                    # 双众数
    [1, 2, 3, 4, 5],                    # 无众数
    ['A', 'B', 'B', 'C', 'C', 'C']     # 分类数据
]

for i, data in enumerate(test_data, 1):
    modes, freq = find_mode(data)
    print(f"数据{i}: {data}")
    print(f"众数: {modes}, 出现次数: {freq}\n")
```

### 3. 连续数据的众数
```python title="连续数据众数处理"
# 连续数据通常需要分组
def continuous_mode(data, bins=10):
    """连续数据的众数估计"""
    hist, bin_edges = np.histogram(data, bins=bins)
    max_freq_idx = np.argmax(hist)
    
 # 众数区间的中点作为众数估计
    mode_estimate = (bin_edges[max_freq_idx] + bin_edges[max_freq_idx + 1]) / 2
    max_freq = hist[max_freq_idx]
    
    return mode_estimate, max_freq, bin_edges[max_freq_idx:max_freq_idx+2]

# 示例：身高数据
heights = np.random.normal(170, 10, 1000)  # 正态分布身高数据
mode_est, freq, interval = continuous_mode(heights, bins=20)

print(f"众数估计: {mode_est:.2f}cm")
print(f"众数区间: [{interval[0]:.2f}, {interval[1]:.2f}]")
print(f"该区间频数: {freq}")
```

### 3. 教育评估
```python title="考试成绩分析"
# 将成绩分组分析
scores = [85, 87, 90, 87, 92, 87, 88, 91, 87, 89]

# 原始众数
modes, freq = find_mode(scores)
print(f"最常见分数: {modes[0]}分")

# 按等级分组
def score_to_grade(score):
    if score >= 90: return 'A'
    elif score >= 80: return 'B'
    elif score >= 70: return 'C'
    elif score >= 60: return 'D'
    else: return 'F'

grades = [score_to_grade(s) for s in scores]
grade_modes, grade_freq = find_mode(grades)
print(f"最常见等级: {grade_modes[0]}级")
```

## 🆚 众数与其他集中趋势指标对比

```python title="三种指标对比"
import numpy as np
from scipy import stats

# 不同分布的数据
datasets = {
    "正态分布": np.random.normal(50, 10, 1000),
    "右偏分布": np.random.exponential(2, 1000),
    "双峰分布": np.concatenate([np.random.normal(30, 5, 500), 
                                np.random.normal(70, 5, 500)])
}

for name, data in datasets.items():
    mean_val = np.mean(data)
    median_val = np.median(data)
    
 # 连续数据的众数估计
    hist, bin_edges = np.histogram(data, bins=30)
    max_freq_idx = np.argmax(hist)
    mode_val = (bin_edges[max_freq_idx] + bin_edges[max_freq_idx+1]) / 2
    
    print(f"{name}:")
    print(f"  均值: {mean_val:.2f}")
    print(f"  中位数: {median_val:.2f}")
    print(f"  众数: {mode_val:.2f}")
    print()
```

## 🔍 众数的优缺点分析

### ✅ 优点
1. **适用性最广**：适用于所有数据类型（定类、定序、定距、定比）
2. **抗异常值**：不受极端值影响
3. **直观易懂**：反映最典型的情况
4. **实际意义强**：在商业决策中很有用

### ❌ 缺点
1. **可能不存在**：所有值出现频率相同时
2. **可能不唯一**：多个众数的情况
3. **信息利用少**：只考虑频率，不考虑数值大小
4. **连续数据处理复杂**：需要分组或特殊方法

## 🛠️ 实用技巧

### 1. 处理多众数情况
```python title="多众数处理策略"
def analyze_modes(data):
    """分析众数情况并给出建议"""
    modes, freq = find_mode(data)
    total_count = len(data)
    if total_count == 0:
        print("空数据：无法计算众数")
        return "empty_data"
    
    if len(modes) == 1:
        print(f"单众数: {modes[0]} (出现{freq}次，占{freq/total_count*100:.1f}%)")
        return "single_mode"
    elif len(modes) == len(set(data)):
        print(f"无众数: 所有值出现频率相同")
        return "no_mode"
    else:
        print(f"多众数: {modes} (各出现{freq}次)")
        print("建议: 考虑数据的分布特征或重新分组")
        return "multi_mode"

# 测试
test_cases = [
    [1, 2, 2, 2, 3, 4],        # 单众数
    [1, 1, 2, 2, 3, 3],        # 所有值等频，按本文约定为无众数
    [1, 2, 3, 4, 5]            # 无众数
]

for i, data in enumerate(test_cases, 1):
    print(f"数据集{i}: {data}")
    analyze_modes(data)
    print()
```

### 2. 分组众数分析
```python title="分组数据众数"
def grouped_mode_analysis(data, group_col, value_col):
    """分组数据的众数分析"""
    import pandas as pd
    
    df = pd.DataFrame({'group': group_col, 'value': value_col})
    
    result = {}
    for group in df['group'].unique():
        group_data = df[df['group'] == group]['value'].tolist()
        modes, freq = find_mode(group_data)
        result[group] = {'modes': modes, 'frequency': freq}
    
    return result

# 示例：不同部门员工年龄众数
departments = ['销售', '技术', '销售', '技术', '销售', '人事', '技术', '销售']
ages = [25, 28, 25, 30, 25, 35, 28, 27]

dept_modes = grouped_mode_analysis(ages, departments, ages)
for dept, info in dept_modes.items():
    print(f"{dept}部门最常见年龄: {info['modes']} (出现{info['frequency']}次)")
```
