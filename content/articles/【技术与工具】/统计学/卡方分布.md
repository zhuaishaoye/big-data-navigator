## 🤔 什么是卡方分布？

> [!note] 📖 **卡方分布的定义**
> 卡方分布是由多个独立标准正态分布的平方和构成的连续概率分布。如果X₁, X₂, ..., Xₖ 是独立的标准正态随机变量，那么它们的平方和 χ² = X₁² + X₂² + ... + Xₖ² 服从自由度为k的卡方分布。

**数学表达式**：
- **概率密度函数**：$$f(x) = \frac{1}{2^{k/2}\Gamma(k/2)} x^{k/2-1} e^{-x/2}, \quad x > 0$$
- **记号**：$\chi^2 \sim \chi^2(k)$，其中 $k$ 是自由度

**重要特性**：
- 只取非负值（$x \geq 0$）
- 右偏分布，随自由度增大趋于对称
- 均值 = 自由度 $k$
- 方差 = $2k$

## 🏗️ 生活中的卡方分布

想象你在做一个"投掷色子"的实验：

> [!example] 🎲 **色子公平性检验**
> 你怀疑一个六面色子是否公平。投掷600次，每个面理论上应该出现100次。实际结果：
> - 面1：95次，面2：110次，面3：85次
> - 面4：120次，面5：90次，面6：100次
> 
> 卡方统计量 = Σ(观察值-期望值)²/期望值；在公平色子的原假设成立、各次投掷独立且期望频数足够大时，该统计量的分布近似为自由度5的卡方分布。

**其他应用场景**：
- 检验药物疗效是否与性别独立
- 分析不同地区销售数据是否存在显著差异
- 验证抽样调查结果是否符合总体分布
- 质量控制中的方差检验

```python title="卡方分布的基本性质"
import numpy as np
import matplotlib.pyplot as plt
from scipy import stats
import seaborn as sns

def plot_chi2_properties():

    ax1.set_ylabel('概率密度')

    means = df_range  # 均值 = df
    variances = 2 * df_range  # 方差 = 2*df
    std_devs = np.sqrt(variances)

    ax2.set_title('卡方分布的均值与标准差')

        cdf = stats.chi2.cdf(x_cdf, df)

    ax4.set_ylabel('分位数值')
    ax4.set_title('卡方分布的分位数')
    ax4.legend()
    ax4.grid(True, alpha=0.3)

    plt.show()
    
    # 输出重要性质
    print("卡方分布的重要性质:")
    print("=" * 40)

    print("2. 右偏分布，df增大时趋于对称")
    print("3. 均值 E(X) = df")
    print("4. 方差 Var(X) = 2·df")
    print("5. 标准差 SD(X) = √(2·df)")
    print("6. 模式 Mode = max(0, df-2)")
    print("7. 可加性: 若X₁ ~ χ²(df₁)、X₂ ~ χ²(df₂)且相互独立，则X₁ + X₂ ~ χ²(df₁+df₂)")

plot_chi2_properties()
```

```python title="卡方分布数值特征验证"
def verify_chi2_properties():
    """验证卡方分布的数值特征"""
    
    df_values = [1, 3, 5, 10, 20]
    n_samples = 100000
    
    print("卡方分布数值特征验证:")
    print("=" * 50)
    print("自由度\t理论均值\t模拟均值\t理论方差\t模拟方差\t理论标准差\t模拟标准差")
    print("-" * 80)
    
    np.random.seed(42)
    
    for df in df_values:
        # 生成随机样本
        samples = stats.chi2.rvs(df, size=n_samples)
        
        # 理论值
        theory_mean = df
        theory_var = 2 * df
        theory_std = np.sqrt(2 * df)
        
        # 模拟值
        sim_mean = np.mean(samples)
        sim_var = np.var(samples, ddof=1)
        sim_std = np.std(samples, ddof=1)
        
        print(f"{df}\t{theory_mean:.2f}\t\t{sim_mean:.2f}\t\t{theory_var:.2f}\t\t{sim_var:.2f}\t\t{theory_std:.2f}\t\t{sim_std:.2f}")
    
    # 可加性验证
    print(f"\n可加性验证:")
    print("-" * 30)
    
    df1, df2 = 3, 5
    df_sum = df1 + df2
    
    # 独立生成两个卡方分布
    chi2_1 = stats.chi2.rvs(df1, size=n_samples)
    chi2_2 = stats.chi2.rvs(df2, size=n_samples)
    chi2_sum = chi2_1 + chi2_2
    
    # 直接生成df1+df2的卡方分布
    chi2_direct = stats.chi2.rvs(df_sum, size=n_samples)
    
    print(f"χ²({df1}) + χ²({df2}) 的均值: {np.mean(chi2_sum):.3f}")
    print(f"χ²({df_sum}) 的均值: {np.mean(chi2_direct):.3f}")
    print(f"理论均值: {df_sum}")
    
    print(f"\nχ²({df1}) + χ²({df2}) 的方差: {np.var(chi2_sum, ddof=1):.3f}")
    print(f"χ²({df_sum}) 的方差: {np.var(chi2_direct, ddof=1):.3f}")
    print(f"理论方差: {2 * df_sum}")

verify_chi2_properties()
```

### 1. 基本计算
```python title="卡方分布的基本计算"
from scipy import stats
import numpy as np

def chi2_calculations():
    """卡方分布的各种计算"""
    
    df = 5  # 自由度为5
    
    print(f"卡方分布 χ²({df}) 的计算:")
    print("=" * 30)
    
    # 1. 概率密度函数值
    x_values = [1, 3, 5, 7, 10, 15]
    print("概率密度函数值:")
    for x in x_values:
        pdf_value = stats.chi2.pdf(x, df)
        print(f"f({x}) = {pdf_value:.4f}")
    
    # 2. 累积分布函数值
    print(f"\n累积分布函数值:")
    for x in [2, 5, 7, 10, 15]:
        cdf_value = stats.chi2.cdf(x, df)
        print(f"P(X ≤ {x}) = {cdf_value:.4f}")
    
    # 3. 上尾概率（生存函数）
    print(f"\n上尾概率:")
    for x in [5, 7, 10, 15]:
        sf_value = stats.chi2.sf(x, df)
        print(f"P(X > {x}) = {sf_value:.4f}")
    
    # 4. 分位数
    print(f"\n重要分位数:")
    percentiles = [0.9, 0.95, 0.975, 0.99, 0.995, 0.999]
    for p in percentiles:
        quantile = stats.chi2.ppf(p, df)
        print(f"{p:.1%}分位数: {quantile:.4f}")
    
    # 5. 区间概率
    print(f"\n区间概率:")
    intervals = [(2, 8), (3, 10), (5, 15)]
    for a, b in intervals:
        prob = stats.chi2.cdf(b, df) - stats.chi2.cdf(a, df)
        print(f"P({a} ≤ X ≤ {b}) = {prob:.4f}")
    
    # 6. 统计特征
    print(f"\n统计特征:")
    mean = stats.chi2.mean(df)
    var = stats.chi2.var(df)
    std = stats.chi2.std(df)
    
    print(f"均值: {mean:.2f}")
    print(f"方差: {var:.2f}")
    print(f"标准差: {std:.2f}")
    
    return df

df = chi2_calculations()
```

```python title="卡方分布随机数生成"
def chi2_simulation_and_visualization():
    """卡方分布模拟和可视化"""
    
    df = 8
    n_samples = 10000
    
    print(f"卡方分布 χ²({df}) 模拟分析:")
    print("=" * 35)
    
    # 生成随机数
    np.random.seed(42)
    samples = stats.chi2.rvs(df, size=n_samples)
    
    # 基本统计
    print(f"样本量: {n_samples:,}")
    print(f"样本均值: {np.mean(samples):.3f} (理论值: {df})")
    print(f"样本方差: {np.var(samples, ddof=1):.3f} (理论值: {2*df})")
    print(f"样本标准差: {np.std(samples, ddof=1):.3f} (理论值: {np.sqrt(2*df):.3f})")
    print(f"样本最小值: {np.min(samples):.3f}")
    print(f"样本最大值: {np.max(samples):.3f}")
    
    # 分位数比较
    print(f"\n分位数比较:")
    print("百分位\t样本分位数\t理论分位数\t误差")
    print("-" * 40)
    
    percentiles_check = [0.25, 0.5, 0.75, 0.9, 0.95]
    for p in percentiles_check:
        sample_quantile = np.percentile(samples, p * 100)
        theory_quantile = stats.chi2.ppf(p, df)
        error = abs(sample_quantile - theory_quantile)
        
        print(f"{p:.0%}\t{sample_quantile:.3f}\t\t{theory_quantile:.3f}\t\t{error:.3f}")
    
    # 可视化
    fig, ((ax1, ax2), (ax3, ax4)) = plt.subplots(2, 2, figsize=(15, 12))
    
    # 1. 直方图vs理论PDF
    ax1.hist(samples, bins=50, density=True, alpha=0.7, color='skyblue', edgecolor='black')
    
    x_theory = np.linspace(0, max(samples), 1000)
    pdf_theory = stats.chi2.pdf(x_theory, df)
    ax1.plot(x_theory, pdf_theory, 'r-', linewidth=2, label=f'理论PDF χ²({df})')
    
    ax1.set_xlabel('x')
    ax1.set_ylabel('概率密度')
    ax1.set_title('样本分布 vs 理论分布')
    ax1.legend()
    ax1.grid(True, alpha=0.3)
    
    # 2. Q-Q图
    from scipy.stats import probplot
    probplot(samples, dist=stats.chi2(df), plot=ax2)
    ax2.set_title('Q-Q图: 样本 vs 理论χ²分布')
    ax2.grid(True, alpha=0.3)
    
    # 3. 经验CDF vs 理论CDF
    x_sorted = np.sort(samples)
    empirical_cdf = np.arange(1, len(x_sorted) + 1) / len(x_sorted)
    theoretical_cdf = stats.chi2.cdf(x_sorted, df)
    
    ax3.plot(x_sorted, empirical_cdf, label='经验CDF', linewidth=1)
    ax3.plot(x_sorted, theoretical_cdf, 'r--', label='理论CDF', linewidth=2)
    ax3.set_xlabel('x')
    ax3.set_ylabel('累积概率')
    ax3.set_title('经验CDF vs 理论CDF')
    ax3.legend()
    ax3.grid(True, alpha=0.3)
    
    # 4. 箱线图分析
    quartiles = np.percentile(samples, [25, 50, 75])
    iqr = quartiles[2] - quartiles[0]
    
    ax4.boxplot(samples, vert=True, patch_artist=True, 
                boxprops=dict(facecolor='lightblue', alpha=0.7))
    ax4.set_ylabel('值')
    ax4.set_title('样本箱线图')
    ax4.grid(True, alpha=0.3)
    
    # 添加统计信息
    ax4.text(1.1, quartiles[1], f'中位数: {quartiles[1]:.2f}', fontsize=10)
    ax4.text(1.1, quartiles[1] - iqr*0.3, f'IQR: {iqr:.2f}', fontsize=10)
    
    plt.tight_layout()
    plt.show()
    
    return samples

samples = chi2_simulation_and_visualization()
```

```python title="拟合优度检验案例"
def goodness_of_fit_test():
    """拟合优度检验案例"""
    
    print("案例1: 色子公平性检验")
    print("=" * 30)
    
    # 投掷600次色子的观察结果
    observed = np.array([95, 110, 85, 120, 90, 100])  # 各面出现次数
    expected = np.array([100, 100, 100, 100, 100, 100])  # 期望次数（公平色子）
    faces = ['面1', '面2', '面3', '面4', '面5', '面6']
    
    print("观察结果:")
    for face, obs, exp in zip(faces, observed, expected):
        print(f"{face}: 观察{obs}次, 期望{exp}次, 偏差{obs-exp:+d}")
    
    # 手动计算卡方统计量
    chi2_manual = np.sum((observed - expected)**2 / expected)
    df = len(observed) - 1  # 本例概率预先指定且未估计额外参数，自由度 = 类别数 - 1
    
    # 使用scipy计算
    chi2_stat, p_value = stats.chisquare(observed, expected)
    
    print(f"\n检验结果:")
    print(f"卡方统计量 χ² = {chi2_stat:.4f}")
    print(f"自由度 df = {df}")
    print(f"p值 = {p_value:.4f}")
    
    # 临界值
    alpha = 0.05
    critical_value = stats.chi2.ppf(1 - alpha, df)
    
    print(f"\n决策 (α = {alpha}):")
    print(f"临界值 χ²_{alpha}({df}) = {critical_value:.4f}")
    
    if chi2_stat > critical_value:
        print(f"拒绝原假设: 色子不公平 (χ² = {chi2_stat:.4f} > {critical_value:.4f})")
    else:
        print(f"不拒绝原假设: 没有足够证据表明色子不公平")
    
    if p_value < alpha:
        print(f"p值判断: 拒绝原假设 (p = {p_value:.4f} < {alpha})")
    else:
        print(f"p值判断: 不拒绝原假设 (p = {p_value:.4f} ≥ {alpha})")
    
    # 可视化
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(15, 6))
    
    # 观察值vs期望值
    x_pos = np.arange(len(faces))
    width = 0.35
    
    bars1 = ax1.bar(x_pos - width/2, observed, width, label='观察值', alpha=0.7, color='skyblue')
    bars2 = ax1.bar(x_pos + width/2, expected, width, label='期望值', alpha=0.7, color='lightcoral')
    
    ax1.set_xlabel('色子面')
    ax1.set_ylabel('频次')
    ax1.set_title('观察频次 vs 期望频次')
    ax1.set_xticks(x_pos)
    ax1.set_xticklabels(faces)
    ax1.legend()
    ax1.grid(True, alpha=0.3)
    
    # 在柱子上标注数值
    for bar1, bar2, obs, exp in zip(bars1, bars2, observed, expected):
        ax1.text(bar1.get_x() + bar1.get_width()/2, bar1.get_height() + 2, 
                str(obs), ha='center', va='bottom', fontsize=10)
        ax1.text(bar2.get_x() + bar2.get_width()/2, bar2.get_height() + 2, 
                str(exp), ha='center', va='bottom', fontsize=10)
    
    # 卡方分布和检验统计量
    x_chi2 = np.linspace(0, 20, 1000)
    pdf_chi2 = stats.chi2.pdf(x_chi2, df)
    
    ax2.plot(x_chi2, pdf_chi2, 'b-', linewidth=2, label=f'χ²({df})分布')
    ax2.axvline(chi2_stat, color='red', linestyle='--', linewidth=2, 
                label=f'检验统计量 = {chi2_stat:.3f}')
    ax2.axvline(critical_value, color='green', linestyle='--', linewidth=2, 
                label=f'临界值 = {critical_value:.3f}')
    
    # 填充拒绝域
    x_reject = x_chi2[x_chi2 >= critical_value]
    pdf_reject = stats.chi2.pdf(x_reject, df)
    ax2.fill_between(x_reject, 0, pdf_reject, alpha=0.3, color='red', label='拒绝域')
    
    ax2.set_xlabel('χ² 值')
    ax2.set_ylabel('概率密度')
    ax2.set_title('卡方检验图示')
    ax2.legend()
    ax2.grid(True, alpha=0.3)
    ax2.set_xlim(0, 20)
    
    plt.tight_layout()
    plt.show()

goodness_of_fit_test()
```

### 2. 独立性检验
```python title="独立性检验案例"
def independence_test():
    """独立性检验案例"""
    
    print("\n\n案例2: 药物疗效与性别独立性检验")
    print("=" * 40)
    
    # 2x2列联表：药物疗效 vs 性别
    # 行：疗效（有效/无效），列：性别（男/女）
    observed_table = np.array([
        [60, 40],   # 有效：男60人，女40人
        [30, 70]    # 无效：男30人，女70人
    ])
    
    # 标签
    effect_labels = ['有效', '无效']
    gender_labels = ['男性', '女性']
    
    print("观察数据 (列联表):")
    print("\t\t男性\t女性\t总计")
    
    row_totals = observed_table.sum(axis=1)
    col_totals = observed_table.sum(axis=0)
    grand_total = observed_table.sum()
    
    for i, effect in enumerate(effect_labels):
        print(f"{effect}\t\t{observed_table[i,0]}\t{observed_table[i,1]}\t{row_totals[i]}")
    
    print(f"总计\t\t{col_totals[0]}\t{col_totals[1]}\t{grand_total}")
    
    # 计算期望频数
    expected_table = np.outer(row_totals, col_totals) / grand_total
    
    print(f"\n期望频数:")
    print("\t\t男性\t女性")
    for i, effect in enumerate(effect_labels):
        print(f"{effect}\t\t{expected_table[i,0]:.1f}\t{expected_table[i,1]:.1f}")
    
    # 卡方检验
    chi2_stat, p_value, dof, expected = stats.chi2_contingency(observed_table, correction=False)  # 使用未作连续性校正的Pearson统计量，与下方标准Cramér's V计算一致
    
    print(f"\n独立性检验结果:")
    print(f"卡方统计量 χ² = {chi2_stat:.4f}")
    print(f"自由度 df = {dof}")
    print(f"p值 = {p_value:.4f}")
    
    # 效应量 - Cramér's V
    n = grand_total
    cramers_v = np.sqrt(chi2_stat / (n * (min(observed_table.shape) - 1)))
```

```python title="方差齐性检验案例"
def variance_homogeneity_test():

    class_B = np.random.normal(82, 12, 30)  # 均值82，标准差12
    class_C = np.random.normal(80, 15, 30)  # 均值80，标准差15

    class_stats = {}
    for name, scores in classes.items():
        n = len(scores)
        mean_score = np.mean(scores)
        std_score = np.std(scores, ddof=1)
        var_score = np.var(scores, ddof=1)
        class_stats[name] = {'n': n, 'mean': mean_score, 'std': std_score, 'var': var_score}
        
        print(f"{name}\t{n}\t{mean_score:.1f}\t{std_score:.2f}\t\t{var_score:.2f}")
    
    # Bartlett检验：用于独立正态总体样本，原假设下统计量渐近服从自由度为组数减一的卡方分布
    from scipy.stats import bartlett

    print(f"\nBartlett方差齐性检验:")
    print(f"检验统计量: {bartlett_stat:.4f}")
    print(f"p值: {bartlett_p:.4f}")

    from scipy.stats import levene
    
    levene_stat, levene_p = levene(class_A, class_B, class_C, center='median')
    
    print(f"\nLevene方差齐性检验:")
    print(f"检验统计量: {levene_stat:.4f}")
    print(f"p值: {levene_p:.4f}")

    alpha = 0.05

    else:
        print(f"Bartlett检验: 不拒绝方差齐性假设 (p = {bartlett_p:.4f} ≥ {alpha})")
```

```python title="样本方差分布的卡方性质"
def sample_variance_distribution():
    """演示样本方差的卡方分布性质"""
    
    print("样本方差的卡方分布性质:")
    print("=" * 35)
    
    # 从正态总体N(μ, σ²)独立同分布抽样；下述样本方差卡方分布及总体方差置信区间依赖这一前提
    true_mean = 10
    true_std = 3
    true_var = true_std**2
    
    sample_sizes = [5, 10, 20, 30]
    n_experiments = 10000
    
    print(f"总体: N({true_mean}, {true_var})")
    print(f"实验次数: {n_experiments:,}")
    
    fig, axes = plt.subplots(2, 2, figsize=(15, 12))
    axes = axes.flatten()
    
    for idx, n in enumerate(sample_sizes):
        print(f"\n样本量 n = {n}:")
        
        # 进行多次抽样实验
        np.random.seed(42)
        sample_variances = []
        
        for _ in range(n_experiments):
            sample = np.random.normal(true_mean, true_std, n)
            sample_var = np.var(sample, ddof=1)  # 使用无偏估计
            sample_variances.append(sample_var)
        
        sample_variances = np.array(sample_variances)
        
        # 理论：(n-1)S²/σ² ~ χ²(n-1)
        chi2_statistics = (n - 1) * sample_variances / true_var
        
        # 统计结果
        print(f"  样本方差均值: {np.mean(sample_variances):.3f} (理论: {true_var:.3f})")
        print(f"  (n-1)S²/σ²的均值: {np.mean(chi2_statistics):.3f} (理论: {n-1})")
        print(f"  (n-1)S²/σ²的方差: {np.var(chi2_statistics, ddof=1):.3f} (理论: {2*(n-1)})")
        
        # 可视化
        ax = axes[idx]
        
        # 直方图
        ax.hist(chi2_statistics, bins=50, density=True, alpha=0.7, 
                color='skyblue', edgecolor='black', label='样本统计量')
        
        # 理论卡方分布
        x_theory = np.linspace(0, max(chi2_statistics), 1000)
        pdf_theory = stats.chi2.pdf(x_theory, n-1)
        ax.plot(x_theory, pdf_theory, 'r-', linewidth=2, 
                label=f'χ²({n-1})理论分布')
        
        ax.set_title(f'n = {n}, (n-1)S²/σ² ~ χ²({n-1})')
        ax.set_xlabel('(n-1)S²/σ²')
        ax.set_ylabel('概率密度')
        ax.legend()
        ax.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
    
    # 置信区间应用
    print(f"\n置信区间应用:")
    print("-" * 20)
    
    n = 20
    confidence_levels = [0.90, 0.95, 0.99]
    
    print(f"当样本量 n = {n} 时，总体方差σ²的置信区间:")
    
    for conf_level in confidence_levels:
        alpha = 1 - conf_level
        
        # 卡方分布的临界值
        chi2_lower = stats.chi2.ppf(alpha/2, n-1)
        chi2_upper = stats.chi2.ppf(1-alpha/2, n-1)
        
        print(f"\n{conf_level:.0%}置信区间:")
        print(f"  χ²_{alpha/2:.3f}({n-1}) = {chi2_lower:.4f}")
        print(f"  χ²_{1-alpha/2:.3f}({n-1}) = {chi2_upper:.4f}")
        print(f"  σ²的置信区间: [(n-1)S²/{chi2_upper:.3f}, (n-1)S²/{chi2_lower:.3f}]")

        sample_var_example = 10.5

        print(f"  若S² = {sample_var_example}, 则σ²的{conf_level:.0%}置信区间: [{ci_lower:.3f}, {ci_upper:.3f}]")
```

```python title="卡方分布的可加性验证"
def chi2_additivity():
    """验证卡方分布的可加性"""
    
    print("\n\n卡方分布的可加性验证:")
    print("=" * 30)
    print("定理: 若 X₁ ~ χ²(df₁), X₂ ~ χ²(df₂) 且独立")
    print("则 X₁ + X₂ ~ χ²(df₁ + df₂)")
    
    # 设定参数
    df1, df2 = 3, 7
    df_sum = df1 + df2
    n_samples = 50000
    
    print(f"\n验证: χ²({df1}) + χ²({df2}) = χ²({df_sum})")
    print(f"样本量: {n_samples:,}")
    
    np.random.seed(42)
    
    # 方法1: 分别生成两个卡方分布然后相加
    chi2_1 = stats.chi2.rvs(df1, size=n_samples)
    chi2_2 = stats.chi2.rvs(df2, size=n_samples)
    chi2_sum = chi2_1 + chi2_2
    
    # 方法2: 直接生成df1+df2的卡方分布
    chi2_direct = stats.chi2.rvs(df_sum, size=n_samples)
    
    # 方法3: 通过正态分布构造
    # χ²(k) = Z₁² + Z₂² + ... + Zₖ²，其中 Zᵢ ~ N(0,1)
    normals_sum = np.sum(np.random.standard_normal((n_samples, df_sum))**2, axis=1)
    
    # 统计比较
    methods = {
        'χ²(3) + χ²(7)': chi2_sum,
        'χ²(10)直接生成': chi2_direct,
        '10个N(0,1)²之和': normals_sum
    }
    
    print("\n统计特征比较:")
    print("方法\t\t\t均值\t方差\t标准差\tQ1\tQ2\tQ3")
    print("-" * 80)
    
    theoretical_mean = df_sum
    theoretical_var = 2 * df_sum
    theoretical_std = np.sqrt(theoretical_var)
    
    print(f"理论值\t\t\t{theoretical_mean:.2f}\t{theoretical_var:.2f}\t{theoretical_std:.2f}\t-\t-\t-")
    
    for method_name, data in methods.items():
        mean_val = np.mean(data)
        var_val = np.var(data, ddof=1)
        std_val = np.std(data, ddof=1)

        print(f"{method_name}\t{mean_val:.2f}\t{var_val:.2f}\t{std_val:.2f}\t{q1:.2f}\t{q2:.2f}\t{q3:.2f}")
    
    # KS检验比较分布相似性
    from scipy.stats import kstest, ks_2samp
    
    print(f"\n分布相似性检验 (KS检验):")
    print("-" * 35)
    
    # 与理论分布比较
    for method_name, data in methods.items():
        ks_stat, p_value = kstest(data, lambda x: stats.chi2.cdf(x, df_sum))
        print(f"{method_name} vs χ²({df_sum})理论分布:")
        print(f"  KS统计量: {ks_stat:.6f}, p值: {p_value:.6f}")
    
    # 两两比较
    print(f"\n方法间比较:")
    method_pairs = [
        ('χ²(3) + χ²(7)', 'χ²(10)直接生成'),
        ('χ²(3) + χ²(7)', '10个N(0,1)²之和'),
        ('χ²(10)直接生成', '10个N(0,1)²之和')
    ]
    
    for method1, method2 in method_pairs:
        ks_stat, p_value = ks_2samp(methods[method1], methods[method2])
        print(f"{method1} vs {method2}:")
        print(f"  KS统计量: {ks_stat:.6f}, p值: {p_value:.6f}")
    
    # 可视化
    fig, ((ax1, ax2), (ax3, ax4)) = plt.subplots(2, 2, figsize=(15, 12))

    ax1.plot(x, theoretical_pdf, 'k-', linewidth=3, label=f'理论χ²({df_sum})')
    
    colors = ['red', 'blue', 'green']
    alphas = [0.7, 0.7, 0.7]
    
    for i, (method_name, data) in enumerate(methods.items()):
        ax1.hist(data, bins=60, density=True, alpha=alphas[i], 
                color=colors[i], label=method_name, edgecolor='white', linewidth=0.5)
    
    ax1.set_xlabel('x')
    ax1.set_ylabel('概率密度')
    ax1.set_title('卡方分布可加性验证 - PDF比较')
    ax1.legend()
    ax1.grid(True, alpha=0.3)
    ax1.set_xlim(0, 30)
    
    # 2. CDF比较
    x_cdf = np.sort(chi2_sum)
    theoretical_cdf = stats.chi2.cdf(x_cdf, df_sum)
    empirical_cdf = np.arange(1, len(x_cdf) + 1) / len(x_cdf)
    
    ax2.plot(x_cdf, theoretical_cdf, 'k-', linewidth=2, label='理论CDF')
    ax2.plot(x_cdf, empirical_cdf, 'r--', alpha=0.7, linewidth=2, label='经验CDF')
    
    ax2.set_xlabel('x')
    ax2.set_ylabel('累积概率')
    ax2.set_title('累积分布函数比较')
    ax2.legend()
    ax2.grid(True, alpha=0.3)
    
    # 3. Q-Q图
    from scipy.stats import probplot
    probplot(chi2_sum, dist=stats.chi2(df_sum), plot=ax3)
    ax3.set_title('Q-Q图: χ²(3) + χ²(7) vs χ²(10)')
    ax3.grid(True, alpha=0.3)
    
    # 4. 残差分析
    # 比较χ²(3) + χ²(7)和χ²(10)直接生成的差异
    differences = chi2_sum - chi2_direct
    
    ax4.hist(differences, bins=50, alpha=0.7, color='purple', edgecolor='black')
    ax4.axvline(np.mean(differences), color='red', linestyle='--', linewidth=2, 
                label=f'均值 = {np.mean(differences):.4f}')
    ax4.set_xlabel('χ²(3) + χ²(7) - χ²(10)')
    ax4.set_ylabel('频数')
    ax4.set_title('两种方法的差异分布')
    ax4.legend()
    ax4.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
    
    print(f"\n结论:")
    print("1. 三种方法生成的分布在统计上几乎相同")
    print("2. 验证了卡方分布的可加性: χ²(df₁) + χ²(df₂) = χ²(df₁+df₂)")
    print("3. 也验证了卡方分布的定义: k个独立标准正态分布的平方和")
```
