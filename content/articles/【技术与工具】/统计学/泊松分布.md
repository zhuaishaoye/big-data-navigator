## 🤔 什么是泊松分布？

> [!note] 📖 **泊松分布的定义**
> 泊松分布是一种描述非负整数计数的概率分布，常用于固定时间或空间内的事件次数建模，但并不限于稀有事件。若二项分布的n趋于无穷、p趋于0且np趋于固定的λ，则其收敛于Poisson(λ)。

**数学表达式**：
$$P(X = k) = \frac{\lambda^k e^{-\lambda}}{k!}, \quad k = 0, 1, 2, \ldots$$

**记号**：$X \sim \text{Poisson}(\lambda)$

其中 $\lambda > 0$ 是参数，表示所讨论观察区间内事件发生的期望次数；若齐次泊松过程的事件率为 $r$，观察时长为 $t$，则该区间计数的参数为 $\lambda=rt$。

**生活例子：**
- 每小时到达银行的顾客数量
- 每天收到的垃圾邮件数量
- 每年某地区发生的地震次数
- 网站每分钟的访问次数
- 电话交换机每小时接到的电话数

### 2. 泊松过程
```python title="泊松过程的理解"
def demonstrate_poisson_process():
    """演示泊松过程"""
    np.random.seed(42)
    
 # 参数设置
    lambda_rate = 3  # 每小时平均事件发生次数
    time_period = 10  # 观察10小时
    
    print("泊松过程演示:")
    print("=" * 40)
    print(f"参数λ = {lambda_rate} (每小时平均{lambda_rate}次事件)")
    print(f"观察时间: {time_period}小时")
    
 # 1. 生成事件发生时间
 # 泊松过程中，事件间隔时间服从指数分布
    inter_arrival_times = np.random.exponential(1/lambda_rate, 100)
    arrival_times = np.cumsum(inter_arrival_times)
    
 # 只取观察期内的事件
    arrival_times = arrival_times[arrival_times <= time_period]
    n_events = len(arrival_times)
    
 # 2. 计算每个时间单位内的事件数
    hourly_counts = []
    for hour in range(time_period):
        count = np.sum((arrival_times >= hour) & (arrival_times < hour + 1))
        hourly_counts.append(count)
    
 # 3. 可视化
    fig, ((ax1, ax2), (ax3, ax4)) = plt.subplots(2, 2, figsize=(15, 12))
    
 # 事件时间线
    ax1.scatter(arrival_times, np.ones(len(arrival_times)), alpha=0.7, s=50)
    ax1.set_xlim(0, time_period)
    ax1.set_ylim(0.5, 1.5)
    ax1.set_xlabel('时间 (小时)')
    ax1.set_title(f'泊松过程事件时间线 (总共{n_events}个事件)')
    ax1.grid(True, alpha=0.3)
    
 # 每小时事件计数
    hours = np.arange(time_period)
    ax2.bar(hours, hourly_counts, alpha=0.7, color='skyblue', edgecolor='black')
    ax2.axhline(y=lambda_rate, color='red', linestyle='--', 
               label=f'理论均值 λ = {lambda_rate}')
    ax2.set_xlabel('时间 (小时)')
    ax2.set_ylabel('事件数量')
    ax2.set_title('每小时事件计数')
    ax2.legend()
    ax2.grid(True, alpha=0.3)
    
 # 事件计数分布
    unique_counts, count_freq = np.unique(hourly_counts, return_counts=True)
    count_prob = count_freq / len(hourly_counts)
    
 # 理论泊松分布
    x_theory = np.arange(0, max(hourly_counts) + 3)
    theory_prob = stats.poisson.pmf(x_theory, lambda_rate)
    
    ax3.bar(unique_counts, count_prob, alpha=0.7, color='lightgreen', 
           width=0.4, label='观测分布')
    ax3.plot(x_theory, theory_prob, 'ro-', linewidth=2, markersize=6, 
            label=f'理论Poisson({lambda_rate})')
    ax3.set_xlabel('每小时事件数')
    ax3.set_ylabel('概率')
    ax3.set_title('事件计数分布对比')
    ax3.legend()
    ax3.grid(True, alpha=0.3)
    
 # 累积事件数
    cumulative_counts = np.arange(1, len(arrival_times) + 1)
    ax4.step(arrival_times, cumulative_counts, where='post', linewidth=2)
    
 # 理论期望累积
    time_points = np.linspace(0, time_period, 100)
    expected_cumulative = lambda_rate * time_points
    ax4.plot(time_points, expected_cumulative, 'r--', linewidth=2, 
            label=f'理论期望 λt = {lambda_rate}t')
    
    ax4.set_xlabel('时间 (小时)')
    ax4.set_ylabel('累积事件数')
    ax4.set_title('累积事件数 vs 时间')
    ax4.legend()
    ax4.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
    
 # 统计分析
    print(f"\n统计分析结果:")
    print(f"总事件数: {n_events}")
    print(f"平均每小时事件数: {np.mean(hourly_counts):.2f}")
    print(f"方差: {np.var(hourly_counts, ddof=1):.2f}")
    print(f"理论均值 = 方差 = {lambda_rate}")
    
 # 拟合优度检验
    observed_freq = np.bincount(hourly_counts, minlength=max(hourly_counts)+1)
    expected_freq = stats.poisson.pmf(np.arange(len(observed_freq)), lambda_rate) * time_period
    
 # 只考虑期望频数 >= 5 的类别
    valid_categories = expected_freq >= 5
    if np.sum(valid_categories) >= 2:
        chi2_stat = np.sum((observed_freq[valid_categories] - expected_freq[valid_categories])**2 / 
                          expected_freq[valid_categories])
        df = np.sum(valid_categories) - 1 - 1  # -1 for estimated parameter
        p_value = 1 - stats.chi2.cdf(chi2_stat, df) if df > 0 else None
        
        print(f"\n拟合优度检验:")
        print(f"卡方统计量: {chi2_stat:.4f}")
        if p_value is not None:
            print(f"p值: {p_value:.4f}")
            if p_value > 0.05:
                print("结论: 若检验分类及前提有效，在0.05显著性水平下未拒绝指定泊松分布假设；这不证明数据服从泊松分布")
            else:
                print("结论: 若检验分类及前提有效，在0.05显著性水平下拒绝指定泊松分布假设")

demonstrate_poisson_process()
```

### 1. 基本计算
```python title="泊松分布的基本计算"
from scipy import stats
import numpy as np

def poisson_calculations():
    """泊松分布的各种计算"""
    lam = 4  # λ = 4
    
    print(f"泊松分布 Poisson({lam}) 的计算:")
    print("=" * 40)
    
 # 1. 概率质量函数值
    k_values = [0, 1, 2, 3, 4, 5, 8, 10]
    print("概率质量函数值:")
    for k in k_values:
        pmf_value = stats.poisson.pmf(k, lam)
        print(f"P(X = {k}) = {pmf_value:.4f}")
    
 # 2. 累积分布函数值
    print(f"\n累积分布函数值:")
    for k in [2, 4, 6, 8]:
        cdf_value = stats.poisson.cdf(k, lam)
        print(f"P(X ≤ {k}) = {cdf_value:.4f}")
    
 # 3. 分位数
    print(f"\n分位数:")
    percentiles = [0.1, 0.25, 0.5, 0.75, 0.9, 0.95]
    for p in percentiles:
        quantile = stats.poisson.ppf(p, lam)
        print(f"{p:.0%}分位数: {quantile}")
    
 # 4. 区间概率
    print(f"\n区间概率:")
    intervals = [(2, 6), (1, 5), (3, 8)]
    for a, b in intervals:
        prob = stats.poisson.cdf(b, lam) - stats.poisson.cdf(a-1, lam)
        print(f"P({a} ≤ X ≤ {b}) = {prob:.4f}")
    
 # 5. 特殊概率
    print(f"\n特殊概率:")
    
 # 至少发生一次
    prob_at_least_one = 1 - stats.poisson.pmf(0, lam)
    print(f"P(X ≥ 1) = {prob_at_least_one:.4f}")
    
 # 超过平均值
    prob_above_mean = 1 - stats.poisson.cdf(lam, lam)
    print(f"P(X > λ) = P(X > {lam}) = {prob_above_mean:.4f}")
    
 # 在均值±标准差范围内
    std_dev = np.sqrt(lam)
    lower = max(0, int(lam - std_dev))
    upper = int(lam + std_dev)
    prob_within_1std = stats.poisson.cdf(upper, lam) - stats.poisson.cdf(lower-1, lam)
    print(f"P({lower} ≤ X ≤ {upper}) = {prob_within_1std:.4f}")

poisson_calculations()
```

### 2. 参数估计
```python title="泊松分布参数估计"
def poisson_parameter_estimation():
    """泊松分布的参数估计"""
 # 真实参数
    true_lambda = 6.5
    n_samples = 1000
    
    np.random.seed(42)
    sample_data = np.random.poisson(true_lambda, n_samples)
    
    print("泊松分布参数估计:")
    print("=" * 40)
    print(f"真实参数 λ = {true_lambda}")
    print(f"样本量 n = {n_samples}")
    
 # 1. 矩估计法 (方法1)
    sample_mean = np.mean(sample_data)
    lambda_moment = sample_mean  # 泊松分布的矩估计就是样本均值
    
    print(f"\n矩估计法:")
    print(f"样本均值: {sample_mean:.4f}")
    print(f"λ的矩估计: {lambda_moment:.4f}")
    
 # 2. 最大似然估计 (方法2)
 # 对于泊松分布，MLE也是样本均值
    lambda_mle = sample_mean
    
    print(f"\n最大似然估计:")
    print(f"λ的MLE: {lambda_mle:.4f}")
    
 # 3. 置信区间
 # 使用正态近似 (当λ较大时)
    se = np.sqrt(lambda_mle / n_samples)  # 标准误差
    z_975 = stats.norm.ppf(0.975)  # 97.5%分位数
    
    ci_lower = lambda_mle - z_975 * se
    ci_upper = lambda_mle + z_975 * se
    
    print(f"\n95%置信区间:")
    print(f"[{ci_lower:.4f}, {ci_upper:.4f}]")
    print(f"真实值是否在置信区间内: {ci_lower <= true_lambda <= ci_upper}")
    
 # 4. 精确置信区间 (基于卡方分布)
 # 对于泊松分布的精确置信区间
    sum_x = np.sum(sample_data)
    
 # 下界: χ²_{2Σx, α/2} / (2n)
 # 上界: χ²_{2(Σx+1), 1-α/2} / (2n)
    chi2_lower = 0.0 if sum_x == 0 else stats.chi2.ppf(0.025, 2 * sum_x)
    chi2_upper = stats.chi2.ppf(0.975, 2 * (sum_x + 1))
    
    exact_ci_lower = chi2_lower / (2 * n_samples)
    exact_ci_upper = chi2_upper / (2 * n_samples)
    
    print(f"\n精确95%置信区间:")
    print(f"[{exact_ci_lower:.4f}, {exact_ci_upper:.4f}]")
    
 # 5. 不同样本量下的估计精度
    sample_sizes = [50, 100, 500, 1000, 5000]
    
    print(f"\n不同样本量下的估计精度:")
    print("样本量\t估计值\t\t标准误差\t95%置信区间宽度")
    print("-" * 60)
    
    for n in sample_sizes:
        sample = np.random.poisson(true_lambda, n)
        estimate = np.mean(sample)
        se_n = np.sqrt(estimate / n)
        ci_width = 2 * z_975 * se_n
        
        print(f"{n}\t{estimate:.4f}\t\t{se_n:.4f}\t\t{ci_width:.4f}")

poisson_parameter_estimation()
```

### 1. 客服呼叫中心分析
```python title="呼叫中心业务分析"
def call_center_analysis():
    """呼叫中心的泊松分布应用"""
    np.random.seed(42)
    
 # 模拟一天24小时的呼叫数据
 # 不同时段的呼叫率不同
    hourly_rates = [
        1, 1, 0.5, 0.5, 0.5, 1,     # 0-5点 (夜间)
        3, 5, 8, 10, 12, 12,        # 6-11点 (上午)
        15, 14, 12, 13, 15, 16,     # 12-17点 (下午)
        12, 10, 8, 6, 4, 2          # 18-23点 (晚上)
    ]
    
    print("呼叫中心泊松分布分析:")
    print("=" * 50)
    
 # 生成一周的数据 (7天)
    daily_calls = []
    hourly_calls_all = []
    
    for day in range(7):
        daily_total = 0
        for hour, rate in enumerate(hourly_rates):
            n_calls = np.random.poisson(rate)
            hourly_calls_all.append(n_calls)
            daily_total += n_calls
        daily_calls.append(daily_total)
    
 # 分析结果
    print(f"一周总呼叫量: {sum(daily_calls)}")
    print(f"日均呼叫量: {np.mean(daily_calls):.1f}")
    print(f"日呼叫量标准差: {np.std(daily_calls, ddof=1):.1f}")
    
 # 高峰时段分析 (12-17点)
    peak_hours = hourly_rates[12:18]
    peak_mean = np.mean(peak_hours)
    
    print(f"\n高峰时段分析 (12-17点):")
    print(f"平均每小时呼叫量: {peak_mean:.1f}")
    
 # 服务水平分析
 # 假设每个客服每小时能处理8个电话
    agents_per_hour = 8
    
    print(f"\n人员配置分析:")
    for hour in range(24):
        expected_calls = hourly_rates[hour]
        min_agents = int(np.ceil(expected_calls / agents_per_hour))
        
 # 假设每人每小时固定处理8通，以小时呼叫量95%分位数估算座席数；不保证等待时间或接通服务水平
 # P(X ≤ k) ≥ 0.95
        calls_95 = stats.poisson.ppf(0.95, expected_calls)
        agents_95 = int(np.ceil(calls_95 / agents_per_hour))
        
        if hour in [9, 10, 14, 15]:  # 只显示几个关键时段
            print(f"{hour:2d}:00 - 期望{expected_calls:4.1f}通, "
                  f"按期望工作量估算{min_agents}人, 按小时呼叫量95%分位数估算{agents_95}人（非服务水平保证）")
    
 # 等待时间分析
 # 假设服务时间服从指数分布，平均3分钟
    service_rate = 20  # 每小时服务20个客户 (3分钟/个)
    
    print(f"\n等待时间分析 (高峰期):")
    peak_arrival_rate = 15  # 高峰期每小时15个电话
    
 # 单座席M/M/1稳态模型：泊松到达、独立指数服务时间，且到达率小于服务率；此处服务率20通/小时与前述排班假设不同
    utilization = peak_arrival_rate / service_rate
    print(f"系统利用率: {utilization:.1%}")
    
    if utilization < 1:
        avg_wait_time = utilization / (service_rate * (1 - utilization)) * 60  # 转换为分钟
        print(f"预期等待时间: {avg_wait_time:.1f}分钟")
    else:
        print("系统过载，需要增加座席")
    
 # 可视化
    fig, ((ax1, ax2), (ax3, ax4)) = plt.subplots(2, 2, figsize=(15, 12))
    
 # 24小时呼叫率
    hours = np.arange(24)
    ax1.bar(hours, hourly_rates, alpha=0.7, color='skyblue')
    ax1.set_xlabel('小时')
    ax1.set_ylabel('平均呼叫量')
    ax1.set_title('24小时呼叫率分布')
    ax1.grid(True, alpha=0.3)
    
 # 某个高峰时段的实际分布
    peak_hour_calls = [np.random.poisson(15) for _ in range(100)]  # 模拟100天的14点数据
    
    unique_calls, call_counts = np.unique(peak_hour_calls, return_counts=True)
    call_probs = call_counts / len(peak_hour_calls)
```

```python title="泊松分布的正态近似"
def poisson_normal_approximation():
    """演示泊松分布的正态近似"""
    lambda_values = [1, 5, 10, 25, 50]
    
    print("泊松分布的正态近似:")
    print("=" * 40)
    
    fig, axes = plt.subplots(2, 3, figsize=(15, 10))
    axes = axes.flatten()
    
    for i, lam in enumerate(lambda_values):
        if i >= len(axes):
            break
            
 # 泊松分布
        x_max = int(lam + 4 * np.sqrt(lam))
        x = np.arange(0, x_max + 1)
        poisson_pmf = stats.poisson.pmf(x, lam)
        
 # 正态近似 (连续性修正)
        x_normal = np.linspace(0, x_max, 1000)
        normal_pdf = stats.norm.pdf(x_normal, lam, np.sqrt(lam))
        
 # 绘图
        axes[i].bar(x, poisson_pmf, alpha=0.7, width=0.8, 
                   color='skyblue', label=f'Poisson({lam})')
        axes[i].plot(x_normal, normal_pdf, 'r-', linewidth=2, 
                    label=f'N({lam}, {lam})')
        
        axes[i].set_title(f'λ = {lam}')
        axes[i].set_xlabel('k')
        axes[i].set_ylabel('概率/密度')
        axes[i].legend()
        axes[i].grid(True, alpha=0.3)
        
 # 计算近似误差
 # 选择几个点计算误差
        test_points = [int(lam), int(lam + np.sqrt(lam)), int(lam - np.sqrt(lam))]
        test_points = [max(0, p) for p in test_points if p <= x_max]
        
        print(f"\nλ = {lam} 时的近似误差:")
        for point in test_points:
            poisson_prob = stats.poisson.pmf(point, lam)
 # 连续性修正: P(X = k) ≈ P(k-0.5 < Y < k+0.5)
            normal_prob = (stats.norm.cdf(point + 0.5, lam, np.sqrt(lam)) - 
                          stats.norm.cdf(point - 0.5, lam, np.sqrt(lam)))
            error = abs(poisson_prob - normal_prob)
            
            print(f"  P(X = {point}): 泊松={poisson_prob:.4f}, "
                  f"正态={normal_prob:.4f}, 误差={error:.4f}")
    
 # 删除多余的子图
    if len(lambda_values) < len(axes):
        for j in range(len(lambda_values), len(axes)):
            fig.delaxes(axes[j])
    
    plt.tight_layout()
    plt.show()
    
 # 近似准则
    print(f"\n正态近似的使用准则:")
    print("1. λ增大时，标准化泊松分布渐近于标准正态分布")
    print("2. λ=5、10、30可作为比较近似误差的示例，不是通用精度阈值")
    print("3. 是否采用正态近似需结合概率区域及绝对或相对误差要求，尤其应检查尾部概率")
    print("4. 使用连续性修正可以提高小λ时的近似精度")

poisson_normal_approximation()
```

```python title="二项分布到泊松分布的极限"
 # 不同的 n 值，相应调整 p = λ/n
    n_values = [10, 50, 100, 500, 1000]
    
    print("二项分布收敛到泊松分布:")
    print("=" * 40)
    print(f"固定 λ = np = {lambda_fixed}")
    
    fig, axes = plt.subplots(2, 3, figsize=(15, 10))
    axes = axes.flatten()
    
    for i, n in enumerate(n_values):
        if i >= len(axes):
            break
            
        p = lambda_fixed / n
        
 # 计算合理的x范围
        x_max = min(n, int(lambda_fixed + 4 * np.sqrt(lambda_fixed)))
        x = np.arange(0, x_max + 1)
        
 # 二项分布
        binomial_pmf = stats.binom.pmf(x, n, p)
        
 # 泊松分布
        poisson_pmf = stats.poisson.pmf(x, lambda_fixed)
        
 # 绘图
        axes[i].plot(x, binomial_pmf, 'bo-', markersize=4, alpha=0.7, 
                    label=f'Binomial({n}, {p:.4f})')
        axes[i].plot(x, poisson_pmf, 'r^-', markersize=4, 
                    label=f'Poisson({lambda_fixed})')
        
        axes[i].set_title(f'n = {n}, p = {p:.4f}')
        axes[i].set_xlabel('k')
        axes[i].set_ylabel('P(X = k)')
        axes[i].legend()
        axes[i].grid(True, alpha=0.3)
        
 # 计算KL散度作为距离度量
 # 只在两个分布都有意义的范围内计算
        valid_range = (binomial_pmf > 1e-10) & (poisson_pmf > 1e-10)
        if np.sum(valid_range) > 0:
            kl_div = np.sum(binomial_pmf[valid_range] * 
                           np.log(binomial_pmf[valid_range] / poisson_pmf[valid_range]))
        else:
            kl_div = np.inf
        
        print(f"n = {n:4d}, p = {p:.6f}: 截断并过滤后的KL项部分和 = {kl_div:.6f}（非完整KL散度）")
    
 # 删除多余的子图
    if len(n_values) < len(axes):
        for j in range(len(n_values), len(axes)):
            fig.delaxes(axes[j])
    
    plt.tight_layout()
    plt.show()
    
 # 收敛条件
    print(f"\n泊松近似的使用条件:")
    print("1. n 很大 (通常 n ≥ 20)")
    print("2. p 很小 (通常 p ≤ 0.1)")
    print("3. np 为中等大小 (通常 1 ≤ np ≤ 10)")
    print("4. 满足以上条件时，Binomial(n,p) ≈ Poisson(np)")
```
