## 🤔 什么是四分位距？

> [!note] 📖 **四分位距的定义**
> 四分位距（IQR）是第三四分位数（Q3）与第一四分位数（Q1）之间的差值，表示中间50%数据的分散程度。

**公式**：$IQR = Q_3 - Q_1$

**生活例子：**
- 班级成绩的四分位距告诉我们"中等学生"的成绩差距有多大
- 员工工资的四分位距反映"普通员工"间的薪资差异
- 房价的四分位距显示"主流房源"的价格跨度

## 📊 四分位距的计算

### 基本计算步骤
1. 将数据从小到大排序
2. 找到Q1（第25百分位数）
3. 找到Q3（第75百分位数）
4. 计算IQR = Q3 - Q1

```python title="四分位距计算示例"
import numpy as np
import matplotlib.pyplot as plt

# 示例数据：学生考试成绩
scores = [65, 70, 72, 75, 78, 80, 82, 85, 88, 90, 92, 95, 98]

# 计算四分位数
Q1 = np.percentile(scores, 25)
Q2 = np.percentile(scores, 50)  # 中位数
Q3 = np.percentile(scores, 75)

# 计算四分位距
IQR = Q3 - Q1

print(f"数据: {scores}")
print(f"Q1 (第一四分位数): {Q1}")
print(f"Q2 (中位数): {Q2}")
print(f"Q3 (第三四分位数): {Q3}")
print(f"IQR (四分位距): {IQR}")

print(f"\n数据分布解释:")
print(f"第25百分位数: {Q1}")
print(f"第50百分位数（中位数）: {Q2}")
print(f"第75百分位数: {Q3}")
print(f"第25至第75百分位数的阈值跨度为{IQR}分；有限样本中落在该区间的人数比例不一定恰好为50%")
```

## 💻 Python实现方法

### 方法1：使用NumPy
```python title="NumPy计算四分位距"
import numpy as np

def calculate_iqr_numpy(data):
    """使用NumPy计算四分位距"""
    Q1 = np.percentile(data, 25)
    Q3 = np.percentile(data, 75)
    IQR = Q3 - Q1
    return Q1, Q3, IQR

# 示例数据
data = [10, 15, 20, 25, 30, 35, 40, 45, 50, 55, 60]
Q1, Q3, IQR = calculate_iqr_numpy(data)

print(f"NumPy方法:")
print(f"Q1: {Q1}, Q3: {Q3}, IQR: {IQR}")
```

## 📈 四分位距的特性

### 1. 抗异常值特性
```python title="四分位距vs标准差的抗异常值能力"

def compare_dispersion_measures(data, label):
    """比较不同离散程度指标"""
    std = np.std(data, ddof=1)
    IQR = stats.iqr(data)
    range_val = np.max(data) - np.min(data)
    
    print(f"{label}:")
    print(f"  标准差: {std:.2f}")
    print(f"  四分位距: {IQR:.2f}")
    print(f"  极差: {range_val:.2f}")
    return std, IQR, range_val

print("抗异常值能力对比:")

measures = ['标准差', '四分位距', '极差']

colors = ['red' if x > 50 else 'orange' if x > 10 else 'green' for x in changes]

ax2.bar(measures, changes, color=colors)
```

## 🎯 四分位距的应用

### 1. 异常值检测（IQR法则）
```python title="IQR异常值检测"
def detect_outliers_iqr(data, k=1.5):
    """
    使用IQR方法检测异常值
    k=1.5: 温和异常值
    k=3.0: 极端异常值
    """
    Q1 = np.percentile(data, 25)
    Q3 = np.percentile(data, 75)
    IQR = Q3 - Q1
    
 # 定义异常值边界
    lower_bound = Q1 - k * IQR
    upper_bound = Q3 + k * IQR
    
 # 识别异常值
    outliers = []
    normal_data = []
    
    for value in data:
        if value < lower_bound or value > upper_bound:
            outliers.append(value)
        else:
            normal_data.append(value)
    
    return {
        'outliers': outliers,
        'normal_data': normal_data,
        'lower_bound': lower_bound,
        'upper_bound': upper_bound,
        'Q1': Q1,
        'Q3': Q3,
        'IQR': IQR
    }

# 示例：员工工资数据
salaries = [3000, 3200, 3500, 3800, 4000, 4200, 4500, 4800, 
           5000, 5200, 5500, 8000, 12000, 25000]

# 检测异常值
result = detect_outliers_iqr(salaries, k=1.5)

print(f"工资异常值检测结果:")
print(f"Q1: {result['Q1']:,.0f}元")
print(f"Q3: {result['Q3']:,.0f}元")
print(f"IQR: {result['IQR']:,.0f}元")
print(f"正常范围: [{result['lower_bound']:,.0f}, {result['upper_bound']:,.0f}]元")
print(f"异常值: {[f'{x:,.0f}' for x in result['outliers']]}元")
print(f"异常值占比: {len(result['outliers'])/len(salaries)*100:.1f}%")

# 可视化
plt.figure(figsize=(12, 6))

# 箱线图
plt.subplot(1, 2, 1)
plt.boxplot(salaries, vert=True)
plt.ylabel('工资 (元)')
plt.title('工资分布箱线图')

# 散点图标注异常值
plt.subplot(1, 2, 2)
normal_indices = [i for i, x in enumerate(salaries) if x in result['normal_data']]
outlier_indices = [i for i, x in enumerate(salaries) if x in result['outliers']]

plt.scatter(normal_indices, [salaries[i] for i in normal_indices], 
           color='blue', label='正常值', alpha=0.7)
plt.scatter(outlier_indices, [salaries[i] for i in outlier_indices], 
           color='red', label='异常值', s=100)

plt.axhline(result['lower_bound'], color='orange', linestyle='--', alpha=0.7, label='下边界')
plt.axhline(result['upper_bound'], color='orange', linestyle='--', alpha=0.7, label='上边界')
plt.axhline(result['Q1'], color='green', linestyle=':', alpha=0.7, label='Q1')
plt.axhline(result['Q3'], color='green', linestyle=':', alpha=0.7, label='Q3')

plt.xlabel('员工编号')
plt.ylabel('工资 (元)')
plt.title('异常值检测结果')
plt.legend()

plt.tight_layout()
plt.show()
```

### 2. 数据质量评估
```python title="数据质量评估"
def data_quality_assessment(data, label="数据"):
    """基于IQR、偏度和人为扣分阈值的分布评分演示，不是通用数据质量标准；分布偏斜或超出IQR围栏不等于数据错误，IQR/中位数的变异解释仅适用于中位数为正且远离零的适当指标。"""
    Q1 = np.percentile(data, 25)
    Q2 = np.percentile(data, 50)
    Q3 = np.percentile(data, 75)
    IQR = Q3 - Q1
    
 # 检测异常值
    outlier_result = detect_outliers_iqr(data)
    outlier_rate = len(outlier_result['outliers']) / len(data)
    
 # 数据分布评估
    mean = np.mean(data)
    skewness = stats.skew(data)
    
 # 数据质量评分
    quality_score = 100
    
 # 异常值扣分
    if outlier_rate > 0.1:  # 超过10%异常值
        quality_score -= 30
    elif outlier_rate > 0.05:  # 5%-10%异常值
        quality_score -= 20
    elif outlier_rate > 0.02:  # 2%-5%异常值
        quality_score -= 10
    
 # 偏度扣分
    if abs(skewness) > 2:  # 严重偏斜
        quality_score -= 20
    elif abs(skewness) > 1:  # 中度偏斜
        quality_score -= 10
    
 # 稳定性评估（基于IQR相对变异）
    cv_iqr = IQR / Q2 if Q2 != 0 else float('inf')
    if cv_iqr > 1:  # 高变异
        quality_score -= 15
    elif cv_iqr > 0.5:  # 中等变异
        quality_score -= 5
    
    print(f"{label}质量评估报告:")
    print(f"{'='*40}")
    print(f"样本量: {len(data)}")
    print(f"四分位数: Q1={Q1:.2f}, Q2={Q2:.2f}, Q3={Q3:.2f}")
    print(f"四分位距: {IQR:.2f}")
    print(f"异常值数量: {len(outlier_result['outliers'])} ({outlier_rate:.1%})")
    print(f"偏度: {skewness:.2f}")
    print(f"IQR变异系数: {cv_iqr:.2f}")
    print(f"示例规则评分（不代表实际数据质量）: {quality_score}/100")
    
 # 质量等级
    if quality_score >= 90:
        grade = "优秀"
    elif quality_score >= 80:
        grade = "良好"
    elif quality_score >= 70:
        grade = "一般"
    else:
        grade = "较差"
    
    print(f"示例规则等级（不代表实际数据质量）: {grade}")
    print(f"{'='*40}")
    
    return {
        'quality_score': quality_score,
        'grade': grade,
        'outlier_rate': outlier_rate,
        'skewness': skewness,
        'cv_iqr': cv_iqr
    }

# 评估不同质量的数据集
np.random.seed(42)

# 正态分布模拟数据（假设在示例评分规则下得分较高，并非已确认高质量）
high_quality_data = np.random.normal(100, 10, 1000)

# 正态主体与高值混合的模拟数据（假设在示例评分规则下得分居中，并非已确认中等质量）
medium_quality_data = np.concatenate([
    np.random.normal(100, 15, 950),
    np.random.uniform(150, 200, 50)  # 添加一些异常值
])

# 指数分布与高值混合的模拟数据（假设在示例评分规则下得分较低，并非已确认低质量）
low_quality_data = np.concatenate([
    np.random.exponential(20, 800),
    np.random.uniform(200, 300, 200)  # 大量异常值
])

# 评估三个数据集
data_sets = [
    (high_quality_data, "正态分布模拟数据集"),
    (medium_quality_data, "正态主体与高值混合模拟数据集"),
    (low_quality_data, "指数分布与高值混合模拟数据集")
]

results = []
for data, label in data_sets:
    result = data_quality_assessment(data, label)
    results.append((label, result))
```

## 🔍 四分位距的优缺点

### ✅ 优点
1. **抗异常值**：相较极差和标准差，通常对尾部极端值更稳健，但并非完全不受异常值或样本构成变化影响
2. **直观易懂**：表示中间50%数据的分散程度
3. **适用性广**：适用于各种分布形状
4. **计算简单**：只需要两个分位数
5. **业务意义清晰**：便于制定分层策略

### ❌ 缺点
1. **信息损失**：忽略了25%和75%以外的数据
2. **效率较低**：在正态分布下效率不如标准差
3. **离散程度描述不完整**：只反映中间部分的变异
4. **不适合代数运算**：无法像方差那样进行合并计算

## 🛠️ 实用技巧

### 1. 选择合适的k值
```python title="不同k值的异常值检测效果"
# 生成测试数据
np.random.seed(42)
test_data = np.concatenate([
    np.random.normal(50, 10, 900),  # 正常数据
    np.random.uniform(100, 120, 50),  # 温和异常值
    np.random.uniform(150, 200, 50)   # 极端异常值
])

k_values = [1.0, 1.5, 2.0, 2.5, 3.0]
print("不同k值的异常值检测效果:")
print("k值\t异常值数量\t异常值率\t主要检测范围")
print("-" * 50)

for k in k_values:
    result = detect_outliers_iqr(test_data, k=k)
    outlier_rate = len(result['outliers']) / len(test_data)
    
    if k == 1.5:
        detection_range = "常规异常值"
    elif k >= 3.0:
        detection_range = "极端异常值"
    else:
        detection_range = "中等异常值"
    
    print(f"{k}\t{len(result['outliers'])}\t\t{outlier_rate:.1%}\t\t{detection_range}")

print("\n💡 建议:")
print("k=1.5: 适合一般异常值检测，比较敏感")
print("k=2.0: 平衡敏感性和稳定性")
print("k=3.0: 只检测极端异常值，比较保守")
```

### 2. 时间序列中的滚动IQR
```python title="滚动四分位距分析"
# 生成时间序列数据
dates = pd.date_range('2023-01-01', periods=365, freq='D')
np.random.seed(42)

# 基础趋势 + 季节性 + 噪声 + 偶发异常
trend = np.linspace(100, 120, 365)
seasonal = 10 * np.sin(2 * np.pi * np.arange(365) / 365.25 * 4)  # 季节性
noise = np.random.normal(0, 3, 365)

# 添加一些异常值
outlier_indices = np.random.choice(365, 20, replace=False)
outlier_values = trend + seasonal + noise
for idx in outlier_indices:
    outlier_values[idx] += np.random.choice([-20, 20]) * np.random.uniform(1, 2)

ts_data = pd.DataFrame({
    'date': dates,
    'value': outlier_values
})

# 计算滚动IQR
window = 30  # 30天滚动窗口
ts_data['rolling_Q1'] = ts_data['value'].rolling(window).quantile(0.25)
ts_data['rolling_Q3'] = ts_data['value'].rolling(window).quantile(0.75)
ts_data['rolling_IQR'] = ts_data['rolling_Q3'] - ts_data['rolling_Q1']

# 基于滚动IQR的异常检测
ts_data['lower_bound'] = ts_data['rolling_Q1'] - 1.5 * ts_data['rolling_IQR']
ts_data['upper_bound'] = ts_data['rolling_Q3'] + 1.5 * ts_data['rolling_IQR']
ts_data['has_window'] = ts_data['lower_bound'].notna() & ts_data['upper_bound'].notna()
ts_data['is_outlier'] = ts_data['has_window'] & (
    (ts_data['value'] < ts_data['lower_bound']) |
    (ts_data['value'] > ts_data['upper_bound'])
)

# 可视化
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(15, 10))

# 时间序列和异常值
normal_data = ts_data[ts_data['has_window'] & ~ts_data['is_outlier']]
outlier_data = ts_data[ts_data['is_outlier']]
unassessed_data = ts_data[~ts_data['has_window']]

ax1.plot(normal_data['date'], normal_data['value'], 'b-', alpha=0.7, label='围栏内观测')
ax1.scatter(unassessed_data['date'], unassessed_data['value'], color='gray', s=20, label='窗口不足，未判定')
ax1.scatter(outlier_data['date'], outlier_data['value'], color='red', s=50, label='异常值', zorder=5)
ax1.fill_between(ts_data['date'], ts_data['lower_bound'], ts_data['upper_bound'], 
                alpha=0.2, color='gray', label='正常范围')
ax1.set_title('基于滚动IQR的时间序列异常检测')
ax1.set_ylabel('数值')
ax1.legend()

# 滚动IQR变化
ax2.plot(ts_data['date'], ts_data['rolling_IQR'], 'g-', linewidth=2)
ax2.set_title(f'{window}天滚动IQR变化')
ax2.set_xlabel('日期')
ax2.set_ylabel('IQR')

plt.tight_layout()
plt.show()

print(f"滚动IQR异常检测结果:")
print(f"检测到异常值: {ts_data['is_outlier'].sum()}个")
print(f"有效窗口观测数: {ts_data['has_window'].sum()}个")
print(f"窗口不足、未判定观测数: {(~ts_data['has_window']).sum()}个")
print(f"有效窗口内异常候选率: {ts_data.loc[ts_data['has_window'], 'is_outlier'].mean():.1%}")
```
