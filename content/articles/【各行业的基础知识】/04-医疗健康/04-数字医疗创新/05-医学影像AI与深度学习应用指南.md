## 3. 评估：不要只看一个准确率

| 评估层 | 指标 | 说明 |
| --- | --- | --- |
| 检索 | Hit@K、MRR、版本正确率、权限过滤 | 正确内容是否被召回且可用 |

## 5. Python：检索与任务评估

### 检索、Agent 或模型任务实现

```python
import pandas as pd
from sklearn.model_selection import TimeSeriesSplit
from sklearn.metrics import roc_auc_score
from sklearn.ensemble import HistGradientBoostingClassifier

def evaluate_time_aware(frame: pd.DataFrame, features: list[str], target: str) -> list[dict]:
    frame = frame.sort_values("event_date").reset_index(drop=True)
    scores = []
    for train_idx, test_idx in TimeSeriesSplit(n_splits=5, gap=7).split(frame):
        train, test = frame.iloc[train_idx], frame.iloc[test_idx]
        if train[target].nunique() < 2 or test[target].nunique() < 2:
            continue
        model = HistGradientBoostingClassifier(max_depth=4)
        model.fit(train[features], train[target])
        proba = model.predict_proba(test[features])[:, 1]
        scores.append({"model_auc": roc_auc_score(test[target], proba)})
    return scores
```

```sql
-- 医学影像AI与深度学习应用指南：结构贡献诊断（病种 × 支付方式）
WITH scoped AS (
  SELECT
    patient_id AS entity_key,
    event_id,
    event_date AS event_date,
    dimension_value AS primary_dim,
    stage_code AS secondary_dim,
    metric_value,
    status
  FROM fact_medical_event
  WHERE event_date >= DATE '2026-01-01'
    AND event_date <  DATE '2026-04-01'
    AND subject_key = 'ai'
    AND is_deleted = 0
),
dedup AS (
  SELECT *
  FROM (
    SELECT
      scoped.*,
      ROW_NUMBER() OVER (
        PARTITION BY event_id
        ORDER BY event_date DESC
      ) AS version_no
    FROM scoped
  ) ranked
  WHERE version_no = 1
),
grouped AS (
  SELECT
    primary_dim,
    secondary_dim,
    COUNT(DISTINCT entity_key) AS entity_cnt,
    COUNT(DISTINCT event_id) AS event_cnt,
    SUM(metric_value) AS metric_sum,
    AVG(metric_value) AS metric_avg
  FROM dedup
  WHERE status IN ('valid', 'completed')
  GROUP BY primary_dim, secondary_dim
),
contribution AS (
  SELECT
    *,
    metric_sum * 1.0 / NULLIF(SUM(metric_sum) OVER (PARTITION BY secondary_dim), 0) AS contribution_ratio,
    metric_avg - AVG(metric_avg) OVER (PARTITION BY secondary_dim) AS delta_to_peer
  FROM grouped
)
SELECT
  '费用偏差率' AS checked_metric,
  primary_dim,
  secondary_dim,
  entity_cnt,
  event_cnt,
  ROUND(metric_sum, 4) AS metric_sum,
  ROUND(contribution_ratio, 4) AS contribution_ratio,
  ROUND(delta_to_peer, 4) AS delta_to_peer
FROM contribution
ORDER BY ABS(delta_to_peer) DESC, entity_cnt DESC;
```

- 对失败分类：检索缺失、上下文过长、工具失败、权限拒绝、模型拒答、结果不合规。
- 设置单位任务成本、每租户预算和异常调用告警。

### 扩展代码一：从医学影像AI与深度学习应用指南回到可核对明细

```sql
-- 医学影像AI与深度学习应用指南：源表与结果表按对象对账
WITH source_side AS (
  SELECT
    patient_id AS entity_key,
    COUNT(DISTINCT event_id) AS event_cnt,
    SUM(metric_value) AS metric_sum
  FROM fact_medical_event
  WHERE event_date >= :start_date
    AND event_date <  :end_date
    AND subject_key = 'ai'
    AND is_deleted = 0
  GROUP BY patient_id
),
result_side AS (
  SELECT
    patient_id AS entity_key,
    event_cnt,
    metric_sum
  FROM fact_medical_event_result
  WHERE stat_date BETWEEN :start_date AND :end_date
    AND subject_key = 'ai'
)
SELECT
  COALESCE(s.entity_key, r.entity_key) AS entity_key,
  s.event_cnt AS source_event_cnt,
  r.event_cnt AS result_event_cnt,
  s.metric_sum AS source_metric_sum,
  r.metric_sum AS result_metric_sum,
  COALESCE(r.metric_sum, 0) - COALESCE(s.metric_sum, 0) AS metric_diff
FROM source_side s
FULL OUTER JOIN result_side r USING (entity_key)
WHERE ABS(COALESCE(r.metric_sum, 0) - COALESCE(s.metric_sum, 0)) > 0
   OR COALESCE(s.event_cnt, 0) <> COALESCE(r.event_cnt, 0)
ORDER BY ABS(metric_diff) DESC;
```

执行这段 SQL 后，不要只看总量。应同时保存查询参数、口径版本、异常对象清单和对应原始记录抽样。若结果依赖人工过滤，要把过滤条件登记为规则，而不是留在个人脚本里。
