## 3. 案例一：华南某支付日志平台

### 案例 1：华南某支付日志平台的“Kafka消费积压排查与实时链路优化项目”现场

这是一个结合公开方法、常见工程故障和脱敏数据构造的复合案例，不对应任何单一企业事实。

- **标题落点**：先确认业务对象和系统边界，再讨论工具或参数。本例把“Kafka消费积压排查与实时链路优化项目”映射到“Kafka 消费积压与实时链路”中的生产侧，而不是泛泛讨论概念。
- **触发事件**：夜间 Lag 周期性升高，但白天业务量更高时反而正常。
- **业务与数据范围**：Broker 磁盘队列、分区副本状态、消费者提交位移、定时批任务和网络指标。数据侧重点采集Topic/Partition 写入速率、消息大小、错误码，并与项目管理、需求管理、代码仓库中的时间、状态和责任人信息对齐。
- **需要回答的问题**：消息是否突增、分区键是否导致热点、重试是否放大流量。
- **关键障碍**：夜间快照任务争抢磁盘 IO，导致一个 Broker 请求时延升高，消费者从副本拉取数据变慢。同时要防止范围持续扩张把表面指标带偏。
- **处理过程**：把 Broker 指标与消费指标按分钟对齐，迁移快照窗口并对慢盘做副本隔离和容量告警。实施时把“按 Topic-Partition 同时看生产速率、消费速率、Lag 和 commit rate”写成上线检查项，并为每一项保留日志、样本和负责人。
- **验证方式**：复合案例将问题从业务流量误判纠正为基础设施争抢，并建立跨层时间线排查模板。验收至少同时看主指标、Topic/Partition 写入速率、消息大小、错误码和一次故障回放，不能只截取优化后最快的一次结果。
- **可迁移结论**：遇到类似标题时，先按“把生产速率、分区分布、消费能力、Broker 状态和下游 Sink 时延放到同一条时间线上判断，避免看到 Lag 就直接扩容。”建立判断顺序，再决定是否调整代码、资源、口径或组织流程。

## 4. 代码实现：从可运行到可维护

### 专题实现代码

```python
from kafka import KafkaConsumer, TopicPartition

TOPIC_NAME = "orders.events.v2"
GROUP_ID = "g_kafka_uh2sws"
consumer = KafkaConsumer(
    bootstrap_servers=["broker-a:9092", "broker-b:9092"],
    group_id=GROUP_ID,
    enable_auto_commit=False,
)

partitions = [TopicPartition(TOPIC_NAME, p) for p in consumer.partitions_for_topic(TOPIC_NAME)]
end_offsets = consumer.end_offsets(partitions)
committed = {p: consumer.committed(p) or 0 for p in partitions}
lag_rows = [
    {
        "partition": p.partition,
        "lag": end_offsets[p] - committed[p],
        "committed": committed[p],
        "end_offset": end_offsets[p],
    }
    for p in partitions
]
lag_rows.sort(key=lambda row: row["lag"], reverse=True)
print({"topic": "Kafka消费积压排查与实时链路优化项目", "hottest": lag_rows[:5], "total_lag": sum(r["lag"] for r in lag_rows)})

# 诊断顺序：单分区 Lag -> 处理耗时 -> Rebalance -> Broker -> 下游 Sink。
# 只有实例吞吐已达上限且分区可分配时才扩容，否则扩容会放大重平衡。
```

```sql
-- Kafka消费积压排查与实时链路优化项目：结构贡献诊断（负责人 × 风险等级）
WITH scoped AS (
  SELECT
    requirement_id AS entity_key,
    event_id,
    event_date AS event_date,
    dimension_value AS primary_dim,
    stage_code AS secondary_dim,
    metric_value,
    status
  FROM fact_project_delivery
  WHERE event_date >= DATE '2026-05-01'
    AND event_date <  DATE '2026-08-01'
    AND subject_key = 'kafka'
    AND is_deleted = 0
),
dedup AS (
  SELECT *
  FROM (
    SELECT
      scoped.*,
      ROW_NUMBER() OVER (
        PARTITION BY event_id
        ORDER BY event_date DESC
      ) AS version_no
    FROM scoped
  ) ranked
  WHERE version_no = 1
),
grouped AS (
  SELECT
    primary_dim,
    secondary_dim,
    COUNT(DISTINCT entity_key) AS entity_cnt,
    COUNT(DISTINCT event_id) AS event_cnt,
    SUM(metric_value) AS metric_sum,
    AVG(metric_value) AS metric_avg
  FROM dedup
  WHERE status IN ('valid', 'completed')
  GROUP BY primary_dim, secondary_dim
),
contribution AS (
  SELECT
    *,
    metric_sum * 1.0 / NULLIF(SUM(metric_sum) OVER (PARTITION BY secondary_dim), 0) AS contribution_ratio,
    metric_avg - AVG(metric_avg) OVER (PARTITION BY secondary_dim) AS delta_to_peer
  FROM grouped
)
SELECT
  '投资回收周期' AS checked_metric,
  primary_dim,
  secondary_dim,
  entity_cnt,
  event_cnt,
  ROUND(metric_sum, 4) AS metric_sum,
  ROUND(contribution_ratio, 4) AS contribution_ratio,
  ROUND(delta_to_peer, 4) AS delta_to_peer
FROM contribution
ORDER BY ABS(delta_to_peer) DESC, entity_cnt DESC;
```

### 质量门禁与失败样本

```python
# Kafka消费积压排查与实时链路优化项目：发布前的数据质量与分层检查
import pandas as pd
import numpy as np
import hashlib

TOPIC = "Kafka消费积压排查与实时链路优化项目"
TOPIC_KEY = "kafka"
RULE_ID = "rule_project_uh2sws"
GUIDE_CHECKS = [
    "按 Topic-Partition 同时看生产速率、消费速率、Lag 和 commit rate",
    "检查消费者组成员、分配分区、Rebalance 原因与每次处理耗时",
    "把 Broker 磁盘、网络、ISR 与下游写入 P95 放在同一时间轴",
    "压测必须验证回放、幂等、死信和积压恢复后的追平时间"
]
RULES = {
    "required_columns": ["event_id", "requirement_id", "event_date", "status"],
    "duplicate_key": "event_id",
    "entity_key": "requirement_id",
    "time_field": "event_date",
    "max_duplicate_rate": 0.0008,
    "max_null_rate": 0.004,
    "allowed_status": {"valid", "completed", "cancelled", "refunded"},
}

def validate(frame: pd.DataFrame) -> pd.DataFrame:
    missing = [c for c in RULES["required_columns"] if c not in frame.columns]
    if missing:
        raise ValueError(f"{TOPIC}/{TOPIC_KEY} missing required columns: {missing}")

    checks = []
    checks.append(("duplicate_rate", frame[RULES["duplicate_key"]].duplicated().mean(), RULES["max_duplicate_rate"]))
    checks.append(("entity_null_rate", frame[RULES["entity_key"]].isna().mean(), RULES["max_null_rate"]))
    event_time = pd.to_datetime(frame[RULES["time_field"]], errors="coerce")
    checks.append(("invalid_time_rate", event_time.isna().mean(), 0.0))
    bad_status = (~frame["status"].isin(RULES["allowed_status"])).mean()
    checks.append(("invalid_status_rate", bad_status, 0.0))

    result = pd.DataFrame(checks, columns=["check_name", "value", "threshold"])
    result["rule_id"] = RULE_ID + "." + result["check_name"]
    result["topic"] = TOPIC
    result["topic_key"] = TOPIC_KEY
    result["passed"] = result["value"].le(result["threshold"])
    result["sample_fingerprint"] = hashlib.sha1(
        f"{TOPIC}|{RULE_ID}|{len(frame)}".encode("utf-8")
    ).hexdigest()[:12]
    return result

def diagnose(frame: pd.DataFrame) -> pd.DataFrame:
    return (
        frame.assign(stat_date=pd.to_datetime(frame[RULES["time_field"]]).dt.date)
        .groupby(["stat_date", "status"], dropna=False)
        .agg(
            event_cnt=(RULES["duplicate_key"], "nunique"),
            entity_cnt=(RULES["entity_key"], "nunique"),
        )
        .reset_index()
        .assign(topic=TOPIC, topic_key=TOPIC_KEY, rule_id=RULE_ID)
        .sort_values("stat_date")
    )

def release_gate(result: pd.DataFrame) -> None:
    failed = result.loc[~result["passed"], ["topic_key", "check_name", "value", "threshold"]]
    if not failed.empty:
        raise RuntimeError({"topic": TOPIC, "failed_checks": failed.to_dict("records")})

# 生产化时把结果写入质量平台，并携带批次、分区、代码版本、规则版本和处理工单。
```

WITH changed_columns AS (
  SELECT column_name
  FROM information_schema.columns
  WHERE table_schema = :source_schema
    AND table_name = :source_table
    AND last_altered_at >= :change_time

downstream_assets AS (

    asset_type,

    last_used_at

  d.asset_name,
  d.asset_type,
  d.owner_name,
  d.criticality,

FROM downstream_assets d
CROSS JOIN changed_columns c
GROUP BY d.asset_id, d.asset_name, d.asset_type, d.owner_name, d.criticality, d.last_used_at
ORDER BY

```python
from datetime import date
from hashlib import sha256

import pandas as pd

SUBJECT = "Kafka消费积压排查与实时链路优化项目"
BLOCKING_RULES = ["范围与验收标准明确","上下游依赖确认","回滚方案可执行"]


def validate_release(frame: pd.DataFrame, start: date, end: date) -> dict:
    scoped = frame.loc[
        frame["event_date"].between(pd.Timestamp(start), pd.Timestamp(end))
        & (frame["subject_key"] == "kafka")
    ].copy()
    checks = {
        "has_rows": len(scoped) > 0,
        "event_id_unique": not scoped["event_id"].duplicated().any(),
        "entity_key_complete": scoped["entity_key"].notna().all(),
        "event_time_present": scoped["event_time"].notna().all(),
        "scope_matches": scoped["subject_key"].nunique() == 1,
    }
    payload = f"{SUBJECT}|{start}|{end}|{len(scoped)}|{sorted(checks.items())}"
    return {
        "subject": SUBJECT,
        "rows": len(scoped),
        "checks": checks,
        "blocking_rules": BLOCKING_RULES,
        "release_allowed": all(checks.values()),
        "evidence_hash": sha256(payload.encode("utf-8")).hexdigest(),
    }
```
